   136→   136→// ── Helpers ─────────────────────────────────────────────────────────────────
   137→   137→function timeAgo(iso: string): string {
   138→   138→  if (!iso) return "-";
   139→   139→  const diff = Date.now() - new Date(iso).getTime();
   140→   140→  const sec = Math.floor(diff / 1000);
   141→   141→  if (sec < 60) return `${sec}s ago`;
   142→   142→  const min = Math.floor(sec / 60);
   143→   143→  if (min < 60) return `${min}m ago`;
   144→   144→  const hr = Math.floor(min / 60);
   145→   145→  if (hr < 24) return `${hr}h ago`;
   146→   146→  const day = Math.floor(hr / 24);
   147→   147→  if (day < 30) return `${day}d ago`;
   148→   148→  return new Date(iso).toLocaleDateString();
   149→   149→}
   150→   150→
   151→   151→function formatTime(iso: string): string {
   152→   152→  if (!iso) return "-";
   153→   153→  try {
   154→   154→    return new Date(iso).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit", second: "2-digit" });
   155→   155→  } catch {
   156→   156→    return iso;
   157→   157→  }
   158→   158→}

   160→   160→// Build an empty config object from a schema (selects default to first option)
   161→   161→function emptyConfigFor(schema: ConnectorSchema | undefined): Record<string, string> {
   162→   162→  const cfg: Record<string, string> = {};
   163→   163→  if (!schema) return cfg;
   164→   164→  for (const f of schema.configFields) {
   165→   165→    if (f.type === "select" && f.options && f.options.length > 0) {
   166→   166→      cfg[f.key] = f.options[0];
   167→   167→    } else {
   168→   168→      cfg[f.key] = "";
   169→   169→    }
   170→   170→  }
   171→   171→  return cfg;
   172→   172→}

   247→   247→function IntegrationsHub({ onSwitchToLog }: { onSwitchToLog: () => void }) {
   248→   248→  const { toast } = useToast();
   249→   249→  const [schemas, setSchemas] = useState<ConnectorSchema[]>([]);
   250→   250→  const [integrations, setIntegrations] = useState<Integration[]>([]);
   251→   251→  const [loading, setLoading] = useState(true);
   252→   252→  const [sendingTest, setSendingTest] = useState(false);
   253→   253→  const [lastForwardLog, setLastForwardLog] = useState<ForwardingLogEntry | null>(null);
   254→   254→  const [modalState, setModalState] = useState<{
   255→   255→    open: boolean;
   256→   256→    type: string | null;
   257→   257→    editingId: string | null;
   258→   258→  }>({ open: false, type: null, editingId: null });
   259→   259→
   260→   260→  const load = useCallback(async () => {
   261→   261→    try {
   262→   262→      const res = await fetch("/api/integrations?schemas=true");
   263→   263→      const data = await res.json();
   264→   264→      setSchemas((data.schemas as ConnectorSchema[]) || []);
   265→   265→      setIntegrations((data.integrations as Integration[]) || []);
   266→   266→    } catch {
   267→   267→      setSchemas([]);
   268→   268→      setIntegrations([]);
   269→   269→    } finally {
   270→   270→      setLoading(false);
   271→   271→    }
   272→   272→  }, []);
   273→   273→
   274→   274→  useEffect(() => {
   275→   275→    load();
   276→   276→    const id = setInterval(load, 30_000);
   277→   277→    return () => clearInterval(id);
   278→   278→  }, [load]);
   279→   279→
   280→   280→  const activeCount = integrations.filter((i) => i.isActive).length;
   281→   281→  const connectedCount = integrations.length;
   282→   282→
   283→   283→  const handleSendTestEvent = async () => {
   284→   284→    if (activeCount === 0) {
   285→   285→      toast({
   286→   286→        variant: "destructive",
   287→   287→        title: "No active integrations",
   288→   288→        description: "Configure and enable at least one integration before sending a test event.",
   289→   289→      });
   290→   290→      return;
   291→   291→    }
   292→   292→    setSendingTest(true);
   293→   293→    try {
   294→   294→      const res = await fetch("/api/integrations", {
   295→   295→        method: "PATCH",
   296→   296→        headers: { "Content-Type": "application/json" },
   297→   297→        body: JSON.stringify({ action: "forward_test_event" }),
   298→   298→      });
   299→   299→      const data = await res.json();
   300→   300→      if (!res.ok) throw new Error((data as { error?: string }).error || "Failed to forward test event");
   301→   301→      const log = data as ForwardingLogEntry;
   302→   302→      setLastForwardLog(log);
   303→   303→      toast({
   304→   304→        title: `Test event forwarded to ${log.totalSent} connector${log.totalSent === 1 ? "" : "s"}`,
   305→   305→        description: `${log.totalSuccess} succeeded, ${log.totalFailed} failed. See the Forwarding Log tab for details.`,
   306→   306→      });
   307→   307→    } catch (err) {
   308→   308→      toast({
   309→   309→        variant: "destructive",
   310→   310→        title: "Forwarding failed",
   311→   311→        description: err instanceof Error ? err.message : "unknown error",
   312→   312→      });
   313→   313→    } finally {
   314→   314→      setSendingTest(false);
   315→   315→    }
   316→   316→  };
   317→   317→
   318→   318→  const handleToggle = async (integration: Integration) => {
   319→   319→    const next = !integration.isActive;
   320→   320→    // Optimistic update
   321→   321→    setIntegrations((prev) =>
   322→   322→      prev.map((i) => (i.id === integration.id ? { ...i, isActive: next } : i))
   323→   323→    );
   324→   324→    try {
   325→   325→      const res = await fetch("/api/integrations", {
   326→   326→        method: "PATCH",
   327→   327→        headers: { "Content-Type": "application/json" },
   328→   328→        body: JSON.stringify({ action: "update", id: integration.id, isActive: next }),
   329→   329→      });
   330→   330→      if (!res.ok) throw new Error("Update failed");
   331→   331→      toast({
   332→   332→        title: next ? "Integration enabled" : "Integration disabled",
   333→   333→        description: `${integration.type} is now ${next ? "active" : "paused"}.`,
   334→   334→      });
   335→   335→    } catch (err) {
   336→   336→      // Rollback
   337→   337→      setIntegrations((prev) =>
   338→   338→        prev.map((i) => (i.id === integration.id ? { ...i, isActive: !next } : i))
   339→   339→      );
   340→   340→      toast({
   341→   341→        variant: "destructive",
   342→   342→        title: "Toggle failed",
   343→   343→        description: err instanceof Error ? err.message : "unknown error",
   344→   344→      });
   345→   345→    }
   346→   346→  };
   347→   347→
   348→   348→  const handleDelete = async (integration: Integration) => {
   349→   349→    try {
   350→   350→      const res = await fetch(`/api/integrations?id=${encodeURIComponent(integration.id)}`, {
   351→   351→        method: "DELETE",
   352→   352→      });
   353→   353→      if (!res.ok) {
   354→   354→        const data = await res.json().catch(() => ({}));
   355→   355→        throw new Error((data as { error?: string }).error || "Delete failed");
   356→   356→      }
   357→   357→      setIntegrations((prev) => prev.filter((i) => i.id !== integration.id));
   358→   358→      toast({
   359→   359→        title: "Integration removed",
   360→   360→        description: `${integration.type} connector deleted.`,
   361→   361→      });
   362→   362→    } catch (err) {
   363→   363→      toast({
   364→   364→        variant: "destructive",
   365→   365→        title: "Delete failed",
   366→   366→        description: err instanceof Error ? err.message : "unknown error",
   367→   367→      });
   368→   368→    }
   369→   369→  };
   370→   370→
   371→   371→  const openCreate = (type: string) => {
   372→   372→    setModalState({ open: true, type, editingId: null });
   373→   373→  };
   374→   374→
   375→   375→  const openEdit = (integration: Integration) => {
   376→   376→    setModalState({ open: true, type: integration.type, editingId: integration.id });
   377→   377→  };
   378→   378→
   379→   379→  const closeModal = () => {
   380→   380→    setModalState({ open: false, type: null, editingId: null });
   381→   381→  };
   382→   382→
   383→   383→  const onSaved = () => {
   384→   384→    closeModal();
   385→   385→    load();
   386→   386→  };
   387→   387→
   388→   388→  return (
   389→   389→    <div className="space-y-4">
   390→   390→      {/* Action bar */}
   391→   391→      <div className="holo-card-sharp hud-corners flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   392→   392→        <div className="flex flex-wrap items-center gap-3">
   393→   393→          <Stat label="Connected" value={connectedCount} color="emerald" />
   394→   394→          <span className="hidden h-8 w-px bg-zinc-800 sm:block" />
   395→   395→          <Stat label="Active" value={activeCount} color="cyan" />
   396→   396→          <span className="hidden h-8 w-px bg-zinc-800 sm:block" />
   397→   397→          <Stat label="Available" value={schemas.length} color="violet" />
   398→   398→        </div>
   399→   399→        <Button
   400→   400→          onClick={handleSendTestEvent}
   401→   401→          disabled={sendingTest || activeCount === 0}
   402→   402→          className="bg-emerald-600 text-white hover:bg-emerald-500"
   403→   403→        >
   404→   404→          {sendingTest ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   405→   405→          Send Test Event
   406→   406→        </Button>
   407→   407→      </div>
   408→   408→
   409→   409→      {/* Last forward result inline banner */}
   410→   410→      {lastForwardLog && (
   411→   411→        <motion.div
   412→   412→          initial={{ opacity: 0, y: -4 }}
   413→   413→          animate={{ opacity: 1, y: 0 }}
   414→   414→          className="holo-card-sharp hud-corners flex flex-col gap-2 p-4 sm:flex-row sm:items-center sm:justify-between"
   415→   415→        >
   416→   416→          <div className="flex items-center gap-3">
   417→   417→            <div className={`flex size-9 items-center justify-center rounded-md ${lastForwardLog.totalFailed === 0 ? "bg-emerald-500/15" : "bg-amber-500/15"}`}>
   418→   418→              {lastForwardLog.totalFailed === 0 ? (
   419→   419→                <CheckCircle2 className="size-5 text-emerald-400" />
   420→   420→              ) : (
   421→   421→                <AlertCircle className="size-5 text-amber-400" />
   422→   422→              )}
   423→   423→            </div>
   424→   424→            <div>
   425→   425→              <div className="text-sm font-semibold text-zinc-100">
   426→   426→                Last test event: {lastForwardLog.totalSuccess}/{lastForwardLog.totalSent} succeeded
   427→   427→              </div>
   428→   428→              <div className="font-mono text-[11px] text-zinc-500">
   429→   429→                {formatTime(lastForwardLog.timestamp)} - event: {lastForwardLog.eventType}
   430→   430→              </div>
   431→   431→            </div>
   432→   432→          </div>
   433→   433→          <Button
   434→   434→            variant="outline"

   647→   647→function ConfigModal({
   648→   648→  type,
   649→   649→  editingId,
   650→   650→  schemas,
   651→   651→  integrations,
   652→   652→  onClose,
   653→   653→  onSaved,
   654→   654→}: {
   655→   655→  type: string;
   656→   656→  editingId: string | null;
   657→   657→  schemas: ConnectorSchema[];
   658→   658→  integrations: Integration[];
   659→   659→  onClose: () => void;
   660→   660→  onSaved: () => void;
   661→   661→}) {
   662→   662→  const { toast } = useToast();
   663→   663→  const schema = schemas.find((s) => s.type === type);
   664→   664→  const meta = CONNECTOR_META[type] || { icon: Plug, color: "zinc" as AccentColor };
   665→   665→  const Icon = meta.icon;
   666→   666→  const color = meta.color;
   667→   667→
   668→   668→  const existing = editingId ? integrations.find((i) => i.id === editingId) : undefined;
   669→   669→  const isEditing = !!existing;
   670→   670→
   671→   671→  const [config, setConfig] = useState<Record<string, string>>(() => {
   672→   672→    const base = emptyConfigFor(schema);
   673→   673→    if (existing && existing.config) {
   674→   674→      for (const k of Object.keys(base)) {
   675→   675→        const v = (existing.config as Record<string, unknown>)[k];
   676→   676→        if (typeof v === "string" || typeof v === "number") base[k] = String(v);
   677→   677→      }
   678→   678→    }
   679→   679→    return base;
   680→   680→  });
   681→   681→  const [testing, setTesting] = useState(false);
   682→   682→  const [saving, setSaving] = useState(false);
   683→   683→  const [testResult, setTestResult] = useState<ConnectorResult | null>(null);
   684→   684→
   685→   685→  // ESC to close + body scroll lock
   686→   686→  useEffect(() => {
   687→   687→    const onKey = (e: KeyboardEvent) => {
   688→   688→      if (e.key === "Escape") onClose();
   689→   689→    };
   690→   690→    document.addEventListener("keydown", onKey);
   691→   691→    const prevOverflow = document.body.style.overflow;
   692→   692→    document.body.style.overflow = "hidden";
   693→   693→    return () => {
   694→   694→      document.removeEventListener("keydown", onKey);
   695→   695→      document.body.style.overflow = prevOverflow;
   696→   696→    };
   697→   697→  }, [onClose]);
   698→   698→
   699→   699→  const updateField = (key: string, value: string) => {
   700→   700→    setConfig((prev) => ({ ...prev, [key]: value }));
   701→   701→    setTestResult(null);
   702→   702→  };
   703→   703→
   704→   704→  const validate = (): string | null => {
   705→   705→    if (!schema) return "Schema not found";
   706→   706→    for (const f of schema.configFields) {
   707→   707→      if (f.required && !config[f.key]?.trim()) {
   708→   708→        return `${f.label} is required`;
   709→   709→      }
   710→   710→    }
   711→   711→    return null;
   712→   712→  };
   713→   713→
   714→   714→  const handleTest = async () => {
   715→   715→    const err = validate();
   716→   716→    if (err) {
   717→   717→      toast({ variant: "destructive", title: "Missing field", description: err });
   718→   718→      return;
   719→   719→    }
   720→   720→    setTesting(true);
   721→   721→    setTestResult(null);
   722→   722→    try {
   723→   723→      const res = await fetch("/api/integrations", {
   724→   724→        method: "PATCH",
   725→   725→        headers: { "Content-Type": "application/json" },
   726→   726→        body: JSON.stringify({ action: "test", type, config }),
   727→   727→      });
   728→   728→      const data = await res.json();
   729→   729→      if (!res.ok) {
   730→   730→        throw new Error((data as { error?: string }).error || "Test failed");
   731→   731→      }
   732→   732→      const result = data as ConnectorResult;
   733→   733→      setTestResult(result);
   734→   734→      if (result.success) {
   735→   735→        toast({
   736→   736→          title: "Connection test passed",
   737→   737→          description: result.message || `${type} responded successfully.`,
   738→   738→        });
   739→   739→      } else {
   740→   740→        toast({
   741→   741→          variant: "destructive",
   742→   742→          title: "Connection test failed",
   743→   743→          description: result.message || `${type} did not respond correctly.`,
   744→   744→        });
   745→   745→      }
   746→   746→    } catch (err) {
   747→   747→      const msg = err instanceof Error ? err.message : "unknown error";
   748→   748→      setTestResult({
   749→   749→        integrationId: "test",
   750→   750→        type,
   751→   751→        success: false,
   752→   752→        message: msg,
   753→   753→        timestamp: new Date().toISOString(),
   754→   754→      });
   755→   755→      toast({
   756→   756→        variant: "destructive",
   757→   757→        title: "Connection test failed",
   758→   758→        description: msg,
   759→   759→      });
   760→   760→    } finally {
   761→   761→      setTesting(false);
   762→   762→    }
   763→   763→  };
   764→   764→
   765→   765→  const handleSave = async () => {
   766→   766→    const err = validate();
   767→   767→    if (err) {
   768→   768→      toast({ variant: "destructive", title: "Missing field", description: err });
   769→   769→      return;
   770→   770→    }
   771→   771→    setSaving(true);
   772→   772→    try {
   773→   773→      if (isEditing && existing) {
   774→   774→        const res = await fetch("/api/integrations", {
   775→   775→          method: "PATCH",
   776→   776→          headers: { "Content-Type": "application/json" },
   777→   777→          body: JSON.stringify({ action: "update", id: existing.id, config }),
   778→   778→        });
   779→   779→        if (!res.ok) {
   780→   780→          const data = await res.json().catch(() => ({}));
   781→   781→          throw new Error((data as { error?: string }).error || "Update failed");
   782→   782→        }
   783→   783→        toast({
   784→   784→          title: "Integration updated",
   785→   785→          description: `${schema?.name || type} configuration saved.`,
   786→   786→        });
   787→   787→      } else {
   788→   788→        const res = await fetch("/api/integrations", {
   789→   789→          method: "POST",
   790→   790→          headers: { "Content-Type": "application/json" },
   791→   791→          body: JSON.stringify({ type, config }),
   792→   792→        });
   793→   793→        const data = await res.json();
   794→   794→        if (!res.ok) {
   795→   795→          throw new Error((data as { error?: string }).error || "Save failed");
   796→   796→        }
   797→   797→        toast({
   798→   798→          title: "Integration connected",
   799→   799→          description: `${schema?.name || type} is now configured and active.`,
   800→   800→        });
   801→   801→      }
   802→   802→      onSaved();
   803→   803→    } catch (err) {
   804→   804→      toast({
   805→   805→        variant: "destructive",
   806→   806→        title: "Save failed",
   807→   807→        description: err instanceof Error ? err.message : "unknown error",
   808→   808→      });
   809→   809→    } finally {
   810→   810→      setSaving(false);
   811→   811→    }
   812→   812→  };
   813→   813→
   814→   814→  if (!schema) {
   815→   815→    return (
   816→   816→      <div
   817→   817→        className="fixed inset-0 z-50 flex items-center justify-center bg-black/70 p-4"
   818→   818→        onClick={onClose}
   819→   819→      >
   820→   820→        <motion.div
   821→   821→          initial={{ opacity: 0, scale: 0.95 }}
   822→   822→          animate={{ opacity: 1, scale: 1 }}
   823→   823→          onClick={(e) => e.stopPropagation()}
   824→   824→          className="holo-card-sharp hud-corners w-full max-w-md rounded-lg p-6 text-center"
   825→   825→        >
   826→   826→          <AlertCircle className="mx-auto size-8 text-amber-400" />
   827→   827→          <p className="mt-2 text-sm text-zinc-300">Schema not found for type: {type}</p>
   828→   828→          <Button variant="outline" onClick={onClose} className="mt-4 border-zinc-700 bg-zinc-900 text-zinc-300 hover:bg-zinc-800">
   829→   829→            Close
   830→   830→          </Button>
   831→   831→        </motion.div>
   832→   832→      </div>
   833→   833→    );
   834→   834→  }
   835→   835→
   836→   836→  return (
   837→   837→    <div
   838→   838→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-4 backdrop-blur-sm"
   839→   839→      onClick={onClose}
   840→   840→      role="dialog"
   841→   841→      aria-modal="true"
   842→   842→      aria-label={`Configure ${schema.name}`}
   843→   843→    >
   844→   844→      <motion.div
   845→   845→        initial={{ opacity: 0, scale: 0.95, y: 8 }}
   846→   846→        animate={{ opacity: 1, scale: 1, y: 0 }}
   847→   847→        exit={{ opacity: 0, scale: 0.95, y: 8 }}
   848→   848→        transition={{ duration: 0.18 }}
   849→   849→        onClick={(e) => e.stopPropagation()}
   850→   850→        className="holo-card-sharp hud-corners custom-scrollbar w-full max-w-lg overflow-y-auto rounded-lg p-6"
   851→   851→        style={{ maxHeight: "90vh" }}
   852→   852→      >
   853→   853→        {/* Header */}
   854→   854→        <div className="mb-5 flex items-start justify-between gap-3">
   855→   855→          <div className="flex items-center gap-3">
   856→   856→            <div className={`flex size-10 items-center justify-center rounded-md ${COLOR_BG[color]} ${COLOR_BORDER[color]} border`}>
   857→   857→              <Icon className={`size-5 ${COLOR_TEXT[color]}`} />
   858→   858→            </div>
   859→   859→            <div>
   860→   860→              <h2 className="text-lg font-bold text-zinc-50">
   861→   861→                {isEditing ? "Configure" : "Connect"} {schema.name}
   862→   862→              </h2>
   863→   863→              <p className="text-xs text-zinc-500">{schema.description}</p>
   864→   864→            </div>
   865→   865→          </div>
   866→   866→          <Button
   867→   867→            size="icon"
   868→   868→            variant="ghost"
   869→   869→            onClick={onClose}
   870→   870→            className="shrink-0 text-zinc-500 hover:bg-zinc-800 hover:text-zinc-200"
   871→   871→            aria-label="Close modal"
   872→   872→          >
   873→   873→            <X className="size-4" />
   874→   874→          </Button>
   875→   875→        </div>
   876→   876→
   877→   877→        {/* Form fields */}
   878→   878→        <div className="space-y-4">
   879→   879→          {schema.configFields.map((field) => (
   880→   880→            <ConfigFieldRow
   881→   881→              key={field.key}
   882→   882→              field={field}
   883→   883→              value={config[field.key] ?? ""}

   889→   889→        {/* Test result */}

   891→   891→          {testResult && (
   892→   892→            <motion.div

   894→   894→              animate={{ opacity: 1, height: "auto" }}
   895→   895→              exit={{ opacity: 0, height: 0 }}
   896→   896→              className={`mt-4 flex items-start gap-3 rounded-md border p-3 ${
