   199→function route(sx,sy,dir,maxLen){
   200→  let x=sx,y=sy,d=dir;
   201→  const pts=[[x,y]];
   202→  let run=2+(rnd()*5|0), blocked=0;
   203→  const total=maxLen??(14+(rnd()*30|0));
   204→  for(let s=0;s<total;s++){
   205→    const [dx,dy]=DIRS[d];
   206→    const nx=x+dx, ny=y+dy;
   207→    if(nx<1||ny<1||nx>=cols-1||ny>=rows-1||occ.has(key(nx,ny))){
   208→      d=(d+(rnd()<.5?1:7))%8;
   209→      if(++blocked>2)break;
   210→      continue;
   211→    }
   212→    blocked=0;x=nx;y=ny;
   213→    occ.add(key(x,y));pts.push([x,y]);
   214→    if(--run<=0){
   215→      if(rnd()<.7)d=(d+(rnd()<.5?1:7))%8;   // 45° turns — the PCB look
   216→      run=2+(rnd()*6|0);
   217→    }
   218→  }
   219→  if(pts.length<5)return null;
   220→  // simplify collinear runs, convert to px, cumulative lengths
   221→  const px=[],keep=[pts[0]];
   222→  for(let i=1;i<pts.length-1;i++){
   223→    const a=keep[keep.length-1],b=pts[i],c=pts[i+1];
   224→    if((b[0]-a[0])*(c[1]-b[1])!==(b[1]-a[1])*(c[0]-b[0]))keep.push(b);
   225→  }
   226→  keep.push(pts[pts.length-1]);
   227→  keep.forEach(p=>px.push([p[0]*cell+cell/2,p[1]*cell+cell/2]));
   228→  const cum=[0];
   229→  for(let i=1;i<px.length;i++)
   230→    cum.push(cum[i-1]+Math.hypot(px[i][0]-px[i-1][0],px[i][1]-px[i-1][1]));
   231→  return {pts:px,cum,len:cum[cum.length-1],end:pts[pts.length-1],start:pts[0]};
   232→}

   234→function buildBoard(){
   235→  cell=Math.max(16,Math.round(Math.min(W,H)/56));
   236→  cols=Math.ceil(W/cell);rows=Math.ceil(H/cell);
   237→  occ=new Set();traces=[];comps=[];pulses=[];
   238→
   239→  /* ---- the center chip: horizontal, carries the agent's name ---- */
   240→  const chW=Math.round(Math.min(W*.235,620*DPR)), chH=Math.round(chW*.30);
   241→  chip={x:W*.5-chW/2, y:H*.5-chH/2, w:chW, h:chH, glow:0, glowIn:0};
   242→  const cx0=Math.floor(chip.x/cell)-1, cy0=Math.floor(chip.y/cell)-1;
   243→  const cx1=Math.ceil((chip.x+chW)/cell)+1, cy1=Math.ceil((chip.y+chH)/cell)+1;
   244→  for(let gx=cx0;gx<=cx1;gx++)for(let gy=cy0;gy<=cy1;gy++)occ.add(key(gx,gy));
   245→
   246→  /* ---- components ---- */
   247→  const place=(wc,hc)=>{
   248→    for(let tr=0;tr<60;tr++){
   249→      const gx=1+(rnd()*(cols-wc-2)|0), gy=1+(rnd()*(rows-hc-2)|0);
   250→      let free=true;
   251→      for(let a=gx-1;a<=gx+wc&&free;a++)for(let b=gy-1;b<=gy+hc&&free;b++)
   252→        if(occ.has(key(a,b)))free=false;
   253→      if(!free)continue;
   254→      for(let a=gx;a<gx+wc;a++)for(let b=gy;b<gy+hc;b++)occ.add(key(a,b));
   255→      return {gx,gy};
   256→    }
   257→    return null;
   258→  };
   259→  const NC=34;
   260→  for(let i=0;i<NC;i++){
   261→    const roll=rnd();
   262→    const type=roll<.42?"ic":roll<.78?"res":"cap";
   263→    const wc=type==="ic"?4+(rnd()*3|0):type==="res"?1:2;
   264→    const hc=type==="ic"?3+(rnd()*2|0):type==="res"?3+(rnd()*3|0):2;
   265→    const spot=place(wc,hc);
   266→    if(!spot)continue;
   267→    comps.push({type,gx:spot.gx,gy:spot.gy,wc,hc,
   268→      x:spot.gx*cell,y:spot.gy*cell,w:wc*cell,h:hc*cell,
   269→      glow:0,amber:rnd()<.18,out:[]});
   270→  }

   272→  /* ---- traces from the chip pins (the heart pumps outward) ---- */
   273→  const pinsPerSide=Math.max(6,Math.round(chH/cell)+2);
   274→  chip.pins=[];

   276→    const n=side<2?pinsPerSide:6;
   277→    for(let i=0;i<n;i++){
   278→      let gx,gy,dir;
   279→      if(side===0){gx=cx0;gy=cy0+2+(rnd()*(cy1-cy0-3)|0);dir=4}
   280→      else if(side===1){gx=cx1;gy=cy0+2+(rnd()*(cy1-cy0-3)|0);dir=0}
   281→      else if(side===2){gx=cx0+2+(rnd()*(cx1-cx0-3)|0);gy=cy0;dir=6}
   282→      else{gx=cx0+2+(rnd()*(cx1-cx0-3)|0);gy=cy1;dir=2}
   283→      const t=route(gx,gy,dir);
   284→      if(t){t.chip=true;t.amber=rnd()<.15;traces.push(t);
   285→        chip.pins.push({side,gx,gy});}
   286→    }
   287→  };
   288→  addPinTraces(0);addPinTraces(1);addPinTraces(2);addPinTraces(3);

   304→      const dy=Math.max(c.gy-t.end[1],0,t.end[1]-(c.gy+c.hc-1));
   305→      const d=dx*dx+dy*dy;

   312→      const dy=Math.max(c.gy-t.start[1],0,t.start[1]-(c.gy+c.hc-1));
   313→      if(dx*dx+dy*dy<=9)c.out.push(ti);

   350→/* ---------------- the board, drawn as vectors ----------------

   353→   the macro shots stay infinitely crisp. sf scales the shadow params
   354→   (canvas shadows live in device space, so they must scale with the zoom). */
   355→function drawWorld(g,v,sf){
   356→  const[vx0,vy0,vx1,vy1]=v;
   357→  const hit=(x0,y0,x1,y1)=>x1>=vx0&&x0<=vx1&&y1>=vy0&&y0<=vy1;
   358→  // substrate
   359→  const grd=g.createRadialGradient(W/2,H/2,0,W/2,H/2,Math.max(W,H)*.72);
   360→  grd.addColorStop(0,"#07160e");grd.addColorStop(.55,"#04100a");grd.addColorStop(1,"#020705");
   361→  g.fillStyle=grd;g.fillRect(vx0,vy0,vx1-vx0,vy1-vy0);
   362→  // FR4 fiberglass weave — soft-light composite so the darks stay moody
   363→  g.save();
   364→  g.globalCompositeOperation="soft-light";
   365→  const wv=Math.max(5,Math.round(cell*.3));
   366→  g.strokeStyle="rgba(190,255,220,.10)";g.lineWidth=1;
   367→  g.beginPath();
   368→  for(let x=Math.floor(vx0/wv)*wv;x<=vx1;x+=wv){g.moveTo(x,vy0);g.lineTo(x,vy1)}
   369→  for(let y=Math.floor(vy0/wv)*wv;y<=vy1;y+=wv){g.moveTo(vx0,y);g.lineTo(vx1,y)}
   370→  g.stroke();
   371→  g.restore();
   372→  // drill-grid dots
   373→  g.fillStyle="rgba(61,220,132,.045)";

   585→let cam={x:0,y:0,z:1}, cine=null, cineFade=0, camZ=1;
   586→function startCine(){
   587→  if(!traces.length||cine)return;
   588→  const pAt=(t,d)=>{let i=1;while(i<t.cum.length&&t.cum[i]<d)i++;i=Math.min(i,t.cum.length-1);
   589→    const seg=t.cum[i]-t.cum[i-1]||1,f=(d-t.cum[i-1])/seg;
   590→    return[t.pts[i-1][0]+(t.pts[i][0]-t.pts[i-1][0])*f,t.pts[i-1][1]+(t.pts[i][1]-t.pts[i-1][1])*f]};
   591→  // clamp a waypoint so the view mostly stays on the board
   592→  const cl=(x,y,z)=>({x:Math.max(W/z/2*.7,Math.min(W-W/z/2*.7,x)),
   593→                      y:Math.max(H/z/2*.7,Math.min(H-H/z/2*.7,y)),z});
   594→  const pick=a=>a[(rnd()*a.length)|0];
   595→  const dur=()=>4800+(rnd()*1800|0);
   596→  const ccx=chip.x+chip.w/2, ccy=chip.y+chip.h/2;
   597→  const byLen=[...traces].sort((a,b)=>b.len-a.len);
   598→  const ics=comps.filter(c=>c.type==="ic").sort((a,b)=>b.w*b.h-a.w*a.h);
   599→  const caps=comps.filter(c=>c.type==="cap");
   600→  /* the SHOT POOL — every run deals a fresh hand. Each shot travels at
   601→     constant speed with a randomized subject and direction. */
   602→  const POOL=[
   603→    ()=>{ // the trace highway — ride one of the five longest traces
   604→      const t=pick(byLen.slice(0,5)), rev=rnd()<.5;
   605→      const[ax,ay]=pAt(t,t.len*(rev?.8:.15)),[bx,by]=pAt(t,t.len*(rev?.15:.8));
   606→      return{d:dur(),f:cl(ax,ay,3.0),t:cl(bx,by,2.7)};
   607→    },
   608→    ()=>{ // the label slides through frame (the ONE chip shot)
   609→      const rev=rnd()<.5;
   610→      return{d:dur(),f:cl(chip.x+chip.w*(rev?.95:.05),ccy,2.9),
   611→             t:cl(chip.x+chip.w*(rev?.05:.95),ccy,2.6),chipShot:true};
   612→    },
   613→    ()=>{ // sweep across an IC quadrant
   614→      const ic=pick(ics.slice(0,3));if(!ic)return null;
   615→      const sx=rnd()<.5?1:-1, sy=rnd()<.5?1:-1;
   616→      const icx=ic.x+ic.w/2, icy=ic.y+ic.h/2;
   617→      return{d:dur(),f:cl(icx-sx*W*.16,icy-sy*H*.10,2.6),t:cl(icx+sx*W*.16,icy+sy*H*.10,2.5)};
   618→    },
   619→    ()=>{ // drift past a capacitor dome
   620→      const cp=pick(caps);if(!cp)return null;
   621→      const sx=rnd()<.5?1:-1;
   622→      const cpx=cp.x+cp.w/2, cpy=cp.y+cp.h/2;
   623→      return{d:dur(),f:cl(cpx-sx*W*.14,cpy+H*.06,3.1),t:cl(cpx+sx*W*.14,cpy-H*.06,2.9)};
   624→    },
   625→    ()=>{ // grand diagonal — corner region sweeping past center
   626→      const fx=rnd()<.5?W*.18:W*.82, fy=rnd()<.5?H*.18:H*.82;
   627→      const tx=fx<W/2?W*.66:W*.34, ty=fy<H/2?H*.66:H*.34;
   628→      return{d:dur(),f:cl(fx,fy,2.7),t:cl(tx,ty,2.5)};
   629→    },
   630→    ()=>{ // the edge run — skim along the top or bottom of the board
   631→      const y=(rnd()<.5?H*.14:H*.86), rev=rnd()<.5;
   632→      return{d:dur(),f:cl(rev?W*.8:W*.2,y,2.9),t:cl(rev?W*.2:W*.8,y,2.7)};
   633→    },
   634→    ()=>{ // vertical descent — truck down (or up) a random column
   635→      const x=W*(.25+rnd()*.5), rev=rnd()<.5;
   636→      return{d:dur(),f:cl(x,rev?H*.82:H*.18,2.8),t:cl(x,rev?H*.18:H*.82,2.6)};
   637→    },
   638→    ()=>{ // dolly push toward an IC — travel + zoom, never static
   639→      const ic=pick(ics);if(!ic)return null;
   640→      const icx=ic.x+ic.w/2, icy=ic.y+ic.h/2, sx=rnd()<.5?1:-1;
   641→      return{d:dur(),f:cl(icx-sx*W*.10,icy+H*.05,2.4),t:cl(icx+sx*W*.05,icy-H*.03,3.1)};
   642→    },
   643→  ];
   644→  // deal 4 shots; the chip shot may land anywhere EXCEPT last — it would
   645→  // jump-cut into the reveal, which also opens on the chip
   646→  const shots=[];
   647→  for(const gen of [...POOL].sort(()=>rnd()-.5)){
   648→    if(shots.length>=4)break;
   649→    const s=gen();if(s)shots.push(s);
   650→  }
   651→  // ONE screen-speed for the whole deck (one pan read faster) — duration
   652→  // derives from travel distance × zoom, so every drift feels the same to
   653→  // the eye, with just a whisper of variation
   654→  shots.forEach(s=>{
   655→    const zAvg=(s.f.z+s.t.z)/2;
   656→    const distS=Math.hypot(s.t.x-s.f.x,s.t.y-s.f.y)*zAvg;
   657→    s.d=Math.max(4300,Math.min(7800,(distS/.33)*(.92+rnd()*.16)));
   658→  });
   659→  const li=shots.findIndex(s=>s.chipShot);
   660→  if(li===shots.length-1&&shots.length>1){
   661→    const j=(rnd()*(shots.length-1))|0;
   662→    [shots[li],shots[j]]=[shots[j],shots[li]];
   663→  }
   664→  const segs=[...shots,
   665→    {d:6600,f:{x:ccx,y:ccy,z:2.3},t:{x:W/2,y:H/2,z:1},out:true}];  // the reveal, always the finale
   666→  cine={t:0,segs,total:segs.reduce((a,s)=>a+s.d,0)};
   667→  cv.style.transform="perspective(1100px) rotateX(16deg) scale(1.24)";  // tilted from frame one
   668→  document.body.classList.add("cine");
   669→}

   671→function updateCine(dt){
   672→  cine.t+=dt;let t=cine.t;
   673→  for(const s of cine.segs){

   675→      const p=t/s.d;
