   764→// ── Helper: determine stage status from client data ────────────────────────
   765→function getStageStatus(client: ClientSummary, stageKey: string): "pending" | "in-progress" | "completed" {
   766→  const s = client.stats;
   767→  switch (stageKey) {
   768→    case "onboarding":
   769→      return (s.codebases > 0 || s.targets > 0) && client.authorized ? "completed" : (s.codebases > 0 || s.targets > 0) ? "in-progress" : "pending";
   770→    case "scanning":
   771→      return s.patches > 0 || s.findings > 0 ? "completed" : client.status === "scanning" ? "in-progress" : "pending";
   772→    case "testing":
   773→      return s.findings > 0 ? "completed" : client.status === "testing" ? "in-progress" : "pending";
   774→    case "patching":
   775→      return s.approved_patches > 0 ? "completed" : s.pending_patches > 0 ? "in-progress" : "pending";
   776→    case "verifying":
   777→      return s.approved_patches > 0 ? "completed" : "pending";
   778→    case "defending":
   779→      return client.status === "defending" || client.status === "compliant" ? "completed" : "pending";
   780→    case "compliant":
   781→      return client.status === "compliant" ? "completed" : "pending";
   782→    default:
   783→      return "pending";
   784→  }
   785→}
