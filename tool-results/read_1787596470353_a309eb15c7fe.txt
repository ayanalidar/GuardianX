   795→      const bScore = (b.category === post.category ? 1 : 0);
   796→      if (aScore !== bScore) return bScore - aScore;

   798→    })
   799→    .slice(0, limit);

   802→/**
   803→ * Parse the markdown-ish content into a list of headings for the table of
   804→ * contents. Returns entries with { id, text, level }.
   805→ */
   806→export function getTableOfContents(content: string): { id: string; text: string; level: number }[] {
   807→  const lines = content.split("\n");
   808→  const toc: { id: string; text: string; level: number }[] = [];
   809→  for (const line of lines) {
   810→    const m = line.match(/^(#{2,3})\s+(.+)$/);
   811→    if (m) {
   812→      const level = m[1].length;
   813→      const text = m[2].trim();
   814→      const id = text
   815→        .toLowerCase()
   816→        .replace(/[^\w\s-]/g, "")
   817→        .replace(/\s+/g, "-");
   818→      toc.push({ id, text, level });
   819→    }
   820→  }
   821→  return toc;
   822→}
