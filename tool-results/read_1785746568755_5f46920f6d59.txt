     1→"use client";
     2→
     3→import { useState, useEffect, useCallback } from "react";
     4→import { motion } from "framer-motion";
     5→import { Button } from "@/components/ui/button";
     6→import { Badge } from "@/components/ui/badge";
     7→import { useToast } from "@/hooks/use-toast";
     8→import {
     9→  ArrowLeft, Building2, Globe, GitBranch, Shield, Loader2, CheckCircle2,
    10→  Circle, AlertCircle, Phone, Mail, FileText, Crosshair, Bug, ShieldCheck,
    11→  Swords, Heart, Gavel, Plus, Boxes, Target, ExternalLink, X, Trash2, Rocket,
    12→} from "lucide-react";
    13→import { sentinelApi } from "@/lib/sentinel/api";
    14→import { pushTerminalLine } from "./live-exploit-terminal";
    15→
    16→interface Stage {
    17→  id: number;
    18→  key: string;
    19→  label: string;
    20→  desc: string;
    21→  status: "pending" | "in-progress" | "completed";
    22→  metrics: Record<string, unknown>;
    23→}
    24→
    25→interface ClientDetailData {
    26→  client: { id: string; name: string; status: string; authorized: boolean };
    27→  stages: Stage[];
    28→  progress: number;
    29→  current_stage: string;
    30→  summary: Record<string, number>;
    31→}
    32→
    33→interface ClientFullDetail {
    34→  id: string;
    35→  name: string;
    36→  description: string | null;
    37→  contact_name: string | null;
    38→  contact_email: string | null;
    39→  contact_phone: string | null;
    40→  target_url: string | null;
    41→  repo_url: string | null;
    42→  scope: string | null;
    43→  authorized: boolean;
    44→  frameworks: string[];
    45→  status: string;
    46→  codebases: { id: string; name: string; language: string; description: string | null }[];
    47→  targets: { id: string; name: string; base_url: string; authorized: boolean }[];
    48→}
    49→
    50→const STAGE_ICONS: Record<string, typeof Circle> = {
    51→  onboarding: Building2,
    52→  scanning: Bug,
    53→  testing: Crosshair,
    54→  patching: ShieldCheck,
    55→  verifying: Swords,
    56→  defending: Shield,
    57→  compliant: Gavel,
    58→};
    59→
    60→const STAGE_COLORS: Record<string, { text: string; bg: string; border: string; neon: string }> = {
    61→  onboarding: { text: "text-emerald-400", bg: "bg-emerald-500/10", border: "border-emerald-500/40", neon: "neon-emerald" },
    62→  scanning:   { text: "text-cyan-400",    bg: "bg-cyan-500/10",    border: "border-cyan-500/40",    neon: "neon-cyan" },
    63→  testing:    { text: "text-amber-400",   bg: "bg-amber-500/10",   border: "border-amber-500/40",   neon: "neon-amber" },
    64→  patching:   { text: "text-violet-400",  bg: "bg-violet-500/10",  border: "border-violet-500/40",  neon: "neon-violet" },
    65→  verifying:  { text: "text-sky-400",     bg: "bg-sky-500/10",     border: "border-sky-500/40",     neon: "neon-sky" },
    66→  defending:  { text: "text-rose-400",    bg: "bg-rose-500/10",    border: "border-rose-500/40",    neon: "neon-rose" },
    67→  compliant:  { text: "text-emerald-400", bg: "bg-emerald-500/10", border: "border-emerald-500/40", neon: "neon-emerald" },
    68→};
    69→
    70→interface ClientDetailProps {
    71→  clientId: string;
    72→  onBack: () => void;
    73→  onNavigate: (tab: string) => void;
    74→}
    75→
    76→export function ClientDetail({ clientId, onBack }: ClientDetailProps) {
    77→  const { toast } = useToast();
    78→  const [detail, setDetail] = useState<ClientFullDetail | null>(null);
    79→  const [pipeline, setPipeline] = useState<ClientDetailData | null>(null);
    80→  const [loading, setLoading] = useState(true);
    81→  const [authorizing, setAuthorizing] = useState(false);
    82→  const [showAddCodebase, setShowAddCodebase] = useState(false);
    83→  const [showAddTarget, setShowAddTarget] = useState(false);
    84→  const [vaptRunning, setVaptRunning] = useState(false);
    85→  const [vaptResult, setVaptResult] = useState<any>(null);
    86→  const [deleting, setDeleting] = useState(false);
    87→
    88→  // Add codebase form state
    89→  const [cbName, setCbName] = useState("");
    90→  const [cbSource, setCbSource] = useState("");
    91→  const [cbDesc, setCbDesc] = useState("");
    92→  const [cbSaving, setCbSaving] = useState(false);
    93→
    94→  // Add target form state
    95→  const [tgtName, setTgtName] = useState("");
    96→  const [tgtUrl, setTgtUrl] = useState("");
    97→  const [tgtAuth, setTgtAuth] = useState(false);
    98→  const [tgtSaving, setTgtSaving] = useState(false);
    99→
   100→  const load = useCallback(async () => {
   101→    setLoading(true);
   102→    try {
   103→      const [dRes, pRes] = await Promise.all([
   104→        fetch(`/api/clients/${clientId}`),
   105→        fetch(`/api/clients/${clientId}/pipeline`),
   106→      ]);
   107→      const d = await dRes.json();
   108→      const p = await pRes.json();
   109→      if (!d.error) setDetail(d);
   110→      if (!p.error) setPipeline(p);
   111→    } catch {
   112→      // ignore
   113→    } finally {
   114→      setLoading(false);
   115→    }
   116→  }, [clientId]);
   117→
   118→  useEffect(() => {
   119→    load();
   120→  }, [load]);
   121→
   122→  const handleAddCodebase = async () => {
   123→    if (!cbName.trim() || !cbSource.trim()) return;
   124→    setCbSaving(true);
   125→    try {
   126→      await sentinelApi.createCodebase({
   127→        name: cbName.trim(),
   128→        sourceCode: cbSource,
   129→        description: cbDesc.trim() || undefined,
   130→        clientId: clientId,
   131→      });
   132→      toast({ title: "Codebase added!", description: cbName.trim() });
   133→      setCbName(""); setCbSource(""); setCbDesc("");
   134→      setShowAddCodebase(false);
   135→      load();
   136→    } catch (err) {
   137→      toast({ variant: "destructive", title: "Failed", description: err instanceof Error ? err.message : "unknown" });
   138→    } finally {
   139→      setCbSaving(false);
   140→    }
   141→  };
   142→
   143→  const handleAddTarget = async () => {
   144→    if (!tgtName.trim() || !tgtUrl.trim()) return;
   145→    setTgtSaving(true);
   146→    try {
   147→      await sentinelApi.addTarget({
   148→        name: tgtName.trim(),
   149→        baseUrl: tgtUrl.trim(),
   150→        authorized: tgtAuth,
   151→        clientId: clientId,
   152→      });
   153→      toast({ title: "Target added!", description: tgtName.trim() });
   154→      setTgtName(""); setTgtUrl(""); setTgtAuth(false);
   155→      setShowAddTarget(false);
   156→      load();
   157→    } catch (err) {
   158→      toast({ variant: "destructive", title: "Failed", description: err instanceof Error ? err.message : "unknown" });
   159→    } finally {
   160→      setTgtSaving(false);
   161→    }
   162→  };
   163→
   164→  const handleFullVapt = async () => {
   165→    setVaptRunning(true);
   166→    setVaptResult(null);
   167→    pushTerminalLine({ text: `$ guardianx full-vapt --client "${detail?.name}" --clientId ${clientId}`, type: "cmd" });
   168→    pushTerminalLine({ text: "[*] Initiating 7-stage pipeline: discover → recon → scan → test → patch → verify → defend", type: "out" });
   169→    try {
   170→      const res = await fetch("/api/full-vapt", {
   171→        method: "POST",
   172→        headers: { "Content-Type": "application/json" },
   173→        body: JSON.stringify({ clientId }),
   174→      });
   175→      const data = await res.json();
   176→      setVaptResult(data);
   177→      if (res.ok) {
   178→        pushTerminalLine({ text: `[+] Full VAPT launched, ${data.message || "pipeline running"}`, type: "success" });
   179→        toast({ title: "Full VAPT Launched!", description: data.message });
   180→      } else {
   181→        pushTerminalLine({ text: `[!] VAPT failed: ${data.error || "unknown error"}`, type: "err" });
   182→        toast({ variant: "destructive", title: "VAPT failed to start" });
   183→      }
   184→      load();
   185→    } catch (err) {
   186→      pushTerminalLine({ text: `[!] VAPT exception: ${err instanceof Error ? err.message : "unknown"}`, type: "err" });
   187→      toast({ variant: "destructive", title: "VAPT failed to start" });
   188→    }
   189→    setVaptRunning(false);
   190→  };
   191→
   192→  const handleDelete = async () => {
   193→    if (!confirm(`Delete "${detail?.name}"? This will remove the client and all associated data (codebases, targets, scans, findings). This cannot be undone.`)) return;
   194→    setDeleting(true);
   195→    try {
   196→      const res = await fetch(`/api/clients/${clientId}`, { method: "DELETE" });
   197→      if (res.ok) {
   198→        toast({ title: "Client deleted", description: detail?.name });
   199→        onBack();
   200→      } else {
   201→        toast({ variant: "destructive", title: "Delete failed" });
   202→      }
   203→    } catch {
   204→      toast({ variant: "destructive", title: "Delete failed" });
   205→    }
   206→    setDeleting(false);
   207→  };
   208→
   209→  const handleAuthorize = async () => {
   210→    setAuthorizing(true);
   211→    try {
   212→      const res = await fetch(`/api/clients/${clientId}`, {
   213→        method: "PATCH",
   214→        headers: { "Content-Type": "application/json" },
   215→        body: JSON.stringify({ authorized: !detail?.authorized }),
   216→      });
   217→      const data = await res.json();
   218→      if (!res.ok) throw new Error(data.error);
   219→      toast({
   220→        title: detail?.authorized ? "Authorization revoked" : "Client authorized",
   221→        description: detail?.authorized
   222→          ? "Testing is now blocked until re-authorized."
   223→          : "You can now run SAST + DAST scans on this client.",
   224→      });
   225→      load();
   226→    } catch (err) {
   227→      toast({
   228→        variant: "destructive",
   229→        title: "Failed",
   230→        description: err instanceof Error ? err.message : "unknown",
   231→      });
   232→    } finally {
   233→      setAuthorizing(false);
   234→    }
   235→  };
   236→
   237→  if (loading) {
   238→    return (
   239→      <div className="flex items-center justify-center py-20">
   240→        <Loader2 className="size-8 animate-spin text-emerald-400" />
   241→      </div>
   242→    );
   243→  }
   244→
   245→  if (!detail) {
   246→    return (
   247→      <div className="flex flex-col items-center justify-center py-20 text-center">
   248→        <AlertCircle className="size-12 text-red-400" />
   249→        <h3 className="mt-4 text-lg font-semibold text-zinc-200">Client not found</h3>
   250→        <Button onClick={onBack} className="mt-4 bg-emerald-600 text-white hover:bg-emerald-500">
   251→          <ArrowLeft className="size-4" /> Back to Clients
   252→        </Button>
   253→      </div>
   254→    );
   255→  }
   256→
   257→  return (
   258→    <div className="space-y-5">
   259→      {/* Header */}
   260→      <div className="flex items-start justify-between gap-4">
   261→        <div className="flex items-start gap-3">
   262→          <Button variant="ghost" size="icon" onClick={onBack} className="text-zinc-400 hover:text-emerald-400">
   263→            <ArrowLeft className="size-5" />
   264→          </Button>
   265→          <div>
   266→            <div className="flex items-center gap-2">
   267→              <h2 className="text-xl font-bold text-zinc-50 neon-emerald">{detail.name}</h2>
   268→              {detail.authorized ? (
   269→                <Badge className="border-emerald-500/30 bg-emerald-500/10 text-[10px] text-emerald-300">AUTHORIZED</Badge>
   270→              ) : (
   271→                <Badge className="border-amber-500/30 bg-amber-500/10 text-[10px] text-amber-300">PENDING AUTH</Badge>
   272→              )}
   273→            </div>
   274→            {detail.description && (
   275→              <p className="mt-1 text-sm text-zinc-400">{detail.description}</p>
   276→            )}
   277→            <div className="mt-2 flex flex-wrap items-center gap-3 text-xs text-zinc-500">
   278→              {detail.target_url && (
   279→                <a href={detail.target_url} target="_blank" rel="noopener noreferrer" className="flex items-center gap-1 hover:text-cyan-400">
   280→                  <Globe className="size-3" /> {detail.target_url}
   281→                </a>
   282→              )}
   283→              {detail.repo_url && (
   284→                <a href={detail.repo_url} target="_blank" rel="noopener noreferrer" className="flex items-center gap-1 hover:text-violet-400">
   285→                  <GitBranch className="size-3" /> {detail.repo_url}
   286→                </a>
   287→              )}
   288→              {detail.contact_email && (
   289→                <a href={`mailto:${detail.contact_email}`} className="flex items-center gap-1 hover:text-emerald-400">
   290→                  <Mail className="size-3" /> {detail.contact_email}
   291→                </a>
   292→              )}
   293→              {detail.contact_phone && (
   294→                <a href={`tel:${detail.contact_phone}`} className="flex items-center gap-1 hover:text-emerald-400">
   295→                  <Phone className="size-3" /> {detail.contact_phone}
   296→                </a>
   297→              )}
   298→            </div>
   299→          </div>
   300→        </div>
   301→        <div className="flex items-center gap-2">
   302→          <Button
   303→            onClick={handleFullVapt}
   304→            disabled={vaptRunning}
   305→            className="bg-emerald-600 text-white hover:bg-emerald-500 neon-border"
   306→          >
   307→            {vaptRunning ? <Loader2 className="size-4 animate-spin" /> : <Rocket className="size-4" />}
   308→            <span className="hidden sm:inline">One-Click Full VAPT</span>
   309→          </Button>
   310→          <Button
   311→            onClick={handleAuthorize}
   312→            disabled={authorizing}
   313→            variant="outline"
   314→            className={
   315→              detail.authorized
   316→                ? "border-red-500/40 bg-red-500/10 text-red-300 hover:bg-red-500/20"
   317→                : "border-emerald-500/40 bg-emerald-500/10 text-emerald-300 hover:bg-emerald-500/20"
   318→            }
   319→          >
   320→            {authorizing ? <Loader2 className="size-4 animate-spin" /> : <Shield className="size-4" />}
   321→            {detail.authorized ? "Revoke Auth" : "Authorize"}
   322→          </Button>
   323→          <Button
   324→            onClick={handleDelete}
   325→            disabled={deleting}
   326→            variant="ghost"
   327→            size="icon"
   328→            className="text-zinc-500 hover:bg-red-500/10 hover:text-red-400"
   329→            title="Delete client"
   330→          >
   331→            {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   332→          </Button>
   333→        </div>
   334→      </div>
   335→
   336→      {/* Full VAPT result */}
   337→      {vaptResult && (
   338→        <div className="holo-card-sharp hud-corners border-emerald-500/30 p-4">
   339→          <h3 className="mb-2 text-sm font-bold text-emerald-300">Full VAPT Status</h3>
   340→          <div className="space-y-1">
   341→            {vaptResult.steps?.map((s: any, i: number) => (
   342→              <div key={i} className="flex items-center gap-2 text-xs">
   343→                <span className={`size-2 rounded-full ${s.status === "completed" ? "bg-emerald-500" : s.status === "running" ? "bg-cyan-500 animate-pulse" : "bg-zinc-600"}`} />
   344→                <span className="text-zinc-400 font-mono">{s.step}.</span>
   345→                <span className="text-zinc-200">{s.action}</span>
   346→                <span className="text-zinc-500">- {s.detail}</span>
   347→              </div>
   348→            ))}
   349→          </div>
   350→          <p className="mt-2 text-xs text-emerald-300">{vaptResult.message}</p>
   351→        </div>
   352→      )}
   353→
   354→      {/* Frameworks */}
   355→      {detail.frameworks.length > 0 && (
   356→        <div className="flex flex-wrap items-center gap-2">
   357→          <span className="font-mono text-[10px] uppercase tracking-widest text-zinc-600">Compliance:</span>
   358→          {detail.frameworks.map((fw) => (
   359→            <Badge key={fw} className="border-purple-500/30 bg-purple-500/10 text-[10px] text-purple-300">{fw}</Badge>
   360→          ))}
   361→        </div>
   362→      )}
   363→
   364→      {/* Pipeline visualization */}
   365→      {pipeline && (
   366→        <div className="holo-card-sharp hud-corners p-5">
   367→          <div className="mb-4 flex items-center justify-between">
   368→            <h3 className="text-sm font-bold text-zinc-100 section-header">Security Pipeline</h3>
   369→            <div className="flex items-center gap-2">
   370→              <span className="font-mono text-xs text-zinc-500">Progress:</span>
   371→              <span className="font-bold text-emerald-400 neon-emerald">{pipeline.progress}%</span>
   372→            </div>
   373→          </div>
   374→
   375→          {/* Horizontal pipeline stepper, clickable to launch service */}
   376→          <div className="flex items-center gap-1 overflow-x-auto pb-2">
   377→            {pipeline.stages.map((stage, i) => {
   378→              const cfg = STAGE_COLORS[stage.key] || STAGE_COLORS.onboarding;
   379→              const Icon = STAGE_ICONS[stage.key] || Circle;
   380→              const serviceMap: Record<string, string> = {
   381→                scanning: "scan", testing: "test", patching: "patch",
   382→                verifying: "verify", defending: "defend", compliant: "comply",
   383→              };
   384→              const canLaunch = serviceMap[stage.key] && stage.status !== "completed" && detail.authorized;
   385→              return (
   386→                <div key={stage.id} className="flex items-center">
   387→                  <motion.button
   388→                    initial={{ opacity: 0, scale: 0.9 }}
   389→                    animate={{ opacity: 1, scale: 1 }}
   390→                    transition={{ delay: i * 0.08 }}
   391→                    onClick={async () => {
   392→                      if (!canLaunch) return;
   393→                      const service = serviceMap[stage.key];
   394→                      pushTerminalLine({ text: `$ guardianx launch --service ${service} --client "${detail?.name}"`, type: "cmd" });
   395→                      try {
   396→                        const res = await fetch("/api/launch-service", {
   397→                          method: "POST",
   398→                          headers: { "Content-Type": "application/json" },
   399→                          body: JSON.stringify({ service, clientIds: [clientId] }),
   400→                        });
   401→                        const data = await res.json();
   402→                        if (res.ok) {
   403→                          pushTerminalLine({ text: `[+] ${stage.label} launched, ${data.message || "ok"}`, type: "success" });
   404→                          toast({ title: `${stage.label} launched`, description: data.message });
   405→                        } else {
   406→                          pushTerminalLine({ text: `[!] ${stage.label} failed: ${data.error || "error"}`, type: "err" });
   407→                          toast({ variant: "destructive", title: "Launch failed" });
   408→                        }
   409→                        load();
   410→                      } catch (err) {
   411→                        pushTerminalLine({ text: `[!] ${stage.label} exception: ${err instanceof Error ? err.message : "unknown"}`, type: "err" });
   412→                        toast({ variant: "destructive", title: "Launch failed" });
   413→                      }
   414→                    }}
   415→                    disabled={!canLaunch}
   416→                    className={`flex min-w-[120px] flex-col items-center rounded-lg border ${cfg.border} ${cfg.bg} p-3 text-center transition-all ${canLaunch ? "cursor-pointer hover:scale-105 hover:border-emerald-500/60" : ""} disabled:cursor-not-allowed`}
   417→                    title={canLaunch ? `Click to launch ${stage.label}` : stage.status === "completed" ? "Completed" : "Not ready"}
   418→                  >
   419→                    <div className={`mb-1 flex size-8 items-center justify-center rounded-full border ${cfg.border} bg-zinc-950/60`}>
   420→                      {stage.status === "completed" ? (
   421→                        <CheckCircle2 className={`size-5 ${cfg.text}`} />
   422→                      ) : stage.status === "in-progress" ? (
   423→                        <Icon className={`size-4 ${cfg.text} animate-pulse`} />
   424→                      ) : (
   425→                        <Icon className="size-4 text-zinc-600" />
   426→                      )}
   427→                    </div>
   428→                    <div className={`text-[11px] font-bold ${stage.status !== "pending" ? cfg.text : "text-zinc-500"}`}>
   429→                      {i + 1}. {stage.label}
   430→                    </div>
   431→                    <div className="mt-0.5 text-[9px] text-zinc-600">{stage.desc}</div>
   432→                    {/* Metrics */}
   433→                    {stage.status !== "pending" && Object.keys(stage.metrics).length > 0 && (
   434→                      <div className="mt-1.5 flex flex-wrap justify-center gap-1">
   435→                        {Object.entries(stage.metrics).map(([k, v]) => (
   436→                          <span key={k} className="rounded bg-zinc-900/60 px-1 text-[8px] text-zinc-500">
   437→                            {k}: {String(v)}
   438→                          </span>
   439→                        ))}
   440→                      </div>
   441→                    )}
   442→                  </motion.button>
   443→                  {i < pipeline.stages.length - 1 && (
   444→                    <div className={`mx-0.5 h-0.5 w-4 ${stage.status === "completed" ? "bg-emerald-500" : "bg-zinc-800"}`} />
   445→                  )}
   446→                </div>
   447→              );
   448→            })}
   449→          </div>
   450→
   451→          {/* Summary bar */}
   452→          <div className="mt-4 grid grid-cols-2 gap-2 border-t border-zinc-800/60 pt-3 sm:grid-cols-4 lg:grid-cols-7">
   453→            <PipelineStat label="Codebases" value={pipeline.summary.codebases} color="text-sky-400" icon={Boxes} />
   454→            <PipelineStat label="Targets" value={pipeline.summary.targets} color="text-red-400" icon={Target} />
   455→            <PipelineStat label="Scans" value={pipeline.summary.scans} color="text-cyan-400" icon={Bug} />
   456→            <PipelineStat label="Patches" value={pipeline.summary.patches} color="text-emerald-400" icon={ShieldCheck} />
   457→            <PipelineStat label="Findings" value={pipeline.summary.findings} color="text-amber-400" icon={AlertCircle} />
   458→            <PipelineStat label="Critical" value={pipeline.summary.critical_findings} color="text-red-400" icon={AlertCircle} />
   459→            <PipelineStat label="Canaries" value={pipeline.summary.canaries} color="text-rose-400" icon={Shield} />
   460→          </div>
   461→        </div>
   462→      )}
   463→
   464→      {/* Assets */}
   465→      <div className="grid gap-4 lg:grid-cols-2">
   466→        {/* Codebases */}
   467→        <div className="holo-card-sharp hud-corners p-5">
   468→          <div className="mb-3 flex items-center justify-between">
   469→            <h3 className="text-sm font-bold text-zinc-100 section-header">Codebases (SAST)</h3>

   673→function ActionButton({
   674→  label, desc, icon: Icon, color, onClick, disabled,
   675→}: {
   676→  label: string;
   677→  desc: string;
   678→  icon: typeof Circle;
   679→  color: string;
   680→  onClick: () => void;
   681→  disabled?: boolean;
   682→}) {
   683→  const colorMap: Record<string, { text: string; border: string; bg: string; hover: string }> = {
   684→    cyan: { text: "text-cyan-400", border: "border-cyan-500/30", bg: "bg-cyan-500/5", hover: "hover:bg-cyan-500/10" },
   685→    red: { text: "text-red-400", border: "border-red-500/30", bg: "bg-red-500/5", hover: "hover:bg-red-500/10" },
   686→    emerald: { text: "text-emerald-400", border: "border-emerald-500/30", bg: "bg-emerald-500/5", hover: "hover:bg-emerald-500/10" },
   687→    rose: { text: "text-rose-400", border: "border-rose-500/30", bg: "bg-rose-500/5", hover: "hover:bg-rose-500/10" },
   688→  };
   689→  const c = colorMap[color] || colorMap.cyan;
   690→  return (
   691→    <button
   692→      onClick={onClick}
   693→      disabled={disabled}
   694→      className={`flex items-center gap-2 rounded-lg border ${c.border} ${c.bg} ${c.hover} p-3 text-left transition-all disabled:cursor-not-allowed disabled:opacity-40`}
   695→    >
   696→      <Icon className={`size-4 ${c.text}`} />
   697→      <div>
   698→        <div className="text-xs font-bold text-zinc-200">{label}</div>
   699→        <div className="text-[10px] text-zinc-500">{desc}</div>
   700→      </div>
   701→    </button>
   702→  );
   703→}
