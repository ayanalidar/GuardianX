   485→function SectionHeading({
   486→  eyebrow,
   487→  title,
   488→  icon: Icon,
   489→  accent,
   490→  subtitle,
   491→}: {
   492→  eyebrow: string;
   493→  title: string;
   494→  icon: React.ComponentType<{ className?: string }>;
   495→  accent: "emerald" | "violet" | "cyan" | "amber";
   496→  subtitle?: string;
   497→}) {
   498→  const accentMap: Record<string, string> = {
   499→    emerald: "border-emerald-500/40 bg-emerald-500/10 text-emerald-400",
   500→    violet: "border-violet-500/40 bg-violet-500/10 text-violet-400",
   501→    cyan: "border-cyan-500/40 bg-cyan-500/10 text-cyan-400",
   502→    amber: "border-amber-500/40 bg-amber-500/10 text-amber-400",
   503→  };
   504→  const titleColor: Record<string, string> = {
   505→    emerald: "text-emerald-300",
   506→    violet: "text-violet-300",
   507→    cyan: "text-cyan-300",
   508→    amber: "text-amber-300",
   509→  };
   510→  return (
   511→    <motion.div
   512→      initial={{ opacity: 0, x: -24 }}
   513→      whileInView={{ opacity: 1, x: 0 }}
   514→      viewport={{ once: true }}
   515→      transition={{ duration: 0.5 }}
   516→      className="mb-8 flex items-start gap-4"
   517→    >
   518→      <div
   519→        className={`flex size-12 shrink-0 items-center justify-center rounded-xl border ${accentMap[accent]}`}
   520→      >
   521→        <Icon className="size-6" />
   522→      </div>
   523→      <div>
   524→        <div className="font-mono text-[10px] uppercase tracking-widest text-zinc-600">
   525→          {eyebrow}
   526→        </div>
   527→        <h2 className={`text-2xl font-bold sm:text-3xl ${titleColor[accent]}`}>
   528→          {title}
   529→        </h2>
   530→        {subtitle && (
   531→          <p className="mt-2 max-w-2xl text-sm text-zinc-400">{subtitle}</p>
   532→        )}
   533→      </div>
   534→    </motion.div>
   535→  );
   536→}
