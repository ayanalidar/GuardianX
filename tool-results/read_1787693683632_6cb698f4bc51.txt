   156→/** Helper to apply a palette to a feature, keeping call sites terse. */
   157→function withColor(
   158→  icon: LucideIcon,
   159→  title: string,
   160→  category: string,
   161→  desc: string,
   162→  palette: PaletteName,
   163→  isNew?: boolean,
   164→): Feature {
   165→  const p = PALETTES[palette];
   166→  return {
   167→    icon,
   168→    title,
   169→    category,
   170→    desc,
   171→    color: p.color,
   172→    neon: p.neon,
   173→    border: p.border,
   174→    glow: p.glow,
   175→    bg: p.bg,
   176→    ...(isNew ? { isNew: true } : {}),
   177→  };
   178→}
