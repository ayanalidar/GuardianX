     3→/**
     4→ * AgentX
     5→ * ------
     6→ * Always-on conversational AI for the GuardianX Command Center.
     7→ *
     8→ * Voice infrastructure: the War Room's <VoiceControl> component is reused
     9→ * purely as an STT + TTS primitive. We mount it visually-hidden (sr-only)
    10→ * with `continuous` + `speakResponses={false}` so it owns the
    11→ * SpeechRecognition instance and auto-restarts on browser silence
    12→ * timeouts. Its `onCommand` callback hands us every parsed utterance;
    13→ * for unrecognized input we get `{action: "unknown", raw: transcript}`
    14→ * which we forward to `/api/agent-x/chat` for an LLM/heuristic reply.
    15→ *
    16→ * Talkback: we do NOT use VoiceControl's `speak()` for the LLM reply
    17→ * because its default rate is 1.05 / pitch 0.95. Agent X is a "sophisticated
    18→ * + lethal" SOC analyst — we want a slower, lower-pitched, authoritative
    19→ * voice. So we run our own SpeechSynthesisUtterance with rate 0.95 / pitch
    20→ * 0.85 and pick a male-coded voice when available.
    21→ *
    22→ * Continuous listening: once activated (open=true) the mic stays hot
    23→ * until the panel is closed. No per-command mic tap.
    24→ *
    25→ * Proactive monitoring: every 60s while open we re-fetch
    26→ * `/api/agent-x/briefing` and, if pending patches appeared or the
    27→ * posture score dropped, we speak a heads-up aloud + drop an alert
    28→ * message in the conversation.
    29→ *
    30→ * Tab awareness: when `currentTab` changes we fetch
    31→ * `/api/agent-x/context?tab={tab}` and surface 2–3 contextual
    32→ * quick-reply chips.
    33→ *
    34→ * Persistence: the last 20 messages are mirrored to
    35→ * localStorage `agent_x_conversation` so a reload while the panel is
    36→ * open picks up where the user left off.
    37→ */

   160→const STORAGE_KEY = "agent_x_conversation";
   161→const MAX_HISTORY = 20;
   162→const PROACTIVE_POLL_MS = 60_000;
   163→
   164→// ─── Voice persona: lower-pitched voice picker ───────────────────────────────
   165→// SpeechSynthesis voice names vary by OS/browser. We prefer a male-coded
   166→// en-US voice (David/Alex/Daniel/Fred/etc.) for an authoritative SOC
   167→// analyst persona. Falls back to any en voice, then any voice at all.
   168→
   169→function pickPersonaVoice(voices: SpeechSynthesisVoice[]): SpeechSynthesisVoice | null {
   170→  if (!voices.length) return null;
   171→  const maleNames = /david|alex|daniel|fred|james|mark|oliver|george|rishi|arjun/i;
   172→  return (
   173→    voices.find((v) => /^en[-_]US/i.test(v.lang) && maleNames.test(v.name)) ||
   174→    voices.find((v) => /^en[-_]GB/i.test(v.lang) && maleNames.test(v.name)) ||
   175→    voices.find((v) => /^en/i.test(v.lang) && maleNames.test(v.name)) ||
   176→    voices.find((v) => /en[-_]US/i.test(v.lang) && /google|samantha|jenny/i.test(v.name)) ||
   177→    voices.find((v) => /^en/i.test(v.lang)) ||
   178→    voices[0] ||
   179→    null
   180→  );
   181→}
   182→
   183→// ─── Reconstruct a synthetic transcript from a parsed VoiceCommand ────────────
   184→// VoiceControl's parser strips intent + target apart. To preserve the
   185→// natural-language phrasing for the Agent X LLM, we rebuild a short
   186→// utterance that mirrors the original intent. The LLM can then echo it
   187→// back contextually.
   188→
   189→function reconstructTranscript(cmd: VoiceCommand): string | null {
   190→  switch (cmd.action) {
   191→    case "scan":
   192→      return `scan ${cmd.target}`;
   193→    case "navigate":
   194→      return `show ${cmd.target}`;
   195→    case "approve":
   196→      return `approve patch ${cmd.target}`;
   197→    case "search":
   198→      return `search for ${cmd.target}`;
   199→    case "status":
   200→      return "What's my current security posture?";
   201→    case "unknown":
   202→      return cmd.raw;
   203→    case "stop":
   204→      return null; // signal: cancel TTS only, no Agent X round-trip
   205→    default:
   206→      return null;
   207→  }
   208→}

   210→// ─── Component ───────────────────────────────────────────────────────────────
   211→
   212→export function AgentX({
   213→  currentTab,
   214→  currentUser,
   215→  onNavigate,
   216→  onScan,
   217→  onApprovePatch,
   218→  onSearch,
   219→  onOpenWarRoom,
   220→  open,
   221→  onClose,
   222→}: AgentXProps) {
   223→  const voiceRef = useRef<VoiceControlHandle>(null);
   224→  const voicesRef = useRef<SpeechSynthesisVoice[]>([]);
   225→
   226→  // ── Voice state mirror (driven by VoiceControl's onStateChange) ─────────
   227→  const [voiceState, setVoiceState] = useState<VoiceControlState>({
   228→    listening: false,
   229→    speaking: false,
   230→    interim: "",
   231→    supported: true,
   232→  });
   233→
   234→  // ── Local TTS speaking flag — driven by our own speakReply() so the
   235→  //    "Agent X is speaking" indicator tracks our slower-rate utterances
   236→  //    even though VoiceControl's own speaking flag stays false.
   237→  const [speakingLocal, setSpeakingLocal] = useState(false);
   238→
   239→  // ── Conversation + UI state ─────────────────────────────────────────────
   240→  const [messages, setMessages] = useState<AgentMessage[]>([]);
   241→  const [thinking, setThinking] = useState(false);
   242→  const [input, setInput] = useState("");
   243→  const [briefing, setBriefing] = useState<AgentXBriefing | null>(null);
   244→  const [tabContext, setTabContext] = useState<AgentXTabContext | null>(null);
   245→  const [suggestions, setSuggestions] = useState<string[]>([]);
   246→  const [resumedSession, setResumedSession] = useState(false);
   247→
   248→  // ── Proactive-monitoring baseline (last seen counts) ────────────────────
   249→  const lastPatchCountRef = useRef<number | null>(null);
   250→  const lastPostureScoreRef = useRef<number | null>(null);
   251→
   252→  // ── Refs mirroring state for use inside stable callbacks ────────────────
   253→  // The VoiceControl onCommand callback is created once; we need the
   254→  // latest `thinking` + `currentTab` + `messages` without re-subscribing
   255→  // VoiceControl's effect on every render.
   256→  const currentTabRef = useRef(currentTab);
   257→  const messagesRef = useRef<AgentMessage[]>(messages);
   258→  const thinkingRef = useRef(thinking);
   259→  const openRef = useRef(open);
   260→
   261→  useEffect(() => {
   262→    currentTabRef.current = currentTab;
   263→  }, [currentTab]);
   264→  useEffect(() => {
   265→    messagesRef.current = messages;
   266→  }, [messages]);
   267→  useEffect(() => {
   268→    thinkingRef.current = thinking;
   269→  }, [thinking]);
   270→  useEffect(() => {
   271→    openRef.current = open;
   272→  }, [open]);
   273→
   274→  const scrollRef = useRef<HTMLDivElement>(null);
   275→
   276→  // ── Warm TTS voices (some browsers load them async) ────────────────────
   277→  useEffect(() => {
   278→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   279→    const loadVoices = () => {
   280→      voicesRef.current = window.speechSynthesis.getVoices();
   281→    };
   282→    loadVoices();
   283→    window.speechSynthesis.onvoiceschanged = loadVoices;
   284→    return () => {
   285→      window.speechSynthesis.onvoiceschanged = null;
   286→    };
   287→  }, []);
   288→
   289→  // ── Restore prior conversation from localStorage on first open ──────────
   290→  // We do this ONCE per mount (not on every open) so a reload while
   291→  // already-open picks up where the user left off, but re-opening after
   292→  // a close keeps the existing in-memory messages.
   293→  const restoredRef = useRef(false);
   294→  useEffect(() => {
   295→    if (!open || restoredRef.current) return;
   296→    restoredRef.current = true;
   297→    try {
   298→      const raw = localStorage.getItem(STORAGE_KEY);
   299→      if (!raw) return;
   300→      const parsed = JSON.parse(raw) as AgentMessage[];
   301→      if (Array.isArray(parsed) && parsed.length > 0) {
   302→        setMessages(parsed.slice(-MAX_HISTORY));
   303→        setResumedSession(true);
   304→      }
   305→    } catch {
   306→      /* quota / privacy mode — ignore */
   307→    }
   308→  }, [open]);
   309→
   310→  // ── Persist conversation to localStorage ────────────────────────────────
   311→  useEffect(() => {
   312→    if (messages.length === 0) return;
   313→    try {
   314→      localStorage.setItem(STORAGE_KEY, JSON.stringify(messages.slice(-MAX_HISTORY)));
   315→    } catch {
   316→      /* quota / privacy mode — ignore */
   317→    }
   318→  }, [messages]);
   319→
   320→  // ── Auto-scroll to bottom on new messages ───────────────────────────────
   321→  useEffect(() => {
   322→    const el = scrollRef.current;
   323→    if (!el) return;
   324→    el.scrollTo({ top: el.scrollHeight, behavior: "smooth" });
   325→  }, [messages, thinking]);
   326→
   327→  // ── Voice persona talkback ──────────────────────────────────────────────
   328→  // rate 0.95 (slower for clarity) + pitch 0.85 (lower, authoritative).
   329→  // Picks a male-coded en voice when available so Agent X sounds like a
   330→  // seasoned SOC analyst rather than a default assistant chirp.
   331→  const speakReply = useCallback((text: string) => {
   332→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   333→    const trimmed = text.trim();
   334→    if (!trimmed) return;
   335→    const synth = window.speechSynthesis;
   336→    synth.cancel(); // interrupt any in-flight utterance
   337→    const voices = voicesRef.current.length ? voicesRef.current : synth.getVoices();
   338→    const preferred = pickPersonaVoice(voices);
   339→    const u = new SpeechSynthesisUtterance(trimmed);
   340→    if (preferred) u.voice = preferred;
   341→    u.rate = 0.95;
   342→    u.pitch = 0.85;
   343→    u.volume = 1;
   344→    u.onstart = () => setSpeakingLocal(true);
   345→    u.onend = () => setSpeakingLocal(false);
   346→    u.onerror = () => setSpeakingLocal(false);
   347→    synth.speak(u);
   348→  }, []);
   349→
   350→  const stopSpeaking = useCallback(() => {
   351→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   352→    window.speechSynthesis.cancel();
   353→    setSpeakingLocal(false);
   354→  }, []);
   355→
   356→  // ── Execute Agent X actions returned from the chat endpoint ─────────────
   357→  const executeActions = useCallback(
   358→    (actions: AgentXAction[] | undefined) => {
   359→      if (!actions || actions.length === 0) return;
   360→      for (const action of actions) {
   361→        switch (action.type) {
   362→          case "navigate":
   363→            onNavigate?.(action.target);
   364→            break;
   365→          case "scan":
   366→            onScan?.(action.target);
   367→            break;
   368→          case "approve_patch":
   369→            onApprovePatch?.(action.target);
   370→            break;
   371→          case "search":
   372→            onSearch?.(action.target);
   373→            break;
   374→          case "war_room":
   375→            onOpenWarRoom?.();
   376→            break;
   377→          default:
   378→            // Unknown action types are silently ignored — the backend may
   379→            // emit intents the frontend hasn't wired up yet.
   380→            break;
   381→        }
   382→      }
   383→    },
   384→    [onApprovePatch, onNavigate, onOpenWarRoom, onScan, onSearch],
   385→  );
   386→
   387→  // ── Send a message to /api/agent-x/chat ─────────────────────────────────
   388→  const sendMessage = useCallback(
   389→    async (text: string) => {
   390→      const trimmed = text.trim();
   391→      if (!trimmed || thinkingRef.current) return;
   392→
   393→      setInput("");
   394→      setThinking(true);
   395→      setMessages((prev) => [
   396→        ...prev,
   397→        { role: "user", content: trimmed, ts: Date.now() },
   398→      ]);
   399→
   400→      try {
   401→        const res = await fetch("/api/agent-x/chat", {
   402→          method: "POST",
   403→          headers: { "Content-Type": "application/json" },
   404→          body: JSON.stringify({
   405→            message: trimmed,
   406→            context: {
   407→              currentTab: currentTabRef.current,
   408→              history: messagesRef.current
   409→                .slice(-10)
   410→                .map((m) => ({ role: m.role, content: m.content })),
   411→            },
   412→          }),
   413→        });
   414→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   415→        const data = (await res.json()) as AgentXChatResponse;
   416→
   417→        const reply = data.reply?.trim() || "I didn't catch that. Try again.";
   418→        setMessages((prev) => [
   419→          ...prev,
   420→          { role: "agent", content: reply, ts: Date.now() },
   421→        ]);
   422→        speakReply(reply);
   423→        executeActions(data.actions);
   424→        if (Array.isArray(data.suggestions) && data.suggestions.length > 0) {
   425→          setSuggestions(data.suggestions.slice(0, 4));
   426→        }
   427→      } catch {
   428→        const errMsg =
   429→          "I'm having trouble reaching the security core. Try again in a moment.";
   430→        setMessages((prev) => [
   431→          ...prev,
   432→          { role: "agent", content: errMsg, ts: Date.now(), kind: "error" },
   433→        ]);
   434→        speakReply(errMsg);
   435→      } finally {
   436→        setThinking(false);
   437→      }
   438→    },
   439→    [executeActions, speakReply],
   440→  );
   441→
   442→  // ── VoiceControl onCommand — receives parsed VoiceCommands ──────────────
   443→  // For `unknown` actions we forward `raw` to Agent X. For built-in
   444→  // commands we reconstruct a synthetic transcript so the LLM stays in
   445→  // the loop and can produce a contextual reply. `stop` cancels TTS only.
   446→  const handleVoiceCommand = useCallback(
   447→    (cmd: VoiceCommand) => {
   448→      if (cmd.action === "stop") {
   449→        stopSpeaking();
   450→        return;
   451→      }
   452→      const transcript = reconstructTranscript(cmd);
   453→      if (transcript) void sendMessage(transcript);
   454→    },
   455→    [sendMessage, stopSpeaking],
   456→  );
   457→
   458→  const handleStateChange = useCallback((next: VoiceControlState) => {
   459→    setVoiceState(next);
   460→  }, []);
   461→
   462→  // ── On open: fetch briefing, speak greeting, auto-start listening ───────
   463→  useEffect(() => {
   464→    if (!open) return;
   465→    let cancelled = false;
   466→
   467→    (async () => {
   468→      try {
   469→        const res = await fetch("/api/agent-x/briefing");
   470→        if (!res.ok) return;
   471→        const data = (await res.json()) as AgentXBriefing;
   472→        if (cancelled) return;
   473→        setBriefing(data);
   474→        lastPatchCountRef.current = data.pendingTasks ?? null;
   475→        lastPostureScoreRef.current = data.postureScore ?? null;
   476→
   477→        const greeting =
   478→          data.greeting ||
   479→          (currentUser?.name
   480→            ? `Good to see you, ${currentUser.name}. Agent X online.`
   481→            : "Agent X online. Ready when you are.");
   482→
   483→        setMessages((prev) => [
   484→          ...prev,
   485→          { role: "agent", content: greeting, ts: Date.now(), kind: "briefing" },
   486→        ]);
   487→        if (Array.isArray(data.suggestions) && data.suggestions.length > 0) {
   488→          setSuggestions(data.suggestions.slice(0, 4));
   489→        }
   490→        speakReply(greeting);
   491→
   492→        // Auto-start continuous listening once the greeting begins to
   493→        // play — give the TTS a beat to start so it doesn't capture its
   494→        // own voice through the mic.
   495→        const v = voiceRef.current;
   496→        if (v?.isSupported()) {
   497→          window.setTimeout(() => {
   498→            if (cancelled) return;
   499→            try {
   500→              v.startListening();
   501→            } catch {
   502→              /* InvalidStateError if start() raced — ignored */
   503→            }
   504→          }, 600);
   505→        }
   506→      } catch {
   507→        /* network/silent failure — panel still usable via text input */
   508→      }
   509→    })();
   510→
   511→    return () => {
   512→      cancelled = true;
   513→    };
   514→  }, [open, currentUser, speakReply]);
   515→
   516→  // ── Tab-aware context: re-fetch suggestions when currentTab changes ─────
   517→  useEffect(() => {
   518→    if (!open) return;
   519→    let cancelled = false;
   520→    (async () => {
   521→      try {
   522→        const res = await fetch(
   523→          `/api/agent-x/context?tab=${encodeURIComponent(currentTab)}`,
   524→        );
   525→        if (!res.ok) return;
   526→        const data = (await res.json()) as AgentXTabContext;
   527→        if (cancelled || !data) return;
   528→        setTabContext(data);
   529→        if (Array.isArray(data.suggestions) && data.suggestions.length > 0) {
   530→          setSuggestions(data.suggestions.slice(0, 4));
   531→        }
   532→      } catch {
   533→        /* ignore — fall back to existing suggestions */
   534→      }
   535→    })();
   536→    return () => {
   537→      cancelled = true;
   538→    };
   539→  }, [open, currentTab]);
   540→
   541→  // ── Proactive monitoring: poll briefing every 60s, speak alerts ─────────
   542→  useEffect(() => {
   543→    if (!open) return;
   544→    const id = window.setInterval(async () => {
   545→      if (!openRef.current) return;
   546→      try {
   547→        const res = await fetch("/api/agent-x/briefing");
   548→        if (!res.ok) return;
   549→        const data = (await res.json()) as AgentXBriefing;
   550→        const newPatches = data.pendingTasks ?? 0;
   551→        const newScore = data.postureScore ?? 0;
   552→
   553→        if (
   554→          lastPatchCountRef.current !== null &&
   555→          newPatches > lastPatchCountRef.current
   556→        ) {
   557→          const msg =
   558→            "Heads up — a new critical patch was just generated. Want me to show you?";
   559→          setMessages((prev) => [
   560→            ...prev,
   561→            { role: "agent", content: msg, ts: Date.now(), kind: "alert" },
   562→          ]);
   563→          speakReply(msg);
   564→        }
   565→        if (
   566→          lastPostureScoreRef.current !== null &&
   567→          newScore < lastPostureScoreRef.current
   568→        ) {
   569→          const msg = `Your security posture dropped to ${newScore}. Want me to investigate?`;
   570→          setMessages((prev) => [
   571→            ...prev,
   572→            { role: "agent", content: msg, ts: Date.now(), kind: "alert" },
   573→          ]);
   574→          speakReply(msg);
   575→        }
   576→        lastPatchCountRef.current = newPatches;
   577→        lastPostureScoreRef.current = newScore;
   578→      } catch {
   579→        /* swallow — proactive polling must never crash the panel */
   580→      }
   581→    }, PROACTIVE_POLL_MS);
   582→    return () => window.clearInterval(id);
   583→  }, [open, speakReply]);
   584→
   585→  // ── Stop TTS + listening when the panel closes ──────────────────────────
   586→  useEffect(() => {
   587→    if (open) return;
   588→    if (typeof window !== "undefined" && "speechSynthesis" in window) {

  1063→interface BriefingStatProps {

  1087→      </span>
  1088→    </div>
  1089→  );
  1090→}
