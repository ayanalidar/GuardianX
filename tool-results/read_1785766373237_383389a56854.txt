   129→    try {

   131→        fetch("/api/clients"),
   132→        fetch("/api/activity-feed"),
   133→      ]);
   134→      const c = await cRes.json();

   765→// ── Helper: determine stage status from client data ────────────────────────
   766→function getStageStatus(client: ClientSummary, stageKey: string): "pending" | "in-progress" | "completed" {
   767→  const s = client.stats;
   768→  switch (stageKey) {
   769→    case "onboarding":
   770→      return (s.codebases > 0 || s.targets > 0) && client.authorized ? "completed" : (s.codebases > 0 || s.targets > 0) ? "in-progress" : "pending";
   771→    case "scanning":
   772→      return s.patches > 0 || s.findings > 0 ? "completed" : client.status === "scanning" ? "in-progress" : "pending";
   773→    case "testing":
   774→      return s.findings > 0 ? "completed" : client.status === "testing" ? "in-progress" : "pending";
   775→    case "patching":
   776→      return s.approved_patches > 0 ? "completed" : s.pending_patches > 0 ? "in-progress" : "pending";
   777→    case "verifying":
   778→      return s.approved_patches > 0 ? "completed" : "pending";
   779→    case "defending":
   780→      return client.status === "defending" || client.status === "compliant" ? "completed" : "pending";
   781→    case "compliant":
   782→      return client.status === "compliant" ? "completed" : "pending";
   783→    default:
   784→      return "pending";
   785→  }
   786→}
