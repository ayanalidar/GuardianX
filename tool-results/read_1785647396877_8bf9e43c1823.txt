     1→     1→     1→// GuardianX Integration Engine — Connector Framework
     2→     2→     2→//
     3→     3→     3→// Each connector knows how to format + send security events to its target
     4→     4→     4→// platform. The engine loads all active integrations from the DB and forwards
     5→     5→     5→// events to every configured connector.
     6→     6→     6→//
     7→     7→     7→// Supported connectors:
     8→     8→     8→//   - splunk    (Splunk HEC live forwarding)
     9→     9→     9→//   - elk       (Elasticsearch bulk API)
    10→    10→    10→//   - datadog   (Datadog Events API)
    11→    11→    11→//   - jira      (Create tickets on critical findings)
    12→    12→    12→//   - pagerduty (Trigger alerts on incidents)
    13→    13→    13→//   - securityhub (AWS Security Hub ASFF batch import)
    14→    14→    14→//   - teams     (Microsoft Teams interactive cards)
    15→    15→    15→//   - slack     (Slack incoming webhook)
    16→    16→    16→//   - webhook   (Generic HTTP webhook)

    84→    84→    84→// ── In-memory forwarding log (last 100 events) ─────────────────────────────
    85→    85→    85→const forwardingLog: ForwardingLog[] = [];
    86→    86→    86→const MAX_LOG = 100;
    87→    87→    87→
    88→    88→    88→function logForwarding(event: SecurityEvent, results: ConnectorResult[]): void {
    89→    89→    89→  const entry: ForwardingLog = {
    90→    90→    90→    id: randomUUID(),
    91→    91→    91→    eventId: event.incidentId || event.patchId || event.findingId || randomUUID(),
    92→    92→    92→    eventType: event.type,
    93→    93→    93→    results,
    94→    94→    94→    totalSent: results.length,
    95→    95→    95→    totalSuccess: results.filter((r) => r.success).length,
    96→    96→    96→    totalFailed: results.filter((r) => !r.success).length,
    97→    97→    97→    timestamp: new Date().toISOString(),
    98→    98→    98→  };
    99→    99→    99→  forwardingLog.unshift(entry);
   100→   100→   100→  if (forwardingLog.length > MAX_LOG) forwardingLog.pop();
   101→   101→   101→}
   102→   102→   102→

   104→   104→   104→  return forwardingLog;

   136→   136→   136→  async send(formatted, config) {
   137→   137→   137→    const hecUrl = config.hecUrl as string;
   138→   138→   138→    const token = config.token as string;

   140→   140→   140→    try {
   141→   141→   141→      const res = await fetch(`${hecUrl}/services/collector`, {
   142→   142→   142→        method: "POST",
   143→   143→   143→        headers: { Authorization: `Splunk ${token}`, "Content-Type": "application/json" },
   144→   144→   144→        body: JSON.stringify(formatted),
   145→   145→   145→        signal: AbortSignal.timeout(10_000),
   146→   146→   146→      });
   147→   147→   147→      return {
   148→   148→   148→        success: res.ok,
   149→   149→   149→        message: res.ok ? "Forwarded to Splunk HEC" : `Splunk returned ${res.status}`,
   150→   150→   150→        responseCode: res.status,
   151→   151→   151→        responseBody: await res.text().catch(() => ""),
   152→   152→   152→      };
   153→   153→   153→    } catch (err) {
   154→   154→   154→      return { success: false, message: err instanceof Error ? err.message : "Splunk unreachable" };

   598→   598→   598→/**
   599→   599→   599→ * Forward a security event to all active + configured integrations.
   600→   600→   600→ * Called by the pipeline when findings/patches/incidents are created.
   601→   601→   601→ */
   602→   602→   602→export async function forwardEvent(event: SecurityEvent): Promise<ForwardingLog> {
   603→   603→   603→  // Load all active integrations from DB
   604→   604→   604→  let integrations: Array<{ id: string; type: string; config: string; isActive: boolean }> = [];
   605→   605→   605→  try {
   606→   606→   606→    integrations = (await db.integration.findMany({
   607→   607→   607→      where: { isActive: true },
   608→   608→   608→    })) as Array<{ id: string; type: string; config: string; isActive: boolean }>;
   609→   609→   609→  } catch {
   610→   610→   610→    // DB may not be ready — skip forwarding
   611→   611→   611→    const emptyLog: ForwardingLog = {
   612→   612→   612→      id: randomUUID(),
   613→   613→   613→      eventId: event.incidentId || event.patchId || event.findingId || randomUUID(),
   614→   614→   614→      eventType: event.type,
   615→   615→   615→      results: [],
   616→   616→   616→      totalSent: 0,
   617→   617→   617→      totalSuccess: 0,
   618→   618→   618→      totalFailed: 0,
   619→   619→   619→      timestamp: new Date().toISOString(),
   620→   620→   620→    };
   621→   621→   621→    return emptyLog;
   622→   622→   622→  }
   623→   623→   623→
   624→   624→   624→  const results: ConnectorResult[] = [];
   625→   625→   625→
   626→   626→   626→  for (const integration of integrations) {
   627→   627→   627→    const connector = connectors[integration.type];
   628→   628→   628→    if (!connector) continue;
   629→   629→   629→
   630→   630→   630→    let config: IntegrationConfig = {};
   631→   631→   631→    try {
   632→   632→   632→      config = integration.config ? JSON.parse(integration.config) : {};
   633→   633→   633→    } catch {
   634→   634→   634→      config = {};
   635→   635→   635→    }
   636→   636→   636→
   637→   637→   637→    // Check if this connector should forward this event
   638→   638→   638→    if (!connector.shouldForward(event, config)) continue;
   639→   639→   639→
   640→   640→   640→    // Format + send
   641→   641→   641→    try {
   642→   642→   642→      const formatted = connector.format(event, config);
   643→   643→   643→      const result = await connector.send(formatted, config);
   644→   644→   644→      results.push({
   645→   645→   645→        integrationId: integration.id,
   646→   646→   646→        type: integration.type,
   647→   647→   647→        success: result.success,
   648→   648→   648→        message: result.message,
   649→   649→   649→        responseCode: result.responseCode,
   650→   650→   650→        responseBody: result.responseBody,
   651→   651→   651→        timestamp: new Date().toISOString(),
   652→   652→   652→      });
   653→   653→   653→    } catch (err) {
   654→   654→   654→      results.push({
   655→   655→   655→        integrationId: integration.id,
   656→   656→   656→        type: integration.type,
   657→   657→   657→        success: false,
   658→   658→   658→        message: err instanceof Error ? err.message : "Connector error",
   659→   659→   659→        timestamp: new Date().toISOString(),
   660→   660→   660→      });
   661→   661→   661→    }
   662→   662→   662→  }
   663→   663→   663→
   664→   664→   664→  const log: ForwardingLog = {
   665→   665→   665→    id: randomUUID(),
   666→   666→   666→    eventId: event.incidentId || event.patchId || event.findingId || randomUUID(),
   667→   667→   667→    eventType: event.type,
   668→   668→   668→    results,
   669→   669→   669→    totalSent: results.length,
   670→   670→   670→    totalSuccess: results.filter((r) => r.success).length,
   671→   671→   671→    totalFailed: results.filter((r) => !r.success).length,
   672→   672→   672→    timestamp: new Date().toISOString(),
   673→   673→   673→  };
   674→   674→   674→
   675→   675→   675→  logForwarding(event, results);
   676→   676→   676→  return log;
   677→   677→   677→}
