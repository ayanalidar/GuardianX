     3→/**
     4→ * AgentX — Sidebar Tab Edition
     5→ * ----------------------------
     6→ * Always-on conversational AI for the GuardianX Command Center.
     7→ *
     8→ * Previously a floating drawer at `fixed bottom-20 right-4 z-[90]`, Agent X
     9→ * is now a full-tab view that fills the main content area (just like
    10→ * Overview / All Clients / Patches / etc.). The parent mounts it inside
    11→ * the main panel whenever `tab === "agent-x"` and controls visibility
    12→ * via the `open` prop. When `open={false}` the component renders nothing.
    13→ *
    14→ * Voice architecture (rebuilt — no longer uses <VoiceControl>):
    15→ *
    16→ *   - We own a `SpeechRecognition` instance directly so we can hook the
    17→ *     `onsoundstart` / `onsoundend` events for voice-activity detection.
    18→ *   - `speakingRef` gates the recognition `onend` auto-restart: when
    19→ *     Agent X is speaking, a browser-side silence timeout that fires
    20→ *     `onend` is suppressed (no auto-restart). The mic is *still live*
    21→ *     to the extent the browser allows it — so if the user starts
    22→ *     speaking mid-TTS, `onsoundstart` fires and we cancel TTS
    23→ *     immediately (interruptible).
    24→ *   - When TTS finishes (`utterance.onend`), we flip `speakingRef` off
    25→ *     and restart recognition if it had stopped.
    26→ *
    27→ * Streaming TTS:
    28→ *
    29→ *   - Replies are split into sentences. The first sentence is spoken
    30→ *     immediately as a "fast first response"; subsequent sentences are
    31→ *     queued (`onend` of utterance N triggers start of N+1). Display
    32→ *     is also progressive: the message bubble starts with the first
    33→ *     sentence and grows as subsequent chunks arrive.
    34→ *
    35→ * Layout (when open):
    36→ *
    37→ *   ┌─────────────────────────────────────────────────────────────────┐
    38→ *   │ Header: AGENT X · provider badge · mic toggle · clear · export    │
    39→ *   ├──────────────────────────────────────┬──────────────────────────┤
    40→ *   │                                       │  Briefing panel (30%)    │
    41→ *   │  Conversation (70%, scrollable)       │  - Posture score         │
    42→ *   │                                       │  - Pending patches       │
    43→ *   │  [user msg]                           │  - Critical findings     │
    44→ *   │  [agent msg with thinking dots]       │  - Recent activity       │
    45→ *   │                                       │                          │
    46→ *   ├───────────────────────────────────────┤                          │
    47→ *   │ Text input + Send                     │                          │
    48→ *   │ [Brief me][Show patches][Explain…][Next]                         │
    49→ *   │ ▁▂▄█▆▃▁▂▄ waveform (when listening)                              │
    50→ *   └───────────────────────────────────────┴──────────────────────────┘
    51→ *
    52→ * Greeting behavior:
    53→ *
    54→ *   - On the first `open={true}` transition, fetch /api/agent-x/briefing
    55→ *     ONCE. Build a short greeting ("Good morning, Ayan. Your security
    56→ *     posture is excellent — score 100/100 (grade A). What are you up
    57→ *     to today?"). Speak it. Only start listening after the utterance
    58→ *     finishes (utterance.onend) — no 600ms timer.
    59→ *
    60→ * Proactive monitoring:
    61→ *
    62→ *   - Poll /api/agent-x/briefing every 5 minutes. Compare pending-patch
    63→ *     IDs (not counts). If a NEW patch ID appears, drop an alert message
    64→ *     + speak a heads-up. Never auto-explain per-tab.
    65→ */

   242→// ─── Voice persona: lower-pitched voice picker ───────────────────────────────
   243→
   244→function pickPersonaVoice(
   245→  voices: SpeechSynthesisVoice[],
   246→): SpeechSynthesisVoice | null {
   247→  if (!voices.length) return null;
   248→  const maleNames = /david|alex|daniel|fred|james|mark|oliver|george|rishi|arjun/i;
   249→  return (
   250→    voices.find((v) => /^en[-_]US/i.test(v.lang) && maleNames.test(v.name)) ||
   251→    voices.find((v) => /^en[-_]GB/i.test(v.lang) && maleNames.test(v.name)) ||
   252→    voices.find((v) => /^en/i.test(v.lang) && maleNames.test(v.name)) ||
   253→    voices.find((v) => /en[-_]US/i.test(v.lang) && /google|samantha|jenny/i.test(v.name)) ||
   254→    voices.find((v) => /^en/i.test(v.lang)) ||
   255→    voices[0] ||
   256→    null
   257→  );
   258→}

   260→// ─── Sentence splitter (for streaming TTS) ──────────────────────────────────
   261→// Splits a reply into speakable chunks. Splits on `.`, `!`, `?`, and
   262→// newlines, keeping the punctuation with the sentence. Filters out empty
   263→// chunks (e.g. trailing periods). Keeps the splitter simple — we don't
   264→// need NLP-grade segmentation for SOC analyst replies.
   265→
   266→function splitIntoSentences(text: string): string[] {
   267→  const trimmed = text.trim();
   268→  if (!trimmed) return [];
   269→  // Match runs that end with . ! ? or end-of-string, including the punctuation.
   270→  const matches = trimmed.match(/[^.!?]*[.!?]+|[^.!?]+$/g);
   271→  if (!matches) return [trimmed];
   272→  return matches
   273→    .map((s) => s.trim())
   274→    .filter((s) => s.length > 0);
   275→}

   293→// PERFORMANCE: `React.memo` wrapper so AgentX doesn't re-render on every
   294→// parent re-render (e.g. the 1Hz live clock in ConsoleView, the periodic
   295→// patch-list refresh, sidebar toggles). The props are stable in ConsoleView
   296→// (all callbacks are `useCallback`-wrapped; `currentTab`, `currentUser`, and
   297→// `open` are scalar). Only meaningful prop changes trigger a re-render.
   298→function AgentXInner({
   299→  currentTab,
   300→  currentUser,
   301→  onNavigate,
   302→  onScan,
   303→  onApprovePatch,
   304→  onSearch,
   305→  onOpenWarRoom,
   306→  open,
   307→  onClose,
   308→}: AgentXProps) {
   309→  // ── Conversation + UI state ─────────────────────────────────────────────
   310→  const [messages, setMessages] = useState<AgentMessage[]>([]);
   311→  const [thinking, setThinking] = useState(false);
   312→  const [input, setInput] = useState("");
   313→  const [briefing, setBriefing] = useState<AgentXBriefing | null>(null);
   314→  const [resumedSession, setResumedSession] = useState(false);
   315→  const [providerName, setProviderName] = useState<string>("…");
   316→
   317→  // ── Voice state ─────────────────────────────────────────────────────────
   318→  const [listening, setListening] = useState(false);
   319→  const [speakingLocal, setSpeakingLocal] = useState(false);
   320→  const [interim, setInterim] = useState("");
   321→  const [supported, setSupported] = useState(false);
   322→  const [voiceError, setVoiceError] = useState<string | null>(null);
   323→
   324→  // ── Refs mirroring state for use inside stable callbacks ────────────────
   325→  // The SpeechRecognition handlers are bound once; we need the latest
   326→  // `thinking` + `currentTab` + `messages` + `speaking` without re-subscribing.
   327→  const currentTabRef = useRef(currentTab);
   328→  const messagesRef = useRef<AgentMessage[]>(messages);
   329→  const thinkingRef = useRef(thinking);
   330→  const openRef = useRef(open);
   331→  const speakingRef = useRef(false);
   332→  const greetedRef = useRef(false);
   333→  const userStoppedRef = useRef(true); // true until the user (or greeting) starts listening
   334→  const latestTranscriptRef = useRef("");
   335→
   336→  // ── SpeechRecognition + Web Audio refs ──────────────────────────────────
   337→  const ctorRef = useRef<SpeechRecognitionCtor | null>(null);
   338→  const recRef = useRef<SpeechRecognitionLike | null>(null);
   339→  const voicesRef = useRef<SpeechSynthesisVoice[]>([]);
   340→  const audioCtxRef = useRef<AudioContext | null>(null);
   341→  const analyserRef = useRef<AnalyserNode | null>(null);
   342→  const micStreamRef = useRef<MediaStream | null>(null);
   343→  const rafRef = useRef<number | null>(null);
   344→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   345→
   346→  // ── Proactive-monitoring baseline (last seen patch IDs) ─────────────────
   347→  const lastPatchIdsRef = useRef<Set<string>>(new Set());
   348→
   349→  const scrollRef = useRef<HTMLDivElement>(null);
   350→
   351→  // ── Mirror state into refs ──────────────────────────────────────────────
   352→  useEffect(() => {
   353→    currentTabRef.current = currentTab;
   354→  }, [currentTab]);
   355→  useEffect(() => {
   356→    messagesRef.current = messages;
   357→  }, [messages]);
   358→  useEffect(() => {
   359→    thinkingRef.current = thinking;
   360→  }, [thinking]);
   361→  useEffect(() => {
   362→    openRef.current = open;
   363→  }, [open]);
   364→
   365→  // ── Feature detection (post-hydration to avoid SSR mismatch) ────────────
   366→  useEffect(() => {
   367→    ctorRef.current = getSpeechRecognitionCtor();
   368→    setSupported(!!ctorRef.current);
   369→
   370→    if (typeof window !== "undefined" && "speechSynthesis" in window) {
   371→      const loadVoices = () => {
   372→        voicesRef.current = window.speechSynthesis.getVoices();
   373→      };
   374→      loadVoices();
   375→      window.speechSynthesis.onvoiceschanged = loadVoices;
   376→      return () => {
   377→        window.speechSynthesis.onvoiceschanged = null;
   378→      };
   379→    }
   380→    return;
   381→  }, []);
   382→
   383→  // ── Restore prior conversation from localStorage on first open ──────────
   384→  const restoredRef = useRef(false);
   385→  useEffect(() => {
   386→    if (!open || restoredRef.current) return;
   387→    restoredRef.current = true;
   388→    try {
   389→      const raw = localStorage.getItem(STORAGE_KEY);
   390→      if (!raw) return;
   391→      const parsed = JSON.parse(raw) as AgentMessage[];
   392→      if (Array.isArray(parsed) && parsed.length > 0) {
   393→        setMessages(parsed.slice(-MAX_HISTORY));
   394→        setResumedSession(true);
   395→      }
   396→    } catch {
   397→      /* quota / privacy mode — ignore */
   398→    }
   399→  }, [open]);
   400→
   401→  // ── Persist conversation to localStorage ────────────────────────────────
   402→  useEffect(() => {
   403→    if (messages.length === 0) return;
   404→    try {
   405→      localStorage.setItem(
   406→        STORAGE_KEY,
   407→        JSON.stringify(messages.slice(-MAX_HISTORY)),
   408→      );
   409→    } catch {
   410→      /* quota / privacy mode — ignore */
   411→    }
   412→  }, [messages]);
   413→
   414→  // ── Auto-scroll to bottom on new messages ───────────────────────────────
   415→  useEffect(() => {
   416→    const el = scrollRef.current;
   417→    if (!el) return;
   418→    el.scrollTo({ top: el.scrollHeight, behavior: "smooth" });
   419→  }, [messages, thinking, interim]);
   420→
   421→  // ── Fetch LLM provider name once per activation ─────────────────────────
   422→  useEffect(() => {
   423→    if (!open) return;
   424→    let cancelled = false;
   425→    (async () => {
   426→      try {
   427→        const res = await fetch("/api/agent-x/provider");
   428→        if (!res.ok) return;
   429→        const data = (await res.json()) as { provider?: string };
   430→        if (cancelled || !data.provider) return;
   431→        setProviderName(data.provider);
   432→      } catch {
   433→        /* silent — badge just stays "…" */
   434→      }
   435→    })();
   436→    return () => {
   437→      cancelled = true;
   438→    };
   439→  }, [open]);
   440→
   441→  // ── Build a SpeechRecognition instance (lazy) ───────────────────────────
   442→  const ensureRecognition = useCallback((): SpeechRecognitionLike | null => {
   443→    if (!ctorRef.current) return null;
   444→    if (recRef.current) return recRef.current;
   445→    const rec = new ctorRef.current();
   446→    rec.lang = "en-US";
   447→    rec.continuous = true;
   448→    rec.interimResults = true;
   449→    rec.maxAlternatives = 1;
   450→
   451→    rec.onstart = () => {
   452→      userStoppedRef.current = false;
   453→      setListening(true);
   454→      setVoiceError(null);
   455→    };
   456→
   457→    rec.onsoundstart = () => {
   458→      // User started making noise — if we're speaking, that's an interrupt.
   459→      if (speakingRef.current) {
   460→        try {
   461→          window.speechSynthesis.cancel();
   462→        } catch {
   463→          /* noop */
   464→        }
   465→        speakingRef.current = false;
   466→        setSpeakingLocal(false);
   467→      }
   468→    };
   469→
   470→    rec.onspeechstart = () => {
   471→      // Same interrupt semantics — speech is a stronger signal than sound.
   472→      if (speakingRef.current) {
   473→        try {
   474→          window.speechSynthesis.cancel();
   475→        } catch {
   476→          /* noop */
   477→        }
   478→        speakingRef.current = false;
   479→        setSpeakingLocal(false);
   480→      }
   481→    };
   482→
   483→    rec.onsoundend = () => {
   484→      /* Just VAD bookkeeping — no action needed. */
   485→    };
   486→
   487→    rec.onresult = (ev: SpeechRecognitionEventLike) => {
   488→      const results = ev.results;
   489→      let interimText = "";
   490→      let finalText = "";
   491→      for (let i = ev.resultIndex; i < results.length; i++) {
   492→        const alt = results[i]?.[0];
   493→        if (!alt) continue;
   494→        const t = alt.transcript;
   495→        if (results[i].isFinal) finalText += t;
   496→        else interimText += t;
   497→      }
   498→      setInterim(interimText);
   499→      if (finalText.trim()) {
   500→        latestTranscriptRef.current = finalText.trim();
   501→        setInterim("");
   502→        // Send the final transcript to the chat endpoint.
   503→        void sendMessage(latestTranscriptRef.current);
   504→      }
   505→    };
   506→
   507→    rec.onerror = (ev: SpeechRecognitionErrorEventLike) => {
   508→      // "no-speech" and "aborted" are routine in continuous mode — don't surface.
   509→      if (ev.error === "no-speech" || ev.error === "aborted") return;
   510→      if (ev.error === "not-allowed" || ev.error === "service-not-allowed") {
   511→        setVoiceError("Mic permission denied");
   512→      } else if (ev.error === "network") {
   513→        setVoiceError("Network error during recognition");
   514→      }
   515→    };
   516→
   517→    rec.onend = () => {
   518→      setListening(false);
   519→      setInterim("");
   520→      // Auto-restart gate: don't restart if (a) the user explicitly stopped,
   521→      // (b) Agent X is currently speaking (let the utterance.onend handler
   522→      // restart), or (c) the panel is no longer open.
   523→      if (
   524→        !openRef.current ||
   525→        userStoppedRef.current ||
   526→        speakingRef.current
   527→      ) {
   528→        return;
   529→      }
   530→      try {
   531→        rec.start();
   532→      } catch {
   533→        /* InvalidStateError if start() raced — ignored; onend will fire again */
   534→      }
   535→    };
   536→
   537→    recRef.current = rec;
   538→    return rec;
   539→  }, []);
   540→
   541→  // ── Start listening ──────────────────────────────────────────────────────
   542→  const startListening = useCallback(() => {
   543→    const rec = ensureRecognition();
   544→    if (!rec) return;
   545→    userStoppedRef.current = false;
   546→    try {
   547→      rec.start();
   548→    } catch {
   549→      /* InvalidStateError if start() raced — ignored */
   550→    }
   551→    void ensureMicAnalyser();
   552→  }, [ensureRecognition]);
   553→
   554→  // ── Stop listening (user-initiated) ─────────────────────────────────────
   555→  const stopListening = useCallback(() => {
   556→    userStoppedRef.current = true;
   557→    const rec = recRef.current;
   558→    if (!rec) return;
   559→    try {
   560→      rec.stop();
   561→    } catch {
   562→      /* noop */
   563→    }
   564→    stopMicAnalyser();
   565→  }, []);
   566→
   567→  // ── Mic toggle (button click) ────────────────────────────────────────────
   568→  const micToggle = useCallback(() => {
   569→    if (listening) stopListening();
   570→    else startListening();
   571→  }, [listening, startListening, stopListening]);
   572→
   573→  // ── Web Audio API: wire up AnalyserNode on the mic stream ────────────────
   574→  // Used by the waveform canvas. Created lazily on first startListening and
   575→  // torn down on stopListening + on unmount.
   576→  const ensureMicAnalyser = useCallback(async () => {
   577→    if (analyserRef.current) return;
   578→    if (typeof window === "undefined") return;
   579→    if (!navigator.mediaDevices?.getUserMedia) return;
   580→    try {
   581→      const stream = await navigator.mediaDevices.getUserMedia({
   582→        audio: { echoCancellation: true, noiseSuppression: true },
   583→      });
   584→      micStreamRef.current = stream;
   585→      const Ctx =
   586→        window.AudioContext ||
   587→        (window as unknown as { webkitAudioContext?: typeof AudioContext })
   588→          .webkitAudioContext;
   589→      if (!Ctx) return;
   590→      const ctx = new Ctx();
   591→      audioCtxRef.current = ctx;
   592→      const src = ctx.createMediaStreamSource(stream);
   593→      const analyser = ctx.createAnalyser();
   594→      analyser.fftSize = 64;
   595→      analyser.smoothingTimeConstant = 0.7;
   596→      src.connect(analyser);
   597→      analyserRef.current = analyser;
   598→    } catch {
   599→      /* Permission denied or unavailable — waveform just stays empty */
   600→    }
   601→  }, []);
   602→
   603→  const stopMicAnalyser = useCallback(() => {
   604→    if (rafRef.current != null) {
   605→      cancelAnimationFrame(rafRef.current);
   606→      rafRef.current = null;
   607→    }
   608→    if (micStreamRef.current) {
   609→      for (const track of micStreamRef.current.getTracks()) {
   610→        try {
   611→          track.stop();
   612→        } catch {
   613→          /* noop */
   614→        }
   615→      }
   616→      micStreamRef.current = null;
   617→    }
   618→    if (audioCtxRef.current) {
   619→      try {
   620→        void audioCtxRef.current.close();
   621→      } catch {
   622→        /* noop */
   623→      }
   624→      audioCtxRef.current = null;
   625→    }
   626→    analyserRef.current = null;
   627→  }, []);
   628→
   629→  // ── Waveform render loop (only when listening && !speaking) ──────────────
   630→  useEffect(() => {
   631→    if (!listening || speakingLocal) {
   632→      if (rafRef.current != null) {
   633→        cancelAnimationFrame(rafRef.current);
   634→        rafRef.current = null;
   635→      }
   636→      // Clear the canvas when not actively drawing.
   637→      const canvas = canvasRef.current;
   638→      if (canvas) {
   639→        const ctx2d = canvas.getContext("2d");
   640→        if (ctx2d) {
   641→          ctx2d.clearRect(0, 0, canvas.width, canvas.height);
   642→        }
   643→      }
   644→      return;
   645→    }
   646→    const canvas = canvasRef.current;
   647→    if (!canvas) return;
   648→    const ctx2d = canvas.getContext("2d");
   649→    if (!ctx2d) return;
   650→
   651→    const draw = () => {
   652→      const analyser = analyserRef.current;
   653→      if (!analyser) {
   654→        rafRef.current = requestAnimationFrame(draw);
   655→        return;
   656→      }
   657→      const bufferLength = analyser.frequencyBinCount;
   658→      const dataArray = new Uint8Array(bufferLength);
   659→      analyser.getByteFrequencyData(dataArray);
   660→
   661→      const w = canvas.width;
   662→      const h = canvas.height;
   663→      ctx2d.clearRect(0, 0, w, h);
   664→
   665→      // Draw a centerline baseline.
   666→      const barCount = Math.min(bufferLength, 32);
   667→      const gap = 2;
   668→      const barWidth = (w - gap * (barCount - 1)) / barCount;
   669→      for (let i = 0; i < barCount; i++) {
   670→        const v = dataArray[i] / 255; // 0..1
   671→        const barH = Math.max(2, v * h);
   672→        const x = i * (barWidth + gap);
   673→        const y = (h - barH) / 2;
   674→        // Gradient: emerald → cyan.
   675→        const grad = ctx2d.createLinearGradient(0, y, 0, y + barH);
   676→        grad.addColorStop(0, "rgba(16,185,129,0.9)");
   677→        grad.addColorStop(1, "rgba(34,211,238,0.7)");
   678→        ctx2d.fillStyle = grad;
   679→        ctx2d.fillRect(x, y, barWidth, barH);
   680→      }
   681→      rafRef.current = requestAnimationFrame(draw);
   682→    };
   683→    rafRef.current = requestAnimationFrame(draw);
   684→    return () => {
   685→      if (rafRef.current != null) {
   686→        cancelAnimationFrame(rafRef.current);
   687→        rafRef.current = null;
   688→      }
   689→    };
   690→  }, [listening, speakingLocal]);
   691→
   692→  // ── Streaming TTS: speak sentences sequentially ──────────────────────────
   693→  // The first sentence is spoken immediately. Each utterance's `onend`
   694→  // triggers the next sentence. While speaking, `speakingRef` is true →
   695→  // the recognition `onend` auto-restart is gated off (no race).
   696→  // The mic stays live to the extent the browser allows — if the user
   697→  // starts talking, `onsoundstart` cancels TTS (see ensureRecognition).
   698→  const speakChunked = useCallback(
   699→    (text: string, opts?: { onAllDone?: () => void }) => {
   700→      if (typeof window === "undefined" || !("speechSynthesis" in window)) {
   701→        opts?.onAllDone?.();
   702→        return;
   703→      }
   704→      const trimmed = text.trim();
   705→      if (!trimmed) {
   706→        opts?.onAllDone?.();
   707→        return;
   708→      }
   709→      const synth = window.speechSynthesis;
   710→      synth.cancel(); // interrupt any in-flight utterance
   711→      const chunks = splitIntoSentences(trimmed);
   712→      if (chunks.length === 0) {
   713→        opts?.onAllDone?.();
   714→        return;
   715→      }
   716→      const voices = voicesRef.current.length
   717→        ? voicesRef.current
   718→        : synth.getVoices();
   719→      const preferred = pickPersonaVoice(voices);
   720→
   721→      speakingRef.current = true;
   722→      setSpeakingLocal(true);
   723→
   724→      // If recognition is currently running, stop it so the browser doesn't
   725→      // pick up its own TTS as "speech" (we'll restart on completion).
   726→      const wasListening = listening;
   727→      if (wasListening) {
   728→        try {
   729→          recRef.current?.stop();
   730→        } catch {
   731→          /* noop */
   732→        }
   733→      }
   734→
   735→      let idx = 0;
   736→      const speakNext = () => {
   737→        if (idx >= chunks.length) {
   738→          speakingRef.current = false;
   739→          setSpeakingLocal(false);
   740→          opts?.onAllDone?.();
   741→          // Restart listening if it was live before we started speaking.
   742→          if (wasListening && openRef.current && !userStoppedRef.current) {
   743→            try {
   744→              recRef.current?.start();
   745→            } catch {
   746→              /* InvalidStateError if start() raced — onend will retry */
   747→            }
   748→          }
   749→          return;
   750→        }
   751→        const chunk = chunks[idx];
   752→        idx += 1;
   753→        const u = new SpeechSynthesisUtterance(chunk);
   754→        if (preferred) u.voice = preferred;
   755→        u.rate = 0.95;
   756→        u.pitch = 0.85;
   757→        u.volume = 1;
   758→        u.onend = () => {
   759→          // Defer to next tick to avoid the synth racing ahead.
   760→          window.setTimeout(speakNext, 0);
   761→        };
   762→        u.onerror = () => {
   763→          speakingRef.current = false;
   764→          setSpeakingLocal(false);
   765→          opts?.onAllDone?.();
   766→        };
   767→        synth.speak(u);
   768→      };
   769→      speakNext();
   770→    },
   771→    [listening],
   772→  );
   773→
   774→  // ── Stop TTS (manual mute) ───────────────────────────────────────────────
   775→  const stopSpeaking = useCallback(() => {
   776→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   777→    window.speechSynthesis.cancel();
   778→    speakingRef.current = false;
   779→    setSpeakingLocal(false);
   780→  }, []);
   781→
   782→  // ── Execute Agent X actions returned from the chat endpoint ─────────────
   783→  const executeActions = useCallback(
   784→    (actions: AgentXAction[] | undefined) => {
   785→      if (!actions || actions.length === 0) return;
   786→      for (const action of actions) {
   787→        switch (action.type) {
   788→          case "navigate":
   789→            if (action.target) onNavigate?.(action.target);
   790→            break;
   791→          case "scan":
   792→            if (action.target) onScan?.(action.target);
   793→            break;
   794→          case "approve":
   795→          case "approve_patch":
   796→            if (action.target) onApprovePatch?.(action.target);
   797→            break;
   798→          case "search":
   799→            onSearch?.(action.target || action.query || "");
   800→            break;
   801→          case "war_room":
   802→            onOpenWarRoom?.();
   803→            break;
   804→          default:
   805→            // Unknown action types are silently ignored — the backend may
   806→            // emit intents the frontend hasn't wired up yet.
   807→            break;
   808→        }
   809→      }
   810→    },
   811→    [onApprovePatch, onNavigate, onOpenWarRoom, onScan, onSearch],
   812→  );
   813→
   814→  // ── Progressive message display ──────────────────────────────────────────
   815→  // Helper to append the next sentence to a specific agent message ID.
   816→  const appendToMessage = useCallback((id: string, chunk: string) => {
   817→    setMessages((prev) =>
   818→      prev.map((m) =>
   819→        m.id === id ? { ...m, content: m.content + chunk } : m,
   820→      ),
   821→    );
   822→  }, []);
   823→
   824→  // ── Send a message to /api/agent-x/chat ─────────────────────────────────
   825→  const sendMessage = useCallback(
   826→    async (text: string) => {
   827→      const trimmed = text.trim();
   828→      if (!trimmed || thinkingRef.current) return;
   829→
   830→      // Stop any in-flight TTS — the user is talking over Agent X.
   831→      if (speakingRef.current) {
   832→        stopSpeaking();
   833→      }
   834→
   835→      setInput("");
   836→      setThinking(true);
   837→      const userMsgId = `u-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   838→      const agentMsgId = `a-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   839→      setMessages((prev) => [
   840→        ...prev,
   841→        { id: userMsgId, role: "user", content: trimmed, ts: Date.now() },
   842→      ]);
   843→
   844→      try {
   845→        const res = await fetch("/api/agent-x/chat", {
   846→          method: "POST",
   847→          headers: { "Content-Type": "application/json" },
   848→          body: JSON.stringify({
   849→            message: trimmed,
   850→            context: {
   851→              currentTab: currentTabRef.current,
   852→              history: messagesRef.current
   853→                .slice(-10)
   854→                .map((m) => ({ role: m.role, content: m.content })),
   855→            },
   856→          }),
   857→        });
   858→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   859→        const data = (await res.json()) as AgentXChatResponse;
   860→
   861→        const reply = data.reply?.trim() || "I didn't catch that. Try again.";
   862→
   863→        // Stream the reply: split into sentences, display first sentence
   864→        // immediately, then progressively append the rest as TTS plays.
   865→        const sentences = splitIntoSentences(reply);
   866→        const firstSentence = sentences[0] || reply;
   867→        const rest = sentences.slice(1);
   868→
   869→        // Seed the agent message with the first sentence.
   870→        setMessages((prev) => [
   871→          ...prev,
   872→          {
   873→            id: agentMsgId,
   874→            role: "agent",
   875→            content: firstSentence,
   876→            ts: Date.now(),
   877→          },
   878→        ]);
   879→
   880→        executeActions(data.actions);
   881→
   882→        // Speak the first sentence immediately; subsequent sentences are
   883→        // queued by the chunked speaker (its onend triggers the next).
   884→        speakChunked(reply, {
   885→          onAllDone: () => {
   886→            // After all chunks have played, ensure the full text is in the
   887→            // message bubble (in case the chunker appended progressively
   888→            // faster/slower than expected — this is idempotent).
   889→          },
   890→        });
   891→
   892→        // Progressive display: append subsequent sentences at a cadence
   893→        // that roughly matches the speaking rate. We use a timer based on
   894→        // sentence length to avoid the message appearing all at once.
   895→        if (rest.length > 0) {
   896→          let delay = 600; // initial delay gives the first utterance a beat
   897→          for (const s of rest) {
   898→            window.setTimeout(() => {
   899→              appendToMessage(agentMsgId, " " + s);
   900→            }, delay);
   901→            delay += Math.max(800, s.length * 55);
   902→          }
   903→        }
   904→      } catch {
   905→        const errMsg =
   906→          "I'm having trouble reaching the security core. Try again in a moment.";
   907→        setMessages((prev) => [
   908→          ...prev,
   909→          {
   910→            id: `err-${Date.now()}`,
   911→            role: "agent",
   912→            content: errMsg,
   913→            ts: Date.now(),
   914→            kind: "error",
   915→          },
   916→        ]);
   917→        speakChunked(errMsg);
   918→      } finally {
   919→        setThinking(false);
   920→      }
   921→    },
   922→    [appendToMessage, executeActions, speakChunked, stopSpeaking],
   923→  );
   924→
   925→  // ── On open: fetch briefing ONCE, greet + speak, start listening after TTS ─
   926→  useEffect(() => {
   927→    if (!open) return;
   928→    if (greetedRef.current) return;
   929→    greetedRef.current = true;
   930→    let cancelled = false;
   931→
   932→    (async () => {
   933→      try {
   934→        const res = await fetch("/api/agent-x/briefing");
   935→        if (!res.ok) return;
   936→        const data = (await res.json()) as AgentXBriefing;
   937→        if (cancelled) return;
   938→        setBriefing(data);
   939→
   940→        // Seed the proactive-monitoring baseline with patch IDs (not counts).
   941→        const patchIds = new Set<string>(
   942→          (data.pendingTasks ?? [])
   943→            .filter((t) => t.type === "patch")
   944→            .map((t) => t.id),
   945→        );
   946→        lastPatchIdsRef.current = patchIds;
   947→
   948→        const firstName = (currentUser?.name || "there").split(" ")[0];
   949→        const postureSummary = buildPostureSummary(data);
   950→        const greeting = `Good ${data.timeOfDay || "day"}, ${firstName}. ${postureSummary} What are you up to today?`;
   951→
   952→        setMessages((prev) => [
   953→          ...prev,
   954→          {
   955→            id: `greeting-${Date.now()}`,
   956→            role: "agent",
   957→            content: greeting,
   958→            ts: Date.now(),
   959→            kind: "briefing",
   960→          },
   961→        ]);
   962→
   963→        // Speak greeting. After TTS finishes (utterance.onend → onAllDone),
   964→        // start listening. This guarantees the mic opens only after the
   965→        // greeting is done — no race with TTS playback.
   966→        speakChunked(greeting, {
   967→          onAllDone: () => {
   968→            if (cancelled || !openRef.current) return;
   969→            startListening();
   970→          },
   971→        });
   972→      } catch {
   973→        /* network/silent failure — tab still usable via text input */
   974→      }
   975→    })();
   976→
   977→    return () => {
   978→      cancelled = true;
   979→    };
   980→    // Intentionally only depend on `open` — we want the greeting to fire
   981→    // exactly once per activation. currentUser + speakChunked + startListening
   982→    // are stable or accessed via refs.
   983→  }, [open, currentUser, speakChunked, startListening]);
   984→
   985→  // ── Proactive monitoring: poll briefing every 5 min, alert on NEW patch IDs ─
   986→  useEffect(() => {
   987→    if (!open) return;
   988→    const id = window.setInterval(async () => {
   989→      if (!openRef.current) return;
   990→      try {
   991→        const res = await fetch("/api/agent-x/briefing");
   992→        if (!res.ok) return;
   993→        const data = (await res.json()) as AgentXBriefing;
   994→        setBriefing(data);
   995→
   996→        const currentPatchIds = new Set<string>(
   997→          (data.pendingTasks ?? [])
   998→            .filter((t) => t.type === "patch")
   999→            .map((t) => t.id),
  1000→        );
  1001→
  1002→        // Find patch IDs that exist now but weren't in the baseline.
  1003→        const newIds: string[] = [];
  1004→        for (const id2 of currentPatchIds) {
  1005→          if (!lastPatchIdsRef.current.has(id2)) newIds.push(id2);
  1006→        }
  1007→
  1008→        if (newIds.length > 0) {
  1009→          const topNew =
  1010→            (data.pendingTasks ?? []).find((t) => t.id === newIds[0])?.title ||
  1011→            "a new patch";
  1012→          const msg = `Heads up — a new critical patch was just generated: ${topNew}. Want me to show you?`;
  1013→          setMessages((prev) => [
  1014→            ...prev,
  1015→            {
  1016→              id: `alert-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
  1017→              role: "agent",
  1018→              content: msg,
  1019→              ts: Date.now(),
  1020→              kind: "alert",
  1021→            },
  1022→          ]);
  1023→          speakChunked(msg);
  1024→        }
  1025→
  1026→        lastPatchIdsRef.current = currentPatchIds;
  1027→      } catch {
  1028→        /* swallow — proactive polling must never crash the tab */
  1029→      }
  1030→    }, PROACTIVE_POLL_MS);
  1031→    return () => window.clearInterval(id);
  1032→  }, [open, speakChunked]);
  1033→
  1034→  // ── Stop TTS + listening when the panel closes ──────────────────────────
  1035→  useEffect(() => {
  1036→    if (open) return;
  1037→    if (typeof window !== "undefined" && "speechSynthesis" in window) {
  1038→      window.speechSynthesis.cancel();
  1039→    }
  1040→    speakingRef.current = false;
  1041→    setSpeakingLocal(false);
  1042→    userStoppedRef.current = true;
  1043→    try {
  1044→      recRef.current?.stop();
  1045→    } catch {
  1046→      /* noop */
  1047→    }
  1048→    stopMicAnalyser();
  1049→  }, [open, stopMicAnalyser]);
  1050→
  1051→  // ── Cleanup TTS + recognition on unmount ──────────────────────────────────
  1052→  useEffect(() => {
  1053→    return () => {
  1054→      userStoppedRef.current = true;
  1055→      try {
  1056→        recRef.current?.abort();
  1057→      } catch {
  1058→        /* noop */
  1059→      }
  1060→      if (typeof window !== "undefined" && "speechSynthesis" in window) {
  1061→        window.speechSynthesis.cancel();
  1062→      }
  1063→      stopMicAnalyser();
  1064→    };
  1065→  }, [stopMicAnalyser]);
  1066→
  1067→  // ── Conversation export (.txt) ───────────────────────────────────────────
  1068→  const exportConversation = useCallback(() => {
  1069→    if (messages.length === 0) return;
  1070→    const lines = messages.map((m) => {
  1071→      const when = new Date(m.ts).toISOString();

  1074→      return `[${when}] ${who}${kindTag}:\n${m.content}`;

  1078→      `Exported: ${new Date().toISOString()}`,
  1079→      `Messages: ${messages.length}`,
  1080→      "",
  1081→      "─".repeat(60),
  1082→      "",

  1091→    document.body.appendChild(a);

  1095→  }, [messages]);
  1096→
  1097→  // ── Clear conversation ────────────────────────────────────────────────────
  1098→  const clearConversation = useCallback(() => {
  1099→    setMessages([]);
  1100→    setResumedSession(false);
  1101→    try {
  1102→      localStorage.removeItem(STORAGE_KEY);
  1103→    } catch {
  1104→      /* noop */
  1105→    }
  1106→  }, []);
  1107→

  1109→  const speaking = speakingLocal;
  1110→

  1112→  if (!open) return null;
