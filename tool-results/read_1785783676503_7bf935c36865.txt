   578→async function http<T>(url: string, init?: RequestInit): Promise<T> {
   579→  // Get JWT token from localStorage (set on login)
   580→  const token = typeof window !== "undefined" ? localStorage.getItem("guardianx-token") : null;
   581→
   582→  const res = await fetch(url, {
   583→    ...init,
   584→    credentials: "same-origin", // send HTTP-only cookie
   585→    headers: {
   586→      "Content-Type": "application/json",
   587→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   588→      ...(init?.headers ?? {}),
   589→    },
   590→  });
   591→
   592→  // Handle 401, token expired, redirect to login
   593→  if (res.status === 401 && typeof window !== "undefined") {
   594→    localStorage.removeItem("guardianx-user");
   595→    localStorage.removeItem("guardianx-token");
   596→    localStorage.setItem("guardianx-view", "auth");
   597→    window.location.reload();
   598→    throw new Error("Session expired. Please log in again.");
   599→  }
   600→
   601→  const data = (await res.json().catch(() => ({}))) as T & { error?: string };
   602→  if (!res.ok) {
   603→    throw new Error(data?.error ?? `Request failed (${res.status})`);
   604→  }
   605→  return data as T;
   606→}

   608→export const sentinelApi = {
   609→  // codebases
   610→  listCodebases: () => http<Codebase[]>("/api/codebases"),
   611→  getCodebase: (id: string) => http<CodebaseDetail>(`/api/codebases/${id}`),
   612→  createCodebase: (data: {
   613→    name: string;
   614→    sourceCode: string;

   617→    clientId?: string;
   618→  }) =>
   619→    http<Codebase>("/api/codebases", {
   620→      method: "POST",
   621→      body: JSON.stringify(data),
   622→    }),
   623→  deleteCodebase: (id: string) =>
   624→    http<{ ok: boolean }>(`/api/codebases/${id}`, { method: "DELETE" }),
   625→
   626→  // scans
   627→  listScans: () => http<Scan[]>("/api/scans"),
   628→  startScan: (codebaseId: string) =>
   629→    http<{ scanId: string; status: string }>("/api/scans", {
   630→      method: "POST",
   631→      body: JSON.stringify({ codebaseId }),
   632→    }),
   633→  getScanEvents: (scanId: string) =>
   634→    http<PipelineEvent[]>(`/api/scans/${scanId}/events`),
   635→
   636→  // patches
   637→  listPending: () => http<PatchSummary[]>("/api/patches/pending"),
   638→  getPatch: (patchId: string) =>
   639→    http<PatchDetail>(`/api/patches/${encodeURIComponent(patchId)}`),
   640→  approve: (patchId: string) =>
   641→    http<{ patch_id: string; status: string; message: string }>(
   642→      `/api/patches/${encodeURIComponent(patchId)}/approve`,
   643→      { method: "POST" }
   644→    ),
   645→  reject: (patchId: string) =>
   646→    http<{ patch_id: string; status: string; message: string }>(
   647→      `/api/patches/${encodeURIComponent(patchId)}/reject`,
   648→      { method: "POST" }
   649→    ),
   650→  chat: (patchId: string, message: string) =>
   651→    http<{ role: "assistant"; content: string; created_at: string }>(
   652→      `/api/patches/${encodeURIComponent(patchId)}/chat`,
   653→      { method: "POST", body: JSON.stringify({ message }) }
   654→    ),
