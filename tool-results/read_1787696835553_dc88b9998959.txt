     3→     3→/**
     4→     4→ * AgentX
     5→     5→ * ------
     6→     6→ * Always-on conversational AI for the GuardianX Command Center.
     7→     7→ *
     8→     8→ * Voice infrastructure: the War Room's <VoiceControl> component is reused
     9→     9→ * purely as an STT + TTS primitive. We mount it visually-hidden (sr-only)
    10→    10→ * with `continuous` + `speakResponses={false}` so it owns the
    11→    11→ * SpeechRecognition instance and auto-restarts on browser silence
    12→    12→ * timeouts. Its `onCommand` callback hands us every parsed utterance;
    13→    13→ * for unrecognized input we get `{action: "unknown", raw: transcript}`
    14→    14→ * which we forward to `/api/agent-x/chat` for an LLM/heuristic reply.
    15→    15→ *
    16→    16→ * Talkback: we do NOT use VoiceControl's `speak()` for the LLM reply
    17→    17→ * because its default rate is 1.05 / pitch 0.95. Agent X is a "sophisticated
    18→    18→ * + lethal" SOC analyst — we want a slower, lower-pitched, authoritative
    19→    19→ * voice. So we run our own SpeechSynthesisUtterance with rate 0.95 / pitch
    20→    20→ * 0.85 and pick a male-coded voice when available.
    21→    21→ *
    22→    22→ * Continuous listening: once activated (open=true) the mic stays hot
    23→    23→ * until the panel is closed. No per-command mic tap.
    24→    24→ *
    25→    25→ * Proactive monitoring: every 60s while open we re-fetch
    26→    26→ * `/api/agent-x/briefing` and, if pending patches appeared or the
    27→    27→ * posture score dropped, we speak a heads-up aloud + drop an alert
    28→    28→ * message in the conversation.
    29→    29→ *
    30→    30→ * Tab awareness: when `currentTab` changes we fetch
    31→    31→ * `/api/agent-x/context?tab={tab}` and surface 2–3 contextual
    32→    32→ * quick-reply chips.
    33→    33→ *
    34→    34→ * Persistence: the last 20 messages are mirrored to
    35→    35→ * localStorage `agent_x_conversation` so a reload while the panel is
    36→    36→ * open picks up where the user left off.
    37→    37→ */

   164→   164→// ─── Voice persona: lower-pitched voice picker ───────────────────────────────
   165→   165→// SpeechSynthesis voice names vary by OS/browser. We prefer a male-coded
   166→   166→// en-US voice (David/Alex/Daniel/Fred/etc.) for an authoritative SOC
   167→   167→// analyst persona. Falls back to any en voice, then any voice at all.
   168→   168→
   169→   169→function pickPersonaVoice(voices: SpeechSynthesisVoice[]): SpeechSynthesisVoice | null {
   170→   170→  if (!voices.length) return null;
   171→   171→  const maleNames = /david|alex|daniel|fred|james|mark|oliver|george|rishi|arjun/i;
   172→   172→  return (
   173→   173→    voices.find((v) => /^en[-_]US/i.test(v.lang) && maleNames.test(v.name)) ||
   174→   174→    voices.find((v) => /^en[-_]GB/i.test(v.lang) && maleNames.test(v.name)) ||
   175→   175→    voices.find((v) => /^en/i.test(v.lang) && maleNames.test(v.name)) ||
   176→   176→    voices.find((v) => /en[-_]US/i.test(v.lang) && /google|samantha|jenny/i.test(v.name)) ||
   177→   177→    voices.find((v) => /^en/i.test(v.lang)) ||
   178→   178→    voices[0] ||
   179→   179→    null
   180→   180→  );
   181→   181→}
   182→   182→
   183→   183→// ─── Reconstruct a synthetic transcript from a parsed VoiceCommand ────────────
   184→   184→// VoiceControl's parser strips intent + target apart. To preserve the
   185→   185→// natural-language phrasing for the Agent X LLM, we rebuild a short
   186→   186→// utterance that mirrors the original intent. The LLM can then echo it
   187→   187→// back contextually.
   188→   188→
   189→   189→function reconstructTranscript(cmd: VoiceCommand): string | null {
   190→   190→  switch (cmd.action) {
   191→   191→    case "scan":
   192→   192→      return `scan ${cmd.target}`;
   193→   193→    case "navigate":
   194→   194→      return `show ${cmd.target}`;
   195→   195→    case "approve":
   196→   196→      return `approve patch ${cmd.target}`;
   197→   197→    case "search":
   198→   198→      return `search for ${cmd.target}`;
   199→   199→    case "status":
   200→   200→      return "What's my current security posture?";
   201→   201→    case "unknown":
   202→   202→      return cmd.raw;
   203→   203→    case "stop":
   204→   204→      return null; // signal: cancel TTS only, no Agent X round-trip
   205→   205→    default:
   206→   206→      return null;
   207→   207→  }
   208→   208→}

   212→   212→export function AgentX({
   213→   213→  currentTab,
   214→   214→  currentUser,
   215→   215→  onNavigate,
   216→   216→  onScan,
   217→   217→  onApprovePatch,
   218→   218→  onSearch,
   219→   219→  onOpenWarRoom,
   220→   220→  open,
   221→   221→  onClose,
   222→   222→}: AgentXProps) {
   223→   223→  const voiceRef = useRef<VoiceControlHandle>(null);
   224→   224→  const voicesRef = useRef<SpeechSynthesisVoice[]>([]);
   225→   225→
   226→   226→  // ── Voice state mirror (driven by VoiceControl's onStateChange) ─────────
   227→   227→  const [voiceState, setVoiceState] = useState<VoiceControlState>({
   228→   228→    listening: false,
   229→   229→    speaking: false,
   230→   230→    interim: "",
   231→   231→    supported: true,
   232→   232→  });
   233→   233→
   234→   234→  // ── Local TTS speaking flag — driven by our own speakReply() so the
   235→   235→  //    "Agent X is speaking" indicator tracks our slower-rate utterances
   236→   236→  //    even though VoiceControl's own speaking flag stays false.
   237→   237→  const [speakingLocal, setSpeakingLocal] = useState(false);
   238→   238→
   239→   239→  // ── Conversation + UI state ─────────────────────────────────────────────
   240→   240→  const [messages, setMessages] = useState<AgentMessage[]>([]);
   241→   241→  const [thinking, setThinking] = useState(false);
   242→   242→  const [input, setInput] = useState("");
   243→   243→  const [briefing, setBriefing] = useState<AgentXBriefing | null>(null);
   244→   244→  const [tabContext, setTabContext] = useState<AgentXTabContext | null>(null);
   245→   245→  const [suggestions, setSuggestions] = useState<string[]>([]);
   246→   246→  const [resumedSession, setResumedSession] = useState(false);
   247→   247→
   248→   248→  // ── Proactive-monitoring baseline (last seen counts) ────────────────────
   249→   249→  const lastPatchCountRef = useRef<number | null>(null);
   250→   250→  const lastPostureScoreRef = useRef<number | null>(null);
   251→   251→
   252→   252→  // ── Refs mirroring state for use inside stable callbacks ────────────────
   253→   253→  // The VoiceControl onCommand callback is created once; we need the
   254→   254→  // latest `thinking` + `currentTab` + `messages` without re-subscribing
   255→   255→  // VoiceControl's effect on every render.
   256→   256→  const currentTabRef = useRef(currentTab);
   257→   257→  const messagesRef = useRef<AgentMessage[]>(messages);
   258→   258→  const thinkingRef = useRef(thinking);
   259→   259→  const openRef = useRef(open);
   260→   260→
   261→   261→  useEffect(() => {
   262→   262→    currentTabRef.current = currentTab;
   263→   263→  }, [currentTab]);
   264→   264→  useEffect(() => {
   265→   265→    messagesRef.current = messages;
   266→   266→  }, [messages]);
   267→   267→  useEffect(() => {
   268→   268→    thinkingRef.current = thinking;
   269→   269→  }, [thinking]);
   270→   270→  useEffect(() => {
   271→   271→    openRef.current = open;
   272→   272→  }, [open]);
   273→   273→
   274→   274→  const scrollRef = useRef<HTMLDivElement>(null);
   275→   275→
   276→   276→  // ── Warm TTS voices (some browsers load them async) ────────────────────
   277→   277→  useEffect(() => {
   278→   278→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   279→   279→    const loadVoices = () => {
   280→   280→      voicesRef.current = window.speechSynthesis.getVoices();
   281→   281→    };
   282→   282→    loadVoices();
   283→   283→    window.speechSynthesis.onvoiceschanged = loadVoices;
   284→   284→    return () => {
   285→   285→      window.speechSynthesis.onvoiceschanged = null;
   286→   286→    };
   287→   287→  }, []);
   288→   288→
   289→   289→  // ── Restore prior conversation from localStorage on first open ──────────
   290→   290→  // We do this ONCE per mount (not on every open) so a reload while
   291→   291→  // already-open picks up where the user left off, but re-opening after
   292→   292→  // a close keeps the existing in-memory messages.
   293→   293→  const restoredRef = useRef(false);
   294→   294→  useEffect(() => {
   295→   295→    if (!open || restoredRef.current) return;
   296→   296→    restoredRef.current = true;
   297→   297→    try {
   298→   298→      const raw = localStorage.getItem(STORAGE_KEY);
   299→   299→      if (!raw) return;
   300→   300→      const parsed = JSON.parse(raw) as AgentMessage[];
   301→   301→      if (Array.isArray(parsed) && parsed.length > 0) {
   302→   302→        setMessages(parsed.slice(-MAX_HISTORY));
   303→   303→        setResumedSession(true);
   304→   304→      }
   305→   305→    } catch {
   306→   306→      /* quota / privacy mode — ignore */
   307→   307→    }
   308→   308→  }, [open]);
   309→   309→
   310→   310→  // ── Persist conversation to localStorage ────────────────────────────────
   311→   311→  useEffect(() => {
   312→   312→    if (messages.length === 0) return;
   313→   313→    try {
   314→   314→      localStorage.setItem(STORAGE_KEY, JSON.stringify(messages.slice(-MAX_HISTORY)));
   315→   315→    } catch {
   316→   316→      /* quota / privacy mode — ignore */
   317→   317→    }
   318→   318→  }, [messages]);
   319→   319→
   320→   320→  // ── Auto-scroll to bottom on new messages ───────────────────────────────
   321→   321→  useEffect(() => {
   322→   322→    const el = scrollRef.current;
   323→   323→    if (!el) return;
   324→   324→    el.scrollTo({ top: el.scrollHeight, behavior: "smooth" });
   325→   325→  }, [messages, thinking]);
   326→   326→
   327→   327→  // ── Voice persona talkback ──────────────────────────────────────────────
   328→   328→  // rate 0.95 (slower for clarity) + pitch 0.85 (lower, authoritative).
   329→   329→  // Picks a male-coded en voice when available so Agent X sounds like a
   330→   330→  // seasoned SOC analyst rather than a default assistant chirp.
   331→   331→  const speakReply = useCallback((text: string) => {
   332→   332→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   333→   333→    const trimmed = text.trim();
   334→   334→    if (!trimmed) return;
   335→   335→    const synth = window.speechSynthesis;
   336→   336→    synth.cancel(); // interrupt any in-flight utterance
   337→   337→    const voices = voicesRef.current.length ? voicesRef.current : synth.getVoices();
   338→   338→    const preferred = pickPersonaVoice(voices);
   339→   339→    const u = new SpeechSynthesisUtterance(trimmed);
   340→   340→    if (preferred) u.voice = preferred;
   341→   341→    u.rate = 0.95;
   342→   342→    u.pitch = 0.85;
   343→   343→    u.volume = 1;
   344→   344→    u.onstart = () => setSpeakingLocal(true);
   345→   345→    u.onend = () => setSpeakingLocal(false);
   346→   346→    u.onerror = () => setSpeakingLocal(false);
   347→   347→    synth.speak(u);
   348→   348→  }, []);
   349→   349→
   350→   350→  const stopSpeaking = useCallback(() => {
   351→   351→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   352→   352→    window.speechSynthesis.cancel();
   353→   353→    setSpeakingLocal(false);
   354→   354→  }, []);
   355→   355→
   356→   356→  // ── Execute Agent X actions returned from the chat endpoint ─────────────
   357→   357→  const executeActions = useCallback(
   358→   358→    (actions: AgentXAction[] | undefined) => {
   359→   359→      if (!actions || actions.length === 0) return;
   360→   360→      for (const action of actions) {
   361→   361→        switch (action.type) {
   362→   362→          case "navigate":
   363→   363→            onNavigate?.(action.target);
   364→   364→            break;
   365→   365→          case "scan":
   366→   366→            onScan?.(action.target);
   367→   367→            break;
   368→   368→          case "approve_patch":
   369→   369→            onApprovePatch?.(action.target);
   370→   370→            break;
   371→   371→          case "search":
   372→   372→            onSearch?.(action.target);
   373→   373→            break;
   374→   374→          case "war_room":
   375→   375→            onOpenWarRoom?.();
   376→   376→            break;
   377→   377→          default:
   378→   378→            // Unknown action types are silently ignored — the backend may
   379→   379→            // emit intents the frontend hasn't wired up yet.
   380→   380→            break;
   381→   381→        }
   382→   382→      }
   383→   383→    },
   384→   384→    [onApprovePatch, onNavigate, onOpenWarRoom, onScan, onSearch],
   385→   385→  );
   386→   386→
   387→   387→  // ── Send a message to /api/agent-x/chat ─────────────────────────────────
   388→   388→  const sendMessage = useCallback(
   389→   389→    async (text: string) => {
   390→   390→      const trimmed = text.trim();
   391→   391→      if (!trimmed || thinkingRef.current) return;
   392→   392→
   393→   393→      setInput("");
   394→   394→      setThinking(true);
   395→   395→      setMessages((prev) => [
   396→   396→        ...prev,
   397→   397→        { role: "user", content: trimmed, ts: Date.now() },
   398→   398→      ]);
   399→   399→
   400→   400→      try {
   401→   401→        const res = await fetch("/api/agent-x/chat", {
   402→   402→          method: "POST",
   403→   403→          headers: { "Content-Type": "application/json" },
   404→   404→          body: JSON.stringify({
   405→   405→            message: trimmed,
   406→   406→            context: {
   407→   407→              currentTab: currentTabRef.current,
   408→   408→              history: messagesRef.current
   409→   409→                .slice(-10)
   410→   410→                .map((m) => ({ role: m.role, content: m.content })),
   411→   411→            },
   412→   412→          }),
   413→   413→        });
   414→   414→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   415→   415→        const data = (await res.json()) as AgentXChatResponse;
   416→   416→
   417→   417→        const reply = data.reply?.trim() || "I didn't catch that. Try again.";
   418→   418→        setMessages((prev) => [
   419→   419→          ...prev,
   420→   420→          { role: "agent", content: reply, ts: Date.now() },
   421→   421→        ]);
   422→   422→        speakReply(reply);
   423→   423→        executeActions(data.actions);
   424→   424→        if (Array.isArray(data.suggestions) && data.suggestions.length > 0) {
   425→   425→          setSuggestions(data.suggestions.slice(0, 4));
   426→   426→        }
   427→   427→      } catch {
   428→   428→        const errMsg =
   429→   429→          "I'm having trouble reaching the security core. Try again in a moment.";
   430→   430→        setMessages((prev) => [
   431→   431→          ...prev,
   432→   432→          { role: "agent", content: errMsg, ts: Date.now(), kind: "error" },
   433→   433→        ]);
   434→   434→        speakReply(errMsg);
   435→   435→      } finally {
   436→   436→        setThinking(false);
   437→   437→      }
   438→   438→    },
   439→   439→    [executeActions, speakReply],
   440→   440→  );
   441→   441→
   442→   442→  // ── VoiceControl onCommand — receives parsed VoiceCommands ──────────────
   443→   443→  // For `unknown` actions we forward `raw` to Agent X. For built-in
   444→   444→  // commands we reconstruct a synthetic transcript so the LLM stays in
   445→   445→  // the loop and can produce a contextual reply. `stop` cancels TTS only.
   446→   446→  const handleVoiceCommand = useCallback(
   447→   447→    (cmd: VoiceCommand) => {
   448→   448→      if (cmd.action === "stop") {
   449→   449→        stopSpeaking();
   450→   450→        return;
   451→   451→      }
   452→   452→      const transcript = reconstructTranscript(cmd);
   453→   453→      if (transcript) void sendMessage(transcript);
   454→   454→    },
   455→   455→    [sendMessage, stopSpeaking],
   456→   456→  );
   457→   457→
   458→   458→  const handleStateChange = useCallback((next: VoiceControlState) => {
   459→   459→    setVoiceState(next);
   460→   460→  }, []);
   461→   461→
   462→   462→  // ── On open: fetch briefing, speak greeting, auto-start listening ───────
   463→   463→  useEffect(() => {
   464→   464→    if (!open) return;
   465→   465→    let cancelled = false;
   466→   466→
   467→   467→    (async () => {
   468→   468→      try {
   469→   469→        const res = await fetch("/api/agent-x/briefing");
   470→   470→        if (!res.ok) return;
   471→   471→        const data = (await res.json()) as AgentXBriefing;
   472→   472→        if (cancelled) return;
   473→   473→        setBriefing(data);
   474→   474→        lastPatchCountRef.current = data.pendingTasks ?? null;
   475→   475→        lastPostureScoreRef.current = data.postureScore ?? null;
   476→   476→
   477→   477→        const greeting =
   478→   478→          data.greeting ||
   479→   479→          (currentUser?.name
   480→   480→            ? `Good to see you, ${currentUser.name}. Agent X online.`
   481→   481→            : "Agent X online. Ready when you are.");
