    25→ * ----------
    26→ * A prominent "Scan Your Website Free" section. User enters a URL → a
    27→ * SIMULATED 30-second scan runs through 4 progress phases → 3-5 realistic

    33→ *    recon-tools Docker service + explicit authorization.

    35→ *    authorized=false) so sales can see which URLs visitors probed. If the
    36→ *    API is unreachable, we silently fall back to mock mode.
    37→ *  - Rate-limited client-side: 1 scan per browser per hour via localStorage
    38→ *    key `gx_scan_last_run`.
    39→ *  - Email leads stored in localStorage key `gx_leads`.
    40→ */

   156→const RATE_LIMIT_MS = 60 * 60 * 1000; // 1 scan per hour
   157→const SCAN_TOTAL_MS = 30_000;
   158→
   159→function normalizeUrl(raw: string): string | null {
   160→  let u = raw.trim();
   161→  if (!u) return null;
   162→  if (!/^https?:\/\//i.test(u)) u = `https://${u}`;
   163→  try {
   164→    const parsed = new URL(u);
   165→    if (!parsed.hostname.includes(".")) return null;
   166→    return parsed.toString();
   167→  } catch {
   168→    return null;
   169→  }
   170→}
   171→
   172→function timeLeftLabel(ms: number): string {
   173→  const mins = Math.ceil(ms / 60000);
   174→  if (mins <= 1) return "less than a minute";
   175→  return `${mins} minutes`;
   176→}
   177→
   178→function isValidEmail(email: string): boolean {
   179→  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim());
   180→}

   182→export function ScanWidget({ onEnter }: { onEnter: () => void }) {
   183→  const [url, setUrl] = useState("");
   184→  const [phase, setPhase] = useState<Phase>("idle");
   185→  const [error, setError] = useState<string | null>(null);
   186→  const [scanPhaseIdx, setScanPhaseIdx] = useState(0);
   187→  const [progress, setProgress] = useState(0); // 0..100
   188→  const [findings, setFindings] = useState<SampleFinding[]>([]);
   189→  const [email, setEmail] = useState("");
   190→  const [targetCreated, setTargetCreated] = useState(false);
   191→  const [rateLimitedUntil, setRateLimitedUntil] = useState<number | null>(null);
   192→  const [now, setNow] = useState(Date.now());
   193→
   194→  // Tick clock so the rate-limit countdown refreshes in UI.
   195→  useEffect(() => {
   196→    if (rateLimitedUntil === null) return;
   197→    const t = setInterval(() => setNow(Date.now()), 1000);
   198→    return () => clearInterval(t);
   199→  }, [rateLimitedUntil]);
   200→
   201→  // On mount, hydrate rate-limit window from localStorage.
   202→  useEffect(() => {
   203→    try {
   204→      const last = Number(localStorage.getItem("gx_scan_last_run") || "0");
   205→      if (last && Date.now() - last < RATE_LIMIT_MS) {
   206→        setRateLimitedUntil(last + RATE_LIMIT_MS);
   207→      }
   208→    } catch {
   209→      // localStorage unavailable (SSR / privacy mode) — proceed without it.
   210→    }
   211→  }, []);
   212→
   213→  const canScan = useMemo(() => {
   214→    if (rateLimitedUntil === null) return true;
   215→    return Date.now() >= rateLimitedUntil;
   216→  }, [rateLimitedUntil, now]);
   217→
   218→  const startScan = async () => {
   219→    setError(null);
   220→    if (!canScan) {
   221→      setError(`Rate limit reached — try again in ${timeLeftLabel(rateLimitedUntil! - now)}.`);
   222→      return;
   223→    }
   224→    const normalized = normalizeUrl(url);
   225→    if (!normalized) {
   226→      setError("Please enter a valid URL (e.g. example.com or https://example.com).");
   227→      return;
   228→    }
   229→    setUrl(normalized);
   230→
   231→    // Best-effort: record this URL as a Target so the sales team can see what
   232→    // visitors are probing. Failure is fine — we proceed with the mock scan.
   233→    try {
   234→      const res = await fetch("/api/targets", {
   235→        method: "POST",
   236→        headers: { "Content-Type": "application/json" },
   237→        body: JSON.stringify({
   238→          name: normalized,
   239→          baseUrl: normalized,
   240→          authorized: false,
   241→          notes: "Submitted via homepage ScanWidget (free scan lead).",
   242→        }),
   243→      });
   244→      setTargetCreated(res.ok);
   245→    } catch {
   246→      setTargetCreated(false);
   247→    }
   248→
   249→    // Persist rate-limit stamp immediately.
   250→    try {
   251→      localStorage.setItem("gx_scan_last_run", String(Date.now()));
   252→      setRateLimitedUntil(Date.now() + RATE_LIMIT_MS);
   253→    } catch {
   254→      // ignore
   255→    }
   256→
   257→    // Pick 3-5 findings at random (deterministic-ish: shuffle + slice).
   258→    const shuffled = [...SAMPLE_FINDINGS].sort(() => Math.random() - 0.5);
   259→    const count = 3 + Math.floor(Math.random() * 3); // 3..5
   260→    setFindings(shuffled.slice(0, count));
   261→
   262→    // Begin the simulated scan.
   263→    setPhase("scanning");
   264→    setScanPhaseIdx(0);
   265→    setProgress(0);
   266→  };
   267→
   268→  // Drive the simulated scan progress.
   269→  useEffect(() => {
   270→    if (phase !== "scanning") return;
   271→    const phaseDuration = SCAN_TOTAL_MS / SCAN_PHASES.length;
   272→    const startedAt = Date.now();
   273→    let raf = 0;
   274→    const tick = () => {
   275→      const elapsed = Date.now() - startedAt;
   276→      const pct = Math.min(100, (elapsed / SCAN_TOTAL_MS) * 100);
   277→      setProgress(pct);
   278→      const idx = Math.min(SCAN_PHASES.length - 1, Math.floor(elapsed / phaseDuration));
   279→      setScanPhaseIdx(idx);
   280→      if (elapsed >= SCAN_TOTAL_MS) {
   281→        setPhase("findings");
   282→        return;
   283→      }
   284→      raf = requestAnimationFrame(tick);
   285→    };
   286→    raf = requestAnimationFrame(tick);
   287→    return () => cancelAnimationFrame(raf);
   288→  }, [phase]);
   289→
   290→  const submitEmail = () => {
   291→    setError(null);
   292→    if (!isValidEmail(email)) {
   293→      setError("Please enter a valid email address.");
   294→      return;
   295→    }
   296→    try {
   297→      const leadsRaw = localStorage.getItem("gx_leads");
   298→      const leads = leadsRaw ? (JSON.parse(leadsRaw) as unknown[]) : [];
   299→      leads.push({
   300→        url,
   301→        email: email.trim(),
   302→        timestamp: new Date().toISOString(),
   303→        source: "homepage_scan_widget",
   304→        findingsCount: findings.length,
   305→      });
   306→      localStorage.setItem("gx_leads", JSON.stringify(leads));
   307→    } catch {
   308→      // localStorage unavailable — we still proceed to the success state.
   309→    }
   310→    setPhase("done");
   311→  };
   312→
   313→  const reset = () => {
   314→    setPhase("idle");
   315→    setUrl("");
   316→    setEmail("");
   317→    setError(null);
   318→    setFindings([]);
   319→    setProgress(0);
   320→    setScanPhaseIdx(0);
   321→    setTargetCreated(false);
   322→  };
   323→
   324→  return (
   325→    <section
   326→      id="scan-widget"
   327→      className="relative isolate overflow-hidden px-4 py-20 sm:px-6"
   328→    >
   329→      {/* Background gradient */}
   330→      <motion.div
   331→        aria-hidden
   332→        className="absolute inset-0 -z-10"
   333→        initial={{ opacity: 0 }}
   334→        whileInView={{ opacity: 1 }}
   335→        viewport={{ once: true }}
   336→        transition={{ duration: 1 }}
   337→      >
   338→        <div
   339→          className="absolute inset-0"
   340→          style={{
   341→            background:
   342→              "radial-gradient(ellipse at 30% 30%, rgba(16,185,129,0.15), transparent 55%)," +
   343→              "radial-gradient(ellipse at 70% 70%, rgba(6,182,212,0.12), transparent 55%)," +
   344→              "linear-gradient(135deg, rgba(24,24,27,0.95) 0%, rgba(9,9,11,0.98) 100%)",
   345→          }}
   346→        />
   347→      </motion.div>
   348→
   349→      <div className="mx-auto max-w-3xl">
   350→        <div className="mb-8 text-center">
   351→          <div className="mb-2 flex items-center justify-center gap-2 font-mono text-[10px] uppercase tracking-widest text-emerald-500/70">
   352→            <Zap className="size-3" /> {"// Free instant scan"}
   353→          </div>
   354→          <h2 className="text-3xl font-bold text-zinc-50 sm:text-4xl">
   355→            Scan Your Website Free
   356→          </h2>
   357→          <p className="mx-auto mt-2 max-w-xl text-sm text-zinc-400">
   358→            Enter your URL below. GuardianX runs a non-intrusive external scan and
   359→            surfaces real vulnerabilities — no signup required.
   360→          </p>
   361→        </div>
   362→
   363→        {/* Main card */}
   364→        <div className="rounded-xl border border-emerald-500/30 bg-zinc-950/80 p-5 shadow-[0_0_40px_rgba(16,185,129,0.10)] sm:p-6">
   365→          <AnimatePresence mode="wait">
   366→            {/* ── IDLE ──────────────────────────────────────────────────── */}
   367→            {phase === "idle" ? (
   368→              <motion.div
   369→                key="idle"
   370→                initial={{ opacity: 0, y: 8 }}
   371→                animate={{ opacity: 1, y: 0 }}
   372→                exit={{ opacity: 0, y: -8 }}
   373→                transition={{ duration: 0.25 }}
   374→              >
   375→                <div className="flex flex-col gap-3 sm:flex-row">
   376→                  <div className="relative flex-1">
   377→                    <Globe className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-zinc-500" />
   378→                    <input
   379→                      type="url"
   380→                      inputMode="url"
   381→                      autoComplete="url"
   382→                      placeholder="example.com or https://your-site.com"
   383→                      value={url}
   384→                      onChange={(e) => setUrl(e.target.value)}
   385→                      onKeyDown={(e) => {
   386→                        if (e.key === "Enter") startScan();
   387→                      }}
   388→                      className="w-full rounded-md border border-zinc-700 bg-zinc-900/60 py-3 pl-9 pr-3 text-sm text-zinc-100 placeholder:text-zinc-600 focus:border-emerald-500/50 focus:outline-none focus:ring-2 focus:ring-emerald-500/20"
   389→                      aria-label="Website URL to scan"
   390→                    />
   391→                  </div>
   392→                  <GlowCTA onClick={startScan} variant="solid" className="!px-6 !py-3 !text-sm">
   393→                    <Radar className="size-4" />
   394→                    Scan Now
   395→                  </GlowCTA>
   396→                </div>
   397→
   398→                <div className="mt-4 flex flex-wrap items-center justify-between gap-2 font-mono text-[10px] text-zinc-600">
   399→                  <div className="flex items-center gap-3">
   400→                    <span className="inline-flex items-center gap-1">
   401→                      <Lock className="size-3" /> Non-intrusive · no payloads sent
   402→                    </span>
   403→                    <span className="inline-flex items-center gap-1">
   404→                      <Clock className="size-3" /> ~30s
   405→                    </span>
   406→                  </div>
   407→                  <span>1 scan / hour</span>
   408→                </div>
   409→
   410→                {error ? (
   411→                  <div className="mt-3 flex items-center gap-2 rounded-md border border-red-500/30 bg-red-500/10 px-3 py-2 text-xs text-red-300">
   412→                    <AlertCircle className="size-3.5 shrink-0" />
   413→                    {error}
   414→                  </div>
   415→                ) : null}
   416→
   417→                {!canScan && rateLimitedUntil !== null ? (
   418→                  <div className="mt-3 flex items-center gap-2 rounded-md border border-amber-500/30 bg-amber-500/10 px-3 py-2 text-xs text-amber-300">
   419→                    <Clock className="size-3.5 shrink-0" />
   420→                    You&apos;ve used your free scan for this hour. Next scan available in{" "}
   421→                    {timeLeftLabel(rateLimitedUntil - now)}.
   422→                  </div>
   423→                ) : null}
   424→              </motion.div>
   425→            ) : null}
   426→
   427→            {/* ── SCANNING ──────────────────────────────────────────────── */}
   428→            {phase === "scanning" ? (
   429→              <motion.div
   430→                key="scanning"
   431→                initial={{ opacity: 0, y: 8 }}
   432→                animate={{ opacity: 1, y: 0 }}
   433→                exit={{ opacity: 0, y: -8 }}
   434→                transition={{ duration: 0.25 }}
   435→              >
   436→                <div className="mb-4 flex items-center gap-2">
   437→                  <Loader2 className="size-4 animate-spin text-emerald-400" />
   438→                  <span className="text-sm font-semibold text-zinc-100">
   439→                    Scanning <span className="text-emerald-300">{url}</span>
   440→                  </span>
   441→                  {targetCreated ? (
   442→                    <span className="ml-auto rounded bg-emerald-500/15 px-1.5 py-0.5 font-mono text-[9px] text-emerald-300">
   443→                      target saved
   444→                    </span>
   445→                  ) : (
   446→                    <span className="ml-auto rounded bg-zinc-800 px-1.5 py-0.5 font-mono text-[9px] text-zinc-400">
   447→                      demo mode
   448→                    </span>
   449→                  )}
   450→                </div>
   451→
   452→                {/* Progress bar */}
   453→                <div className="relative mb-4 h-2 overflow-hidden rounded-full bg-zinc-800">
   454→                  <motion.div
   455→                    className="absolute inset-y-0 left-0 rounded-full bg-gradient-to-r from-emerald-500 via-cyan-500 to-emerald-400"
   456→                    style={{ width: `${progress}%` }}
   457→                    transition={{ ease: "linear" }}
   458→                  />
   459→                </div>
   460→
   461→                {/* Phase list */}
   462→                <div className="space-y-2">
   463→                  {SCAN_PHASES.map((p, i) => {
   464→                    const Icon = p.icon;
   465→                    const state = i < scanPhaseIdx ? "done" : i === scanPhaseIdx ? "active" : "pending";
   466→                    return (
   467→                      <div
   468→                        key={p.label}
   469→                        className={`flex items-center gap-3 rounded-md border px-3 py-2 font-mono text-xs transition-colors ${
   470→                          state === "done"
   471→                            ? "border-emerald-500/30 bg-emerald-500/5"
   472→                            : state === "active"
   473→                            ? "border-cyan-500/40 bg-cyan-500/10"
   474→                            : "border-zinc-800 bg-zinc-900/40 text-zinc-600"
   475→                        }`}
   476→                      >
   477→                        {state === "done" ? (
   478→                          <CheckCircle2 className="size-4 text-emerald-400" />
   479→                        ) : state === "active" ? (
   480→                          <Loader2 className={`size-4 animate-spin ${p.color}`} />
   481→                        ) : (
   482→                          <Icon className={`size-4 ${p.color} opacity-40`} />
   483→                        )}
   484→                        <span
   485→                          className={
   486→                            state === "done"
   487→                              ? "text-emerald-300"
   488→                              : state === "active"
   489→                              ? p.color
   490→                              : "text-zinc-600"
   491→                          }
   492→                        >
   493→                          {p.label}
   494→                        </span>
   495→                        {state === "done" ? (
   496→                          <span className="ml-auto text-[9px] uppercase text-emerald-500/60">ok</span>
   497→                        ) : null}
   498→                      </div>
   499→                    );
   500→                  })}
   501→                </div>
   502→              </motion.div>
   503→            ) : null}
   504→
   505→            {/* ── FINDINGS ──────────────────────────────────────────────── */}
   506→            {phase === "findings" ? (
   507→              <motion.div
   508→                key="findings"
   509→                initial={{ opacity: 0, y: 8 }}
   510→                animate={{ opacity: 1, y: 0 }}
   511→                exit={{ opacity: 0, y: -8 }}
   512→                transition={{ duration: 0.25 }}
   513→              >
   514→                <div className="mb-4 flex items-center gap-2">
   515→                  <ShieldAlert className="size-5 text-red-400" />
   516→                  <span className="text-sm font-semibold text-zinc-100">
   517→                    Found {findings.length} potential vulnerabilities on{" "}
   518→                    <span className="text-red-300">{url}</span>
   519→                  </span>
   520→                </div>
   521→
   522→                <div className="space-y-2">
   523→                  {findings.map((f, i) => {
   524→                    const m = SEV_META[f.severity];
   525→                    return (
   526→                      <motion.div
   527→                        key={f.id}
   528→                        initial={{ opacity: 0, x: -12 }}
   529→                        animate={{ opacity: 1, x: 0 }}
   530→                        transition={{ duration: 0.3, delay: i * 0.12 }}
   531→                        className={`rounded-md border ${m.border} ${m.bg} p-3`}
   532→                      >
