   136→// ── Helpers ─────────────────────────────────────────────────────────────────
   137→function timeAgo(iso: string): string {
   138→  if (!iso) return "-";
   139→  const diff = Date.now() - new Date(iso).getTime();
   140→  const sec = Math.floor(diff / 1000);
   141→  if (sec < 60) return `${sec}s ago`;
   142→  const min = Math.floor(sec / 60);
   143→  if (min < 60) return `${min}m ago`;
   144→  const hr = Math.floor(min / 60);
   145→  if (hr < 24) return `${hr}h ago`;
   146→  const day = Math.floor(hr / 24);
   147→  if (day < 30) return `${day}d ago`;
   148→  return new Date(iso).toLocaleDateString();
   149→}
   150→
   151→function formatTime(iso: string): string {
   152→  if (!iso) return "-";
   153→  try {
   154→    return new Date(iso).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit", second: "2-digit" });
   155→  } catch {
   156→    return iso;
   157→  }
   158→}

   160→// Build an empty config object from a schema (selects default to first option)
   161→function emptyConfigFor(schema: ConnectorSchema | undefined): Record<string, string> {
   162→  const cfg: Record<string, string> = {};
   163→  if (!schema) return cfg;
   164→  for (const f of schema.configFields) {
   165→    if (f.type === "select" && f.options && f.options.length > 0) {
   166→      cfg[f.key] = f.options[0];
   167→    } else {
   168→      cfg[f.key] = "";
   169→    }
   170→  }
   171→  return cfg;
   172→}

   245→// INTEGRATIONS HUB (Tab 1)
   246→// ════════════════════════════════════════════════════════════════════════════
   247→function IntegrationsHub({ onSwitchToLog }: { onSwitchToLog: () => void }) {
   248→  const { toast } = useToast();
   249→  const [schemas, setSchemas] = useState<ConnectorSchema[]>([]);
   250→  const [integrations, setIntegrations] = useState<Integration[]>([]);
   251→  const [loading, setLoading] = useState(true);
   252→  const [sendingTest, setSendingTest] = useState(false);
   253→  const [lastForwardLog, setLastForwardLog] = useState<ForwardingLogEntry | null>(null);
   254→  const [modalState, setModalState] = useState<{
   255→    open: boolean;
   256→    type: string | null;
   257→    editingId: string | null;
   258→  }>({ open: false, type: null, editingId: null });
   259→
   260→  const load = useCallback(async () => {
   261→    try {
   262→      const res = await fetch("/api/integrations?schemas=true");
   263→      const data = await res.json();
   264→      setSchemas((data.schemas as ConnectorSchema[]) || []);
   265→      setIntegrations((data.integrations as Integration[]) || []);
   266→    } catch {
   267→      setSchemas([]);
   268→      setIntegrations([]);
   269→    } finally {
   270→      setLoading(false);
   271→    }
   272→  }, []);
   273→
   274→  useEffect(() => {
   275→    load();
   276→    const id = setInterval(load, 30_000);
   277→    return () => clearInterval(id);
   278→  }, [load]);
   279→
   280→  const activeCount = integrations.filter((i) => i.isActive).length;
   281→  const connectedCount = integrations.length;
   282→
   283→  const handleSendTestEvent = async () => {
   284→    if (activeCount === 0) {
   285→      toast({
   286→        variant: "destructive",
   287→        title: "No active integrations",
   288→        description: "Configure and enable at least one integration before sending a test event.",
   289→      });
   290→      return;
   291→    }
   292→    setSendingTest(true);
   293→    try {
   294→      const res = await fetch("/api/integrations", {
   295→        method: "PATCH",
   296→        headers: { "Content-Type": "application/json" },
   297→        body: JSON.stringify({ action: "forward_test_event" }),
   298→      });
   299→      const data = await res.json();
   300→      if (!res.ok) throw new Error((data as { error?: string }).error || "Failed to forward test event");
   301→      const log = data as ForwardingLogEntry;
   302→      setLastForwardLog(log);
   303→      toast({
   304→        title: `Test event forwarded to ${log.totalSent} connector${log.totalSent === 1 ? "" : "s"}`,
   305→        description: `${log.totalSuccess} succeeded, ${log.totalFailed} failed. See the Forwarding Log tab for details.`,
   306→      });
   307→    } catch (err) {
   308→      toast({
   309→        variant: "destructive",
   310→        title: "Forwarding failed",
   311→        description: err instanceof Error ? err.message : "unknown error",
   312→      });
   313→    } finally {
   314→      setSendingTest(false);
   315→    }
   316→  };
   317→
   318→  const handleToggle = async (integration: Integration) => {
   319→    const next = !integration.isActive;
   320→    // Optimistic update
   321→    setIntegrations((prev) =>
   322→      prev.map((i) => (i.id === integration.id ? { ...i, isActive: next } : i))
   323→    );
   324→    try {
   325→      const res = await fetch("/api/integrations", {
   326→        method: "PATCH",
   327→        headers: { "Content-Type": "application/json" },
   328→        body: JSON.stringify({ action: "update", id: integration.id, isActive: next }),
   329→      });
   330→      if (!res.ok) throw new Error("Update failed");
   331→      toast({
   332→        title: next ? "Integration enabled" : "Integration disabled",
   333→        description: `${integration.type} is now ${next ? "active" : "paused"}.`,
   334→      });
   335→    } catch (err) {
   336→      // Rollback
   337→      setIntegrations((prev) =>
   338→        prev.map((i) => (i.id === integration.id ? { ...i, isActive: !next } : i))
   339→      );
   340→      toast({
   341→        variant: "destructive",
   342→        title: "Toggle failed",
   343→        description: err instanceof Error ? err.message : "unknown error",
   344→      });
   345→    }
   346→  };
   347→
   348→  const handleDelete = async (integration: Integration) => {
   349→    try {
   350→      const res = await fetch(`/api/integrations?id=${encodeURIComponent(integration.id)}`, {
   351→        method: "DELETE",
   352→      });
   353→      if (!res.ok) {
   354→        const data = await res.json().catch(() => ({}));
   355→        throw new Error((data as { error?: string }).error || "Delete failed");
   356→      }
   357→      setIntegrations((prev) => prev.filter((i) => i.id !== integration.id));
   358→      toast({
   359→        title: "Integration removed",
   360→        description: `${integration.type} connector deleted.`,
   361→      });
   362→    } catch (err) {
   363→      toast({
   364→        variant: "destructive",
   365→        title: "Delete failed",
   366→        description: err instanceof Error ? err.message : "unknown error",
   367→      });
   368→    }
   369→  };
   370→
   371→  const openCreate = (type: string) => {
   372→    setModalState({ open: true, type, editingId: null });
   373→  };
   374→
   375→  const openEdit = (integration: Integration) => {
   376→    setModalState({ open: true, type: integration.type, editingId: integration.id });
   377→  };
   378→
   379→  const closeModal = () => {
   380→    setModalState({ open: false, type: null, editingId: null });
   381→  };
   382→
   383→  const onSaved = () => {
   384→    closeModal();
   385→    load();
   386→  };
   387→
   388→  return (
   389→    <div className="space-y-4">
   390→      {/* Action bar */}
   391→      <div className="holo-card-sharp hud-corners flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   392→        <div className="flex flex-wrap items-center gap-3">
   393→          <Stat label="Connected" value={connectedCount} color="emerald" />
   394→          <span className="hidden h-8 w-px bg-zinc-800 sm:block" />
   395→          <Stat label="Active" value={activeCount} color="cyan" />
   396→          <span className="hidden h-8 w-px bg-zinc-800 sm:block" />
   397→          <Stat label="Available" value={schemas.length} color="violet" />
   398→        </div>
   399→        <Button
   400→          onClick={handleSendTestEvent}
   401→          disabled={sendingTest || activeCount === 0}
   402→          className="bg-emerald-600 text-white hover:bg-emerald-500"
   403→        >
   404→          {sendingTest ? <Loader2 className="size-4 animate-spin" /> : <Send className="size-4" />}
   405→          Send Test Event
   406→        </Button>
   407→      </div>
   408→
   409→      {/* Last forward result inline banner */}
   410→      {lastForwardLog && (
   411→        <motion.div
   412→          initial={{ opacity: 0, y: -4 }}
   413→          animate={{ opacity: 1, y: 0 }}
   414→          className="holo-card-sharp hud-corners flex flex-col gap-2 p-4 sm:flex-row sm:items-center sm:justify-between"
   415→        >
   416→          <div className="flex items-center gap-3">
   417→            <div className={`flex size-9 items-center justify-center rounded-md ${lastForwardLog.totalFailed === 0 ? "bg-emerald-500/15" : "bg-amber-500/15"}`}>
   418→              {lastForwardLog.totalFailed === 0 ? (
   419→                <CheckCircle2 className="size-5 text-emerald-400" />
   420→              ) : (
   421→                <AlertCircle className="size-5 text-amber-400" />
   422→              )}
   423→            </div>
   424→            <div>
   425→              <div className="text-sm font-semibold text-zinc-100">
   426→                Last test event: {lastForwardLog.totalSuccess}/{lastForwardLog.totalSent} succeeded
   427→              </div>
   428→              <div className="font-mono text-[11px] text-zinc-500">
   429→                {formatTime(lastForwardLog.timestamp)} - event: {lastForwardLog.eventType}
   430→              </div>
   431→            </div>
   432→          </div>
   433→          <Button
   434→            variant="outline"
   435→            size="sm"
   436→            onClick={onSwitchToLog}
   437→            className="border-zinc-700 bg-zinc-900 text-zinc-300 hover:bg-zinc-800"
   438→          >
   439→            View Log <ArrowRight className="size-3.5" />
   440→          </Button>
   441→        </motion.div>
   442→      )}
   443→
   444→      {/* Connector cards grid */}
   445→      {loading ? (
   446→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   447→          {Array.from({ length: 9 }).map((_, i) => (
   448→            <Skeleton key={i} className="h-44 w-full rounded-lg bg-zinc-900/60" />
   449→          ))}
   450→        </div>
   451→      ) : schemas.length === 0 ? (
   452→        <div className="holo-card-sharp hud-corners flex flex-col items-center justify-center gap-3 p-12 text-center">
   453→          <AlertCircle className="size-8 text-amber-400" />
   454→          <div className="text-sm font-medium text-zinc-300">No connector schemas loaded</div>
   455→          <div className="text-xs text-zinc-500">The integration engine returned no schemas. Try refreshing.</div>
   456→          <Button variant="outline" size="sm" onClick={load} className="border-zinc-700 bg-zinc-900 text-zinc-300 hover:bg-zinc-800">
   457→            <RefreshCw className="size-3.5" /> Reload
   458→          </Button>
   459→        </div>
   460→      ) : (
   461→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   462→          {schemas.map((schema, idx) => {
   463→            const meta = CONNECTOR_META[schema.type] || { icon: Plug, color: "zinc" as AccentColor };
   464→            const existing = integrations.find((i) => i.type === schema.type);
   465→            const Icon = meta.icon;
   466→            return (
   467→              <motion.div
   468→                key={schema.type}
   469→                initial={{ opacity: 0, y: 8 }}
   470→                animate={{ opacity: 1, y: 0 }}
   471→                transition={{ duration: 0.18, delay: Math.min(idx * 0.03, 0.25) }}
   472→              >
   473→                <ConnectorCard
   474→                  schema={schema}
   475→                  icon={Icon}
   476→                  color={meta.color}
   477→                  existing={existing}
   478→                  onConfigure={() => (existing ? openEdit(existing) : openCreate(schema.type))}
   479→                  onToggle={existing ? () => handleToggle(existing) : undefined}
   480→                  onDelete={existing ? () => handleDelete(existing) : undefined}
   481→                />
   482→              </motion.div>
   483→            );
   484→          })}
   485→        </div>
   486→      )}
   487→
   488→      {/* Config modal */}
   489→      <AnimatePresence>
   490→        {modalState.open && modalState.type && (
   491→          <ConfigModal
   492→            type={modalState.type}
   493→            editingId={modalState.editingId}
   494→            schemas={schemas}
   495→            integrations={integrations}
   496→            onClose={closeModal}
   497→            onSaved={onSaved}
   498→          />
   499→        )}
   500→      </AnimatePresence>
   501→    </div>
   502→  );
   503→}

   834→  }
   835→
   836→  return (
   837→    <div
   838→      className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-4 backdrop-blur-sm"
   839→      onClick={onClose}
   840→      role="dialog"
