     3→     3→/**
     4→     4→ * AgentX — Sidebar Tab Edition
     5→     5→ * ----------------------------
     6→     6→ * Always-on conversational AI for the GuardianX Command Center.
     7→     7→ *
     8→     8→ * Previously a floating drawer at `fixed bottom-20 right-4 z-[90]`, Agent X
     9→     9→ * is now a full-tab view that fills the main content area (just like
    10→    10→ * Overview / All Clients / Patches / etc.). The parent mounts it inside
    11→    11→ * the main panel whenever `tab === "agent-x"` and controls visibility
    12→    12→ * via the `open` prop. When `open={false}` the component renders nothing.
    13→    13→ *
    14→    14→ * Voice architecture (rebuilt — no longer uses <VoiceControl>):
    15→    15→ *
    16→    16→ *   - We own a `SpeechRecognition` instance directly so we can hook the
    17→    17→ *     `onsoundstart` / `onsoundend` events for voice-activity detection.
    18→    18→ *   - `speakingRef` gates the recognition `onend` auto-restart: when
    19→    19→ *     Agent X is speaking, a browser-side silence timeout that fires
    20→    20→ *     `onend` is suppressed (no auto-restart). The mic is *still live*
    21→    21→ *     to the extent the browser allows it — so if the user starts
    22→    22→ *     speaking mid-TTS, `onsoundstart` fires and we cancel TTS
    23→    23→ *     immediately (interruptible).
    24→    24→ *   - When TTS finishes (`utterance.onend`), we flip `speakingRef` off
    25→    25→ *     and restart recognition if it had stopped.
    26→    26→ *
    27→    27→ * Streaming TTS:
    28→    28→ *
    29→    29→ *   - Replies are split into sentences. The first sentence is spoken
    30→    30→ *     immediately as a "fast first response"; subsequent sentences are
    31→    31→ *     queued (`onend` of utterance N triggers start of N+1). Display
    32→    32→ *     is also progressive: the message bubble starts with the first
    33→    33→ *     sentence and grows as subsequent chunks arrive.
    34→    34→ *
    35→    35→ * Layout (when open):
    36→    36→ *
    37→    37→ *   ┌─────────────────────────────────────────────────────────────────┐
    38→    38→ *   │ Header: AGENT X · provider badge · mic toggle · clear · export    │
    39→    39→ *   ├──────────────────────────────────────┬──────────────────────────┤
    40→    40→ *   │                                       │  Briefing panel (30%)    │
    41→    41→ *   │  Conversation (70%, scrollable)       │  - Posture score         │
    42→    42→ *   │                                       │  - Pending patches       │
    43→    43→ *   │  [user msg]                           │  - Critical findings     │
    44→    44→ *   │  [agent msg with thinking dots]       │  - Recent activity       │
    45→    45→ *   │                                       │                          │
    46→    46→ *   ├───────────────────────────────────────┤                          │
    47→    47→ *   │ Text input + Send                     │                          │
    48→    48→ *   │ [Brief me][Show patches][Explain…][Next]                         │
    49→    49→ *   │ ▁▂▄█▆▃▁▂▄ waveform (when listening)                              │
    50→    50→ *   └───────────────────────────────────────┴──────────────────────────┘
    51→    51→ *
    52→    52→ * Greeting behavior:
    53→    53→ *
    54→    54→ *   - On the first `open={true}` transition, fetch /api/agent-x/briefing
    55→    55→ *     ONCE. Build a short greeting ("Good morning, Ayan. Your security
    56→    56→ *     posture is excellent — score 100/100 (grade A). What are you up
    57→    57→ *     to today?"). Speak it. Only start listening after the utterance
    58→    58→ *     finishes (utterance.onend) — no 600ms timer.
    59→    59→ *
    60→    60→ * Proactive monitoring:
    61→    61→ *
    62→    62→ *   - Poll /api/agent-x/briefing every 5 minutes. Compare pending-patch
    63→    63→ *     IDs (not counts). If a NEW patch ID appears, drop an alert message
    64→    64→ *     + speak a heads-up. Never auto-explain per-tab.
    65→    65→ */

   254→   254→    voices.find((v) => /^en/i.test(v.lang)) ||

   256→   256→    null

   260→   260→// ─── Sentence splitter (for streaming TTS) ──────────────────────────────────
   261→   261→// Splits a reply into speakable chunks. Splits on `.`, `!`, `?`, and
   262→   262→// newlines, keeping the punctuation with the sentence. Filters out empty
   263→   263→// chunks (e.g. trailing periods). Keeps the splitter simple — we don't
   264→   264→// need NLP-grade segmentation for SOC analyst replies.
   265→   265→
   266→   266→function splitIntoSentences(text: string): string[] {
   267→   267→  const trimmed = text.trim();
   268→   268→  if (!trimmed) return [];
   269→   269→  // Match runs that end with . ! ? or end-of-string, including the punctuation.
   270→   270→  const matches = trimmed.match(/[^.!?]*[.!?]+|[^.!?]+$/g);
   271→   271→  if (!matches) return [trimmed];
   272→   272→  return matches
   273→   273→    .map((s) => s.trim())
   274→   274→    .filter((s) => s.length > 0);
   275→   275→}

   466→   466→        setSpeakingLocal(false);
   467→   467→      }
   468→   468→    };
   469→   469→
   470→   470→    rec.onspeechstart = () => {
   471→   471→      // Same interrupt semantics — speech is a stronger signal than sound.
   472→   472→      if (speakingRef.current) {
   473→   473→        try {
   474→   474→          window.speechSynthesis.cancel();
   475→   475→        } catch {
   476→   476→          /* noop */
   477→   477→        }
   478→   478→        speakingRef.current = false;
   479→   479→        setSpeakingLocal(false);
   480→   480→      }
   481→   481→    };
   482→   482→
   483→   483→    rec.onsoundend = () => {
   484→   484→      /* Just VAD bookkeeping — no action needed. */
   485→   485→    };
   486→   486→
   487→   487→    rec.onresult = (ev: SpeechRecognitionEventLike) => {
   488→   488→      const results = ev.results;
   489→   489→      let interimText = "";
   490→   490→      let finalText = "";
   491→   491→      for (let i = ev.resultIndex; i < results.length; i++) {
   492→   492→        const alt = results[i]?.[0];
   493→   493→        if (!alt) continue;
   494→   494→        const t = alt.transcript;
   495→   495→        if (results[i].isFinal) finalText += t;
   496→   496→        else interimText += t;
   497→   497→      }
   498→   498→      setInterim(interimText);
   499→   499→      if (finalText.trim()) {
   500→   500→        latestTranscriptRef.current = finalText.trim();
   501→   501→        setInterim("");
   502→   502→        // Send the final transcript to the chat endpoint.
   503→   503→        void sendMessage(latestTranscriptRef.current);
   504→   504→      }
   505→   505→    };
   506→   506→
   507→   507→    rec.onerror = (ev: SpeechRecognitionErrorEventLike) => {
   508→   508→      // "no-speech" and "aborted" are routine in continuous mode — don't surface.
   509→   509→      if (ev.error === "no-speech" || ev.error === "aborted") return;
   510→   510→      if (ev.error === "not-allowed" || ev.error === "service-not-allowed") {
   511→   511→        setVoiceError("Mic permission denied");
   512→   512→      } else if (ev.error === "network") {
   513→   513→        setVoiceError("Network error during recognition");
   514→   514→      }
   515→   515→    };
   516→   516→
   517→   517→    rec.onend = () => {
   518→   518→      setListening(false);
   519→   519→      setInterim("");
   520→   520→      // Auto-restart gate: don't restart if (a) the user explicitly stopped,
   521→   521→      // (b) Agent X is currently speaking (let the utterance.onend handler
   522→   522→      // restart), or (c) the panel is no longer open.
   523→   523→      if (
   524→   524→        !openRef.current ||
   525→   525→        userStoppedRef.current ||
   526→   526→        speakingRef.current
   527→   527→      ) {
   528→   528→        return;
   529→   529→      }
   530→   530→      try {
   531→   531→        rec.start();
   532→   532→      } catch {
   533→   533→        /* InvalidStateError if start() raced — ignored; onend will fire again */
   534→   534→      }
   535→   535→    };
   536→   536→
   537→   537→    recRef.current = rec;
   538→   538→    return rec;
   539→   539→  }, []);
   540→   540→
   541→   541→  // ── Start listening ──────────────────────────────────────────────────────
   542→   542→  const startListening = useCallback(() => {
   543→   543→    const rec = ensureRecognition();
   544→   544→    if (!rec) return;
   545→   545→    userStoppedRef.current = false;
   546→   546→    try {
   547→   547→      rec.start();
   548→   548→    } catch {
   549→   549→      /* InvalidStateError if start() raced — ignored */
   550→   550→    }
   551→   551→    void ensureMicAnalyser();
   552→   552→  }, [ensureRecognition]);
   553→   553→
   554→   554→  // ── Stop listening (user-initiated) ─────────────────────────────────────
   555→   555→  const stopListening = useCallback(() => {
   556→   556→    userStoppedRef.current = true;
   557→   557→    const rec = recRef.current;
   558→   558→    if (!rec) return;
   559→   559→    try {
   560→   560→      rec.stop();
   561→   561→    } catch {
   562→   562→      /* noop */
   563→   563→    }
   564→   564→    stopMicAnalyser();
   565→   565→  }, []);
   566→   566→
   567→   567→  // ── Mic toggle (button click) ────────────────────────────────────────────
   568→   568→  const micToggle = useCallback(() => {
   569→   569→    if (listening) stopListening();
   570→   570→    else startListening();
   571→   571→  }, [listening, startListening, stopListening]);
   572→   572→
   573→   573→  // ── Web Audio API: wire up AnalyserNode on the mic stream ────────────────
   574→   574→  // Used by the waveform canvas. Created lazily on first startListening and
   575→   575→  // torn down on stopListening + on unmount.
   576→   576→  const ensureMicAnalyser = useCallback(async () => {
   577→   577→    if (analyserRef.current) return;
   578→   578→    if (typeof window === "undefined") return;
   579→   579→    if (!navigator.mediaDevices?.getUserMedia) return;
   580→   580→    try {
   581→   581→      const stream = await navigator.mediaDevices.getUserMedia({
   582→   582→        audio: { echoCancellation: true, noiseSuppression: true },
   583→   583→      });
   584→   584→      micStreamRef.current = stream;
   585→   585→      const Ctx =
   586→   586→        window.AudioContext ||
   587→   587→        (window as unknown as { webkitAudioContext?: typeof AudioContext })
   588→   588→          .webkitAudioContext;
   589→   589→      if (!Ctx) return;
   590→   590→      const ctx = new Ctx();
   591→   591→      audioCtxRef.current = ctx;
   592→   592→      const src = ctx.createMediaStreamSource(stream);
   593→   593→      const analyser = ctx.createAnalyser();
   594→   594→      analyser.fftSize = 64;
   595→   595→      analyser.smoothingTimeConstant = 0.7;
   596→   596→      src.connect(analyser);
   597→   597→      analyserRef.current = analyser;
   598→   598→    } catch {
   599→   599→      /* Permission denied or unavailable — waveform just stays empty */
   600→   600→    }
   601→   601→  }, []);
   602→   602→
   603→   603→  const stopMicAnalyser = useCallback(() => {
   604→   604→    if (rafRef.current != null) {
   605→   605→      cancelAnimationFrame(rafRef.current);
   606→   606→      rafRef.current = null;
   607→   607→    }
   608→   608→    if (micStreamRef.current) {
   609→   609→      for (const track of micStreamRef.current.getTracks()) {
   610→   610→        try {
   611→   611→          track.stop();
   612→   612→        } catch {
   613→   613→          /* noop */
   614→   614→        }
   615→   615→      }
   616→   616→      micStreamRef.current = null;
   617→   617→    }
   618→   618→    if (audioCtxRef.current) {
   619→   619→      try {
   620→   620→        void audioCtxRef.current.close();
   621→   621→      } catch {
   622→   622→        /* noop */
   623→   623→      }
   624→   624→      audioCtxRef.current = null;
   625→   625→    }
   626→   626→    analyserRef.current = null;
   627→   627→  }, []);
   628→   628→
   629→   629→  // ── Waveform render loop (only when listening && !speaking) ──────────────
   630→   630→  useEffect(() => {
   631→   631→    if (!listening || speakingLocal) {
   632→   632→      if (rafRef.current != null) {
   633→   633→        cancelAnimationFrame(rafRef.current);
   634→   634→        rafRef.current = null;
   635→   635→      }
   636→   636→      // Clear the canvas when not actively drawing.
   637→   637→      const canvas = canvasRef.current;
   638→   638→      if (canvas) {
   639→   639→        const ctx2d = canvas.getContext("2d");
   640→   640→        if (ctx2d) {
   641→   641→          ctx2d.clearRect(0, 0, canvas.width, canvas.height);
   642→   642→        }
   643→   643→      }
   644→   644→      return;
   645→   645→    }
   646→   646→    const canvas = canvasRef.current;
   647→   647→    if (!canvas) return;
   648→   648→    const ctx2d = canvas.getContext("2d");
   649→   649→    if (!ctx2d) return;
   650→   650→
   651→   651→    const draw = () => {
   652→   652→      const analyser = analyserRef.current;
   653→   653→      if (!analyser) {
   654→   654→        rafRef.current = requestAnimationFrame(draw);
   655→   655→        return;
   656→   656→      }
   657→   657→      const bufferLength = analyser.frequencyBinCount;
   658→   658→      const dataArray = new Uint8Array(bufferLength);
   659→   659→      analyser.getByteFrequencyData(dataArray);
   660→   660→
   661→   661→      const w = canvas.width;
   662→   662→      const h = canvas.height;
   663→   663→      ctx2d.clearRect(0, 0, w, h);
   664→   664→
   665→   665→      // Draw a centerline baseline.
   666→   666→      const barCount = Math.min(bufferLength, 32);
   667→   667→      const gap = 2;
   668→   668→      const barWidth = (w - gap * (barCount - 1)) / barCount;
   669→   669→      for (let i = 0; i < barCount; i++) {
   670→   670→        const v = dataArray[i] / 255; // 0..1
   671→   671→        const barH = Math.max(2, v * h);
   672→   672→        const x = i * (barWidth + gap);
   673→   673→        const y = (h - barH) / 2;
   674→   674→        // Gradient: emerald → cyan.
   675→   675→        const grad = ctx2d.createLinearGradient(0, y, 0, y + barH);
   676→   676→        grad.addColorStop(0, "rgba(16,185,129,0.9)");
   677→   677→        grad.addColorStop(1, "rgba(34,211,238,0.7)");
   678→   678→        ctx2d.fillStyle = grad;
   679→   679→        ctx2d.fillRect(x, y, barWidth, barH);
   680→   680→      }
   681→   681→      rafRef.current = requestAnimationFrame(draw);
   682→   682→    };
   683→   683→    rafRef.current = requestAnimationFrame(draw);
   684→   684→    return () => {
   685→   685→      if (rafRef.current != null) {
   686→   686→        cancelAnimationFrame(rafRef.current);
   687→   687→        rafRef.current = null;
   688→   688→      }
   689→   689→    };
   690→   690→  }, [listening, speakingLocal]);
   691→   691→
   692→   692→  // ── Streaming TTS: speak sentences sequentially ──────────────────────────
   693→   693→  // The first sentence is spoken immediately. Each utterance's `onend`
   694→   694→  // triggers the next sentence. While speaking, `speakingRef` is true →
   695→   695→  // the recognition `onend` auto-restart is gated off (no race).
   696→   696→  // The mic stays live to the extent the browser allows — if the user
   697→   697→  // starts talking, `onsoundstart` cancels TTS (see ensureRecognition).
   698→   698→  const speakChunked = useCallback(
   699→   699→    (text: string, opts?: { onAllDone?: () => void }) => {
   700→   700→      if (typeof window === "undefined" || !("speechSynthesis" in window)) {
   701→   701→        opts?.onAllDone?.();
   702→   702→        return;
   703→   703→      }
   704→   704→      const trimmed = text.trim();
   705→   705→      if (!trimmed) {
   706→   706→        opts?.onAllDone?.();
   707→   707→        return;
   708→   708→      }
   709→   709→      const synth = window.speechSynthesis;
   710→   710→      synth.cancel(); // interrupt any in-flight utterance
   711→   711→      const chunks = splitIntoSentences(trimmed);
   712→   712→      if (chunks.length === 0) {
   713→   713→        opts?.onAllDone?.();
   714→   714→        return;
   715→   715→      }
   716→   716→      const voices = voicesRef.current.length
   717→   717→        ? voicesRef.current
   718→   718→        : synth.getVoices();
   719→   719→      const preferred = pickPersonaVoice(voices);
   720→   720→
   721→   721→      speakingRef.current = true;
   722→   722→      setSpeakingLocal(true);
   723→   723→
   724→   724→      // If recognition is currently running, stop it so the browser doesn't
   725→   725→      // pick up its own TTS as "speech" (we'll restart on completion).
   726→   726→      const wasListening = listening;
   727→   727→      if (wasListening) {
   728→   728→        try {
   729→   729→          recRef.current?.stop();
   730→   730→        } catch {
   731→   731→          /* noop */
   732→   732→        }
   733→   733→      }
   734→   734→
   735→   735→      let idx = 0;
   736→   736→      const speakNext = () => {
   737→   737→        if (idx >= chunks.length) {
   738→   738→          speakingRef.current = false;
   739→   739→          setSpeakingLocal(false);
   740→   740→          opts?.onAllDone?.();
   741→   741→          // Restart listening if it was live before we started speaking.
   742→   742→          if (wasListening && openRef.current && !userStoppedRef.current) {
   743→   743→            try {
   744→   744→              recRef.current?.start();
   745→   745→            } catch {
   746→   746→              /* InvalidStateError if start() raced — onend will retry */
   747→   747→            }
   748→   748→          }
   749→   749→          return;
   750→   750→        }
   751→   751→        const chunk = chunks[idx];
   752→   752→        idx += 1;
   753→   753→        const u = new SpeechSynthesisUtterance(chunk);
   754→   754→        if (preferred) u.voice = preferred;
   755→   755→        u.rate = 0.95;
   756→   756→        u.pitch = 0.85;
   757→   757→        u.volume = 1;
   758→   758→        u.onend = () => {
   759→   759→          // Defer to next tick to avoid the synth racing ahead.
   760→   760→          window.setTimeout(speakNext, 0);
   761→   761→        };
   762→   762→        u.onerror = () => {
   763→   763→          speakingRef.current = false;
   764→   764→          setSpeakingLocal(false);
   765→   765→          opts?.onAllDone?.();
   766→   766→        };
   767→   767→        synth.speak(u);
   768→   768→      };
   769→   769→      speakNext();
   770→   770→    },
   771→   771→    [listening],
   772→   772→  );
   773→   773→
   774→   774→  // ── Stop TTS (manual mute) ───────────────────────────────────────────────
   775→   775→  const stopSpeaking = useCallback(() => {
   776→   776→    if (typeof window === "undefined" || !("speechSynthesis" in window)) return;
   777→   777→    window.speechSynthesis.cancel();
   778→   778→    speakingRef.current = false;
   779→   779→    setSpeakingLocal(false);
   780→   780→  }, []);
   781→   781→
   782→   782→  // ── Execute Agent X actions returned from the chat endpoint ─────────────
   783→   783→  const executeActions = useCallback(
   784→   784→    (actions: AgentXAction[] | undefined) => {
   785→   785→      if (!actions || actions.length === 0) return;
   786→   786→      for (const action of actions) {
   787→   787→        switch (action.type) {
   788→   788→          case "navigate":
   789→   789→            if (action.target) onNavigate?.(action.target);
   790→   790→            break;
   791→   791→          case "scan":
   792→   792→            if (action.target) onScan?.(action.target);
   793→   793→            break;
   794→   794→          case "approve":
   795→   795→          case "approve_patch":
   796→   796→            if (action.target) onApprovePatch?.(action.target);
   797→   797→            break;
   798→   798→          case "search":
   799→   799→            onSearch?.(action.target || action.query || "");
   800→   800→            break;
   801→   801→          case "war_room":
   802→   802→            onOpenWarRoom?.();
   803→   803→            break;
   804→   804→          default:
   805→   805→            // Unknown action types are silently ignored — the backend may
   806→   806→            // emit intents the frontend hasn't wired up yet.
   807→   807→            break;
   808→   808→        }
   809→   809→      }
   810→   810→    },
   811→   811→    [onApprovePatch, onNavigate, onOpenWarRoom, onScan, onSearch],
   812→   812→  );
   813→   813→
   814→   814→  // ── Progressive message display ──────────────────────────────────────────
   815→   815→  // Helper to append the next sentence to a specific agent message ID.
   816→   816→  const appendToMessage = useCallback((id: string, chunk: string) => {
   817→   817→    setMessages((prev) =>
   818→   818→      prev.map((m) =>
   819→   819→        m.id === id ? { ...m, content: m.content + chunk } : m,
   820→   820→      ),
   821→   821→    );
   822→   822→  }, []);
   823→   823→
   824→   824→  // ── Send a message to /api/agent-x/chat ─────────────────────────────────
   825→   825→  const sendMessage = useCallback(
   826→   826→    async (text: string) => {
   827→   827→      const trimmed = text.trim();
   828→   828→      if (!trimmed || thinkingRef.current) return;
   829→   829→
   830→   830→      // Stop any in-flight TTS — the user is talking over Agent X.
   831→   831→      if (speakingRef.current) {
   832→   832→        stopSpeaking();
   833→   833→      }
   834→   834→
   835→   835→      setInput("");
   836→   836→      setThinking(true);
   837→   837→      const userMsgId = `u-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   838→   838→      const agentMsgId = `a-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`;
   839→   839→      setMessages((prev) => [
   840→   840→        ...prev,
   841→   841→        { id: userMsgId, role: "user", content: trimmed, ts: Date.now() },
   842→   842→      ]);
   843→   843→
   844→   844→      try {
   845→   845→        const res = await fetch("/api/agent-x/chat", {
   846→   846→          method: "POST",
   847→   847→          headers: { "Content-Type": "application/json" },
   848→   848→          body: JSON.stringify({
   849→   849→            message: trimmed,
   850→   850→            context: {
   851→   851→              currentTab: currentTabRef.current,
   852→   852→              history: messagesRef.current
   853→   853→                .slice(-10)
   854→   854→                .map((m) => ({ role: m.role, content: m.content })),
   855→   855→            },
   856→   856→          }),
   857→   857→        });
   858→   858→        if (!res.ok) throw new Error(`HTTP ${res.status}`);
   859→   859→        const data = (await res.json()) as AgentXChatResponse;
   860→   860→
   861→   861→        const reply = data.reply?.trim() || "I didn't catch that. Try again.";
   862→   862→
   863→   863→        // Stream the reply: split into sentences, display first sentence
   864→   864→        // immediately, then progressively append the rest as TTS plays.
   865→   865→        const sentences = splitIntoSentences(reply);
   866→   866→        const firstSentence = sentences[0] || reply;
   867→   867→        const rest = sentences.slice(1);
   868→   868→
   869→   869→        // Seed the agent message with the first sentence.
   870→   870→        setMessages((prev) => [
   871→   871→          ...prev,
   872→   872→          {
   873→   873→            id: agentMsgId,
   874→   874→            role: "agent",
   875→   875→            content: firstSentence,
   876→   876→            ts: Date.now(),
   877→   877→          },
   878→   878→        ]);
   879→   879→
   880→   880→        executeActions(data.actions);
   881→   881→
   882→   882→        // Speak the first sentence immediately; subsequent sentences are
   883→   883→        // queued by the chunked speaker (its onend triggers the next).
   884→   884→        speakChunked(reply, {
   885→   885→          onAllDone: () => {
   886→   886→            // After all chunks have played, ensure the full text is in the
   887→   887→            // message bubble (in case the chunker appended progressively
   888→   888→            // faster/slower than expected — this is idempotent).
   889→   889→          },
   890→   890→        });
   891→   891→
   892→   892→        // Progressive display: append subsequent sentences at a cadence
   893→   893→        // that roughly matches the speaking rate. We use a timer based on
   894→   894→        // sentence length to avoid the message appearing all at once.
   895→   895→        if (rest.length > 0) {
   896→   896→          let delay = 600; // initial delay gives the first utterance a beat
   897→   897→          for (const s of rest) {
   898→   898→            window.setTimeout(() => {
   899→   899→              appendToMessage(agentMsgId, " " + s);
   900→   900→            }, delay);
   901→   901→            delay += Math.max(800, s.length * 55);
   902→   902→          }
   903→   903→        }
   904→   904→      } catch {
   905→   905→        const errMsg =
   906→   906→          "I'm having trouble reaching the security core. Try again in a moment.";
   907→   907→        setMessages((prev) => [
   908→   908→          ...prev,
   909→   909→          {
   910→   910→            id: `err-${Date.now()}`,
   911→   911→            role: "agent",
   912→   912→            content: errMsg,
   913→   913→            ts: Date.now(),
   914→   914→            kind: "error",
   915→   915→          },
   916→   916→        ]);
   917→   917→        speakChunked(errMsg);
   918→   918→      } finally {
   919→   919→        setThinking(false);
   920→   920→      }
   921→   921→    },
   922→   922→    [appendToMessage, executeActions, speakChunked, stopSpeaking],
   923→   923→  );
   924→   924→
   925→   925→  // ── On open: fetch briefing ONCE, greet + speak, start listening after TTS ─
   926→   926→  useEffect(() => {
   927→   927→    if (!open) return;
   928→   928→    if (greetedRef.current) return;
   929→   929→    greetedRef.current = true;
   930→   930→    let cancelled = false;
   931→   931→
   932→   932→    (async () => {
   933→   933→      try {
   934→   934→        const res = await fetch("/api/agent-x/briefing");
   935→   935→        if (!res.ok) return;
   936→   936→        const data = (await res.json()) as AgentXBriefing;
   937→   937→        if (cancelled) return;
   938→   938→        setBriefing(data);
   939→   939→
   940→   940→        // Seed the proactive-monitoring baseline with patch IDs (not counts).
   941→   941→        const patchIds = new Set<string>(
   942→   942→          (data.pendingTasks ?? [])
   943→   943→            .filter((t) => t.type === "patch")
   944→   944→            .map((t) => t.id),
   945→   945→        );
   946→   946→        lastPatchIdsRef.current = patchIds;
   947→   947→
   948→   948→        const firstName = (currentUser?.name || "there").split(" ")[0];
   949→   949→        const postureSummary = buildPostureSummary(data);
   950→   950→        const greeting = `Good ${data.timeOfDay || "day"}, ${firstName}. ${postureSummary} What are you up to today?`;
