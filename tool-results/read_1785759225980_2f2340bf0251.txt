   176→// ── Helpers ──────────────────────────────────────────────────────────────────
   177→function timeAgo(iso: string): string {
   178→  if (!iso) return "-";
   179→  const diff = Date.now() - new Date(iso).getTime();
   180→  const sec = Math.floor(diff / 1000);
   181→  if (sec < 60) return `${sec}s ago`;
   182→  const min = Math.floor(sec / 60);
   183→  if (min < 60) return `${min}m ago`;
   184→  const hr = Math.floor(min / 60);
   185→  if (hr < 24) return `${hr}h ago`;
   186→  return new Date(iso).toLocaleDateString();
   187→}
   188→
   189→function formatTimestamp(iso: string): string {
   190→  if (!iso) return "-";
   191→  try {
   192→    return new Date(iso).toLocaleString("en-US", {
   193→      month: "short", day: "numeric", hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false,
   194→    });
   195→  } catch {
   196→    return iso;
   197→  }
   198→}

   297→      const list = Array.isArray(cData) ? cData : (cData.integrations || []);

   301→        variant: "destructive",
