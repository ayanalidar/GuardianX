   669→// ── Helper: determine stage status from client data ────────────────────────
   670→function getStageStatus(client: ClientSummary, stageKey: string): "pending" | "in-progress" | "completed" {
   671→  const s = client.stats;
   672→  switch (stageKey) {
   673→    case "onboarding":
   674→      return (s.codebases > 0 || s.targets > 0) && client.authorized ? "completed" : (s.codebases > 0 || s.targets > 0) ? "in-progress" : "pending";
   675→    case "scanning":
   676→      return s.patches > 0 || s.findings > 0 ? "completed" : client.status === "scanning" ? "in-progress" : "pending";
   677→    case "testing":
   678→      return s.findings > 0 ? "completed" : client.status === "testing" ? "in-progress" : "pending";
   679→    case "patching":
   680→      return s.approved_patches > 0 ? "completed" : s.pending_patches > 0 ? "in-progress" : "pending";
   681→    case "verifying":
   682→      return s.approved_patches > 0 ? "completed" : "pending";
   683→    case "defending":
   684→      return client.status === "defending" || client.status === "compliant" ? "completed" : "pending";
   685→    case "compliant":
   686→      return client.status === "compliant" ? "completed" : "pending";
   687→    default:
   688→      return "pending";
   689→  }
   690→}
