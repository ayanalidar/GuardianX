     8→     8→import {
     9→     9→  ArrowLeft, Building2, Globe, GitBranch, Shield, Loader2, CheckCircle2,

    15→    15→

    59→    59→

    63→    63→  testing:    { text: "text-amber-400",   bg: "bg-amber-500/10",   border: "border-amber-500/40",   neon: "neon-amber" },
    64→    64→  patching:   { text: "text-violet-400",  bg: "bg-violet-500/10",  border: "border-violet-500/40",  neon: "neon-violet" },

    67→    67→  compliant:  { text: "text-emerald-400", bg: "bg-emerald-500/10", border: "border-emerald-500/40", neon: "neon-emerald" },
    68→    68→};
    69→    69→
    70→    70→interface ClientDetailProps {
    71→    71→  clientId: string;
    72→    72→  onBack: () => void;
    73→    73→  onNavigate: (tab: string) => void;

    75→    75→
    76→    76→export function ClientDetail({ clientId, onBack }: ClientDetailProps) {
    77→    77→  const { toast } = useToast();
    78→    78→  const [detail, setDetail] = useState<ClientFullDetail | null>(null);
    79→    79→  const [pipeline, setPipeline] = useState<ClientDetailData | null>(null);
    80→    80→  const [loading, setLoading] = useState(true);
    81→    81→  const [authorizing, setAuthorizing] = useState(false);
    82→    82→  const [showAddCodebase, setShowAddCodebase] = useState(false);
    83→    83→  const [showAddTarget, setShowAddTarget] = useState(false);
    84→    84→  const [vaptRunning, setVaptRunning] = useState(false);
    85→    85→  const [vaptResult, setVaptResult] = useState<any>(null);
    86→    86→  const [deleting, setDeleting] = useState(false);
    87→    87→
    88→    88→  // Add codebase form state
    89→    89→  const [cbName, setCbName] = useState("");
    90→    90→  const [cbSource, setCbSource] = useState("");
    91→    91→  const [cbDesc, setCbDesc] = useState("");
    92→    92→  const [cbSaving, setCbSaving] = useState(false);
    93→    93→
    94→    94→  // Add target form state
    95→    95→  const [tgtName, setTgtName] = useState("");
    96→    96→  const [tgtUrl, setTgtUrl] = useState("");
    97→    97→  const [tgtAuth, setTgtAuth] = useState(false);
    98→    98→  const [tgtSaving, setTgtSaving] = useState(false);
    99→    99→
   100→   100→  const load = useCallback(async () => {
   101→   101→    setLoading(true);
   102→   102→    try {
   103→   103→      const [dRes, pRes] = await Promise.all([
   104→   104→        fetch(`/api/clients/${clientId}`),
   105→   105→        fetch(`/api/clients/${clientId}/pipeline`),
   106→   106→      ]);
   107→   107→      const d = await dRes.json();
   108→   108→      const p = await pRes.json();
   109→   109→      if (!d.error) setDetail(d);
   110→   110→      if (!p.error) setPipeline(p);
   111→   111→    } catch {
   112→   112→      // ignore
   113→   113→    } finally {
   114→   114→      setLoading(false);
   115→   115→    }
   116→   116→  }, [clientId]);
   117→   117→
   118→   118→  useEffect(() => {
   119→   119→    load();
   120→   120→  }, [load]);
   121→   121→
   122→   122→  const handleAddCodebase = async () => {
   123→   123→    if (!cbName.trim() || !cbSource.trim()) return;
   124→   124→    setCbSaving(true);
   125→   125→    try {
   126→   126→      await sentinelApi.createCodebase({
   127→   127→        name: cbName.trim(),
   128→   128→        sourceCode: cbSource,
   129→   129→        description: cbDesc.trim() || undefined,
   130→   130→        clientId: clientId,
   131→   131→      });
   132→   132→      toast({ title: "Codebase added!", description: cbName.trim() });
   133→   133→      setCbName(""); setCbSource(""); setCbDesc("");
   134→   134→      setShowAddCodebase(false);
   135→   135→      load();
   136→   136→    } catch (err) {
   137→   137→      toast({ variant: "destructive", title: "Failed", description: err instanceof Error ? err.message : "unknown" });
   138→   138→    } finally {
   139→   139→      setCbSaving(false);
   140→   140→    }
   141→   141→  };
   142→   142→
   143→   143→  const handleAddTarget = async () => {
   144→   144→    if (!tgtName.trim() || !tgtUrl.trim()) return;
   145→   145→    setTgtSaving(true);
   146→   146→    try {
   147→   147→      await sentinelApi.addTarget({
   148→   148→        name: tgtName.trim(),
   149→   149→        baseUrl: tgtUrl.trim(),
   150→   150→        authorized: tgtAuth,
   151→   151→        clientId: clientId,
   152→   152→      });
   153→   153→      toast({ title: "Target added!", description: tgtName.trim() });
   154→   154→      setTgtName(""); setTgtUrl(""); setTgtAuth(false);
   155→   155→      setShowAddTarget(false);
   156→   156→      load();
   157→   157→    } catch (err) {
   158→   158→      toast({ variant: "destructive", title: "Failed", description: err instanceof Error ? err.message : "unknown" });
   159→   159→    } finally {
   160→   160→      setTgtSaving(false);
   161→   161→    }
   162→   162→  };
   163→   163→
   164→   164→  const handleFullVapt = async () => {
   165→   165→    setVaptRunning(true);
   166→   166→    setVaptResult(null);
   167→   167→    pushTerminalLine({ text: `$ guardianx full-vapt --client "${detail?.name}" --clientId ${clientId}`, type: "cmd" });
   168→   168→    pushTerminalLine({ text: "[*] Initiating 7-stage pipeline: discover → recon → scan → test → patch → verify → defend", type: "out" });
   169→   169→    try {
   170→   170→      const res = await fetch("/api/full-vapt", {
   171→   171→        method: "POST",
   172→   172→        headers: { "Content-Type": "application/json" },
   173→   173→        body: JSON.stringify({ clientId }),
   174→   174→      });
   175→   175→      const data = await res.json();
   176→   176→      setVaptResult(data);
   177→   177→      if (res.ok) {
   178→   178→        pushTerminalLine({ text: `[+] Full VAPT launched, ${data.message || "pipeline running"}`, type: "success" });
   179→   179→        toast({ title: "Full VAPT Launched!", description: data.message });
   180→   180→      } else {
   181→   181→        pushTerminalLine({ text: `[!] VAPT failed: ${data.error || "unknown error"}`, type: "err" });
   182→   182→        toast({ variant: "destructive", title: "VAPT failed to start" });
   183→   183→      }
   184→   184→      load();
   185→   185→    } catch (err) {
   186→   186→      pushTerminalLine({ text: `[!] VAPT exception: ${err instanceof Error ? err.message : "unknown"}`, type: "err" });
   187→   187→      toast({ variant: "destructive", title: "VAPT failed to start" });
   188→   188→    }
   189→   189→    setVaptRunning(false);
   190→   190→  };
   191→   191→
   192→   192→  const handleDelete = async () => {
   193→   193→    if (!confirm(`Delete "${detail?.name}"? This will remove the client and all associated data (codebases, targets, scans, findings). This cannot be undone.`)) return;
   194→   194→    setDeleting(true);
   195→   195→    try {
   196→   196→      const res = await fetch(`/api/clients/${clientId}`, { method: "DELETE" });
   197→   197→      if (res.ok) {
   198→   198→        toast({ title: "Client deleted", description: detail?.name });
   199→   199→        onBack();
   200→   200→      } else {
   201→   201→        toast({ variant: "destructive", title: "Delete failed" });
   202→   202→      }
   203→   203→    } catch {
   204→   204→      toast({ variant: "destructive", title: "Delete failed" });
   205→   205→    }
   206→   206→    setDeleting(false);
   207→   207→  };
   208→   208→
   209→   209→  const handleAuthorize = async () => {
   210→   210→    setAuthorizing(true);
   211→   211→    try {
   212→   212→      const res = await fetch(`/api/clients/${clientId}`, {
   213→   213→        method: "PATCH",
   214→   214→        headers: { "Content-Type": "application/json" },
   215→   215→        body: JSON.stringify({ authorized: !detail?.authorized }),
   216→   216→      });
   217→   217→      const data = await res.json();
   218→   218→      if (!res.ok) throw new Error(data.error);
   219→   219→      toast({
   220→   220→        title: detail?.authorized ? "Authorization revoked" : "Client authorized",
   221→   221→        description: detail?.authorized
   222→   222→          ? "Testing is now blocked until re-authorized."
   223→   223→          : "You can now run SAST + DAST scans on this client.",
   224→   224→      });
   225→   225→      load();
   226→   226→    } catch (err) {
   227→   227→      toast({
   228→   228→        variant: "destructive",
   229→   229→        title: "Failed",
   230→   230→        description: err instanceof Error ? err.message : "unknown",
   231→   231→      });
   232→   232→    } finally {
   233→   233→      setAuthorizing(false);
   234→   234→    }
   235→   235→  };
   236→   236→
   237→   237→  if (loading) {
   238→   238→    return (
   239→   239→      <div className="flex items-center justify-center py-20">
   240→   240→        <Loader2 className="size-8 animate-spin text-emerald-400" />
   241→   241→      </div>
   242→   242→    );
   243→   243→  }
   244→   244→
   245→   245→  if (!detail) {
   246→   246→    return (
   247→   247→      <div className="flex flex-col items-center justify-center py-20 text-center">
   248→   248→        <AlertCircle className="size-12 text-red-400" />
   249→   249→        <h3 className="mt-4 text-lg font-semibold text-zinc-200">Client not found</h3>
   250→   250→        <Button onClick={onBack} className="mt-4 bg-emerald-600 text-white hover:bg-emerald-500">
   251→   251→          <ArrowLeft className="size-4" /> Back to Clients
   252→   252→        </Button>
   253→   253→      </div>
   254→   254→    );
   255→   255→  }
   256→   256→
   257→   257→  return (
   258→   258→    <div className="space-y-5">
   259→   259→      {/* Header */}
   260→   260→      <div className="flex items-start justify-between gap-4">
   261→   261→        <div className="flex items-start gap-3">
   262→   262→          <Button variant="ghost" size="icon" onClick={onBack} className="text-zinc-400 hover:text-emerald-400">
   263→   263→            <ArrowLeft className="size-5" />
   264→   264→          </Button>
   265→   265→          <div>
   266→   266→            <div className="flex items-center gap-2">
   267→   267→              <h2 className="text-xl font-bold text-zinc-50 neon-emerald">{detail.name}</h2>
   268→   268→              {detail.authorized ? (
   269→   269→                <Badge className="border-emerald-500/30 bg-emerald-500/10 text-[10px] text-emerald-300">AUTHORIZED</Badge>
   270→   270→              ) : (
   271→   271→                <Badge className="border-amber-500/30 bg-amber-500/10 text-[10px] text-amber-300">PENDING AUTH</Badge>
   272→   272→              )}
   273→   273→            </div>
   274→   274→            {detail.description && (
   275→   275→              <p className="mt-1 text-sm text-zinc-400">{detail.description}</p>
   276→   276→            )}
   277→   277→            <div className="mt-2 flex flex-wrap items-center gap-3 text-xs text-zinc-500">
   278→   278→              {detail.target_url && (
   279→   279→                <a href={detail.target_url} target="_blank" rel="noopener noreferrer" className="flex items-center gap-1 hover:text-cyan-400">
   280→   280→                  <Globe className="size-3" /> {detail.target_url}
   281→   281→                </a>
   282→   282→              )}
   283→   283→              {detail.repo_url && (
   284→   284→                <a href={detail.repo_url} target="_blank" rel="noopener noreferrer" className="flex items-center gap-1 hover:text-violet-400">
   285→   285→                  <GitBranch className="size-3" /> {detail.repo_url}
   286→   286→                </a>
   287→   287→              )}
   288→   288→              {detail.contact_email && (
   289→   289→                <a href={`mailto:${detail.contact_email}`} className="flex items-center gap-1 hover:text-emerald-400">
   290→   290→                  <Mail className="size-3" /> {detail.contact_email}
   291→   291→                </a>
   292→   292→              )}
   293→   293→              {detail.contact_phone && (
   294→   294→                <a href={`tel:${detail.contact_phone}`} className="flex items-center gap-1 hover:text-emerald-400">
   295→   295→                  <Phone className="size-3" /> {detail.contact_phone}
   296→   296→                </a>
   297→   297→              )}
   298→   298→            </div>
   299→   299→          </div>
   300→   300→        </div>
   301→   301→        <div className="flex items-center gap-2">
   302→   302→          <Button
   303→   303→            onClick={handleFullVapt}
   304→   304→            disabled={vaptRunning}
   305→   305→            className="bg-emerald-600 text-white hover:bg-emerald-500 neon-border"
   306→   306→          >
   307→   307→            {vaptRunning ? <Loader2 className="size-4 animate-spin" /> : <Rocket className="size-4" />}
   308→   308→            <span className="hidden sm:inline">One-Click Full VAPT</span>
   309→   309→          </Button>
   310→   310→          <Button
   311→   311→            onClick={handleAuthorize}
   312→   312→            disabled={authorizing}
   313→   313→            variant="outline"
   314→   314→            className={
   315→   315→              detail.authorized
   316→   316→                ? "border-red-500/40 bg-red-500/10 text-red-300 hover:bg-red-500/20"
   317→   317→                : "border-emerald-500/40 bg-emerald-500/10 text-emerald-300 hover:bg-emerald-500/20"
   318→   318→            }
   319→   319→          >
   320→   320→            {authorizing ? <Loader2 className="size-4 animate-spin" /> : <Shield className="size-4" />}
   321→   321→            {detail.authorized ? "Revoke Auth" : "Authorize"}
   322→   322→          </Button>
   323→   323→          <Button
   324→   324→            onClick={handleDelete}
   325→   325→            disabled={deleting}
   326→   326→            variant="ghost"
   327→   327→            size="icon"
   328→   328→            className="text-zinc-500 hover:bg-red-500/10 hover:text-red-400"
   329→   329→            title="Delete client"
   330→   330→          >
   331→   331→            {deleting ? <Loader2 className="size-4 animate-spin" /> : <Trash2 className="size-4" />}
   332→   332→          </Button>
   333→   333→        </div>
   334→   334→      </div>
   335→   335→
   336→   336→      {/* Full VAPT result */}
   337→   337→      {vaptResult && (
   338→   338→        <div className="holo-card-sharp hud-corners border-emerald-500/30 p-4">
   339→   339→          <h3 className="mb-2 text-sm font-bold text-emerald-300">Full VAPT Status</h3>
   340→   340→          <div className="space-y-1">
   341→   341→            {vaptResult.steps?.map((s: any, i: number) => (
   342→   342→              <div key={i} className="flex items-center gap-2 text-xs">
   343→   343→                <span className={`size-2 rounded-full ${s.status === "completed" ? "bg-emerald-500" : s.status === "running" ? "bg-cyan-500 animate-pulse" : "bg-zinc-600"}`} />
   344→   344→                <span className="text-zinc-400 font-mono">{s.step}.</span>
   345→   345→                <span className="text-zinc-200">{s.action}</span>
   346→   346→                <span className="text-zinc-500">- {s.detail}</span>
   347→   347→              </div>
   348→   348→            ))}
   349→   349→          </div>
   350→   350→          <p className="mt-2 text-xs text-emerald-300">{vaptResult.message}</p>
   351→   351→        </div>
   352→   352→      )}
   353→   353→
   354→   354→      {/* Frameworks */}
   355→   355→      {detail.frameworks.length > 0 && (
   356→   356→        <div className="flex flex-wrap items-center gap-2">
   357→   357→          <span className="font-mono text-[10px] uppercase tracking-widest text-zinc-600">Compliance:</span>
   358→   358→          {detail.frameworks.map((fw) => (
   359→   359→            <Badge key={fw} className="border-purple-500/30 bg-purple-500/10 text-[10px] text-purple-300">{fw}</Badge>
   360→   360→          ))}
   361→   361→        </div>
   362→   362→      )}
   363→   363→
   364→   364→      {/* Pipeline visualization */}
   365→   365→      {pipeline && (
   366→   366→        <div className="holo-card-sharp hud-corners p-5">
   367→   367→          <div className="mb-4 flex items-center justify-between">
   368→   368→            <h3 className="text-sm font-bold text-zinc-100 section-header">Security Pipeline</h3>
   369→   369→            <div className="flex items-center gap-2">
   370→   370→              <span className="font-mono text-xs text-zinc-500">Progress:</span>
   371→   371→              <span className="font-bold text-emerald-400 neon-emerald">{pipeline.progress}%</span>
   372→   372→            </div>
   373→   373→          </div>
   374→   374→
   375→   375→          {/* Horizontal pipeline stepper, clickable to launch service */}
   376→   376→          <div className="flex items-center gap-1 overflow-x-auto pb-2">
   377→   377→            {pipeline.stages.map((stage, i) => {
   378→   378→              const cfg = STAGE_COLORS[stage.key] || STAGE_COLORS.onboarding;
   379→   379→              const Icon = STAGE_ICONS[stage.key] || Circle;
   380→   380→              const serviceMap: Record<string, string> = {
   381→   381→                scanning: "scan", testing: "test", patching: "patch",
   382→   382→                verifying: "verify", defending: "defend", compliant: "comply",
   383→   383→              };
   384→   384→              const canLaunch = serviceMap[stage.key] && stage.status !== "completed" && detail.authorized;
   385→   385→              return (
   386→   386→                <div key={stage.id} className="flex items-center">
   387→   387→                  <motion.button
   388→   388→                    initial={{ opacity: 0, scale: 0.9 }}
   389→   389→                    animate={{ opacity: 1, scale: 1 }}
   390→   390→                    transition={{ delay: i * 0.08 }}
   391→   391→                    onClick={async () => {
   392→   392→                      if (!canLaunch) return;
   393→   393→                      const service = serviceMap[stage.key];
   394→   394→                      pushTerminalLine({ text: `$ guardianx launch --service ${service} --client "${detail?.name}"`, type: "cmd" });
   395→   395→                      try {
   396→   396→                        const res = await fetch("/api/launch-service", {
   397→   397→                          method: "POST",
   398→   398→                          headers: { "Content-Type": "application/json" },
   399→   399→                          body: JSON.stringify({ service, clientIds: [clientId] }),
   400→   400→                        });
   401→   401→                        const data = await res.json();
   402→   402→                        if (res.ok) {
   403→   403→                          pushTerminalLine({ text: `[+] ${stage.label} launched, ${data.message || "ok"}`, type: "success" });
   404→   404→                          toast({ title: `${stage.label} launched`, description: data.message });
   405→   405→                        } else {
   406→   406→                          pushTerminalLine({ text: `[!] ${stage.label} failed: ${data.error || "error"}`, type: "err" });
   407→   407→                          toast({ variant: "destructive", title: "Launch failed" });
   408→   408→                        }
   409→   409→                        load();
   410→   410→                      } catch (err) {
   411→   411→                        pushTerminalLine({ text: `[!] ${stage.label} exception: ${err instanceof Error ? err.message : "unknown"}`, type: "err" });
   412→   412→                        toast({ variant: "destructive", title: "Launch failed" });
   413→   413→                      }
