   142→   142→// Map a raw timeline "type" string to one of our EVENT_TYPE_CONFIG keys.
   143→   143→function normalizeEventType(rawType: string): string {
   144→   144→  if (!rawType) return "note";
   145→   145→  // strip "incident." prefix if present
   146→   146→  const t = rawType.startsWith("incident.") ? rawType.slice("incident.".length) : rawType;
   147→   147→  if (t in EVENT_TYPE_CONFIG) return t;
   148→   148→  if (t === "audit") return "note";
   149→   149→  if (t === "api_access" || t === "api_log") return "anomaly";
   150→   150→  return "note";
   151→   151→}

   164→   164→function timeAgo(iso: string): string {
   165→   165→  if (!iso) return "-";
   166→   166→  const diff = Date.now() - new Date(iso).getTime();
   167→   167→  const sec = Math.floor(diff / 1000);
   168→   168→  if (sec < 60) return `${sec}s ago`;
   169→   169→  const min = Math.floor(sec / 60);
   170→   170→  if (min < 60) return `${min}m ago`;
   171→   171→  const hr = Math.floor(min / 60);
   172→   172→  if (hr < 24) return `${hr}h ago`;
   173→   173→  const day = Math.floor(hr / 24);
   174→   174→  if (day < 30) return `${day}d ago`;
   175→   175→  return new Date(iso).toLocaleDateString();
   176→   176→}

   178→   178→function truncateHash(h: string, len = 16): string {
   179→   179→  if (!h) return "";
   180→   180→  if (h.length <= len * 2) return h;
   181→   181→  return `${h.slice(0, len)}...${h.slice(-8)}`;
   182→   182→}

   184→   184→// Safely unwrap an API response that may be either a bare array or an
   185→   185→// object wrapping the array under a known key. The parallel-agent routes
   186→   186→// wrap responses as { iocs: [...] }, { evidence: [...] }, { playbooks: [...] },
   187→   187→// { timeline: [...] }, while the older list endpoints return bare arrays.
   188→   188→function unwrapList<T>(data: unknown, key: string): T[] {
   189→   189→  if (Array.isArray(data)) return data as T[];
   190→   190→  if (data && typeof data === "object") {
   191→   191→    const obj = data as Record<string, unknown>;
   192→   192→    if (Array.isArray(obj[key])) return obj[key] as T[];
   193→   193→  }
   194→   194→  return [];
   195→   195→}
   196→   196→
   197→   197→// Normalize a raw timeline entry (which may come from the unified timeline
   198→   198→// endpoint with { timestamp, type, title, description, severity, source,
   199→   199→// metadata } OR from a raw IncidentEvent with { occurredAt, eventType, ... })
   200→   200→// into our TimelineEntry shape.
   201→   201→function normalizeTimelineEntry(raw: Record<string, unknown>): TimelineEntry {
   202→   202→  const meta = raw.metadata as Record<string, unknown> | string | null;
   203→   203→  const metaObj = typeof meta === "string" ? (() => { try { return JSON.parse(meta); } catch { return {}; } })() : (meta || {});
   204→   204→  const ts = (raw.occurredAt || raw.timestamp || new Date().toISOString()) as string;
   205→   205→  return {
   206→   206→    id: (raw.id as string) || (metaObj?.eventId as string) || `${ts}-${raw.title}`,
   207→   207→    eventType: normalizeEventType((raw.eventType as string) || (raw.type as string) || "note"),
   208→   208→    source: (raw.source as string) || "system",
   209→   209→    sourceId: (raw.sourceId as string) || null,
   210→   210→    title: (raw.title as string) || "Untitled event",
   211→   211→    description: (raw.description as string) || null,
   212→   212→    severity: (raw.severity as string) || "info",
   213→   213→    metadata: typeof meta === "string" ? meta : (meta ? JSON.stringify(meta) : null),
   214→   214→    actor: (raw.actor as string) || (metaObj?.actor as string) || null,
   215→   215→    occurredAt: ts,
   216→   216→  };
   217→   217→}
