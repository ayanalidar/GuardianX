   260→// ── Hook: load all settings once on mount ───────────────────────────────────
   261→function useSettingsLoader() {
   262→  const [settings, setSettings] = useState<SettingsMap>(DEFAULTS);
   263→  const [loaded, setLoaded] = useState(false);
   264→
   265→  const load = useCallback(async () => {
   266→    try {
   267→      const res = await fetch("/api/settings");
   268→      const data = await res.json();
   269→      if (!res.ok) throw new Error(data.error || "Failed to load settings");
   270→      const next: SettingsMap = { ...DEFAULTS };
   271→      const incoming = (data.settings || {}) as Partial<SettingsMap>;
   272→      (Object.keys(DEFAULTS) as SettingKey[]).forEach((k) => {
   273→        const row = incoming[k];
   274→        if (row && row.config && typeof row.config === "object") {
   275→          // Strip the internal _key marker if it slipped through
   276→          const clean = { ...row.config };
   277→          delete (clean as Record<string, unknown>)._key;
   278→          next[k] = {
   279→            id: row.id || "",
   280→            isActive: !!row.isActive,
   281→            createdAt: row.createdAt || "",
   282→            config: { ...(DEFAULTS[k].config as Record<string, unknown>), ...clean },
   283→          };
   284→        }
   285→      });
   286→      setSettings(next);
   287→    } catch {
   288→      // silently fall back to defaults
   289→      setSettings(DEFAULTS);
   290→    } finally {
   291→      setLoaded(true);
   292→    }
   293→  }, []);
   294→
   295→  useEffect(() => {
   296→    load();
   297→  }, [load]);
   298→
   299→  return { settings, loaded, reload: load, setSettings };
   300→}

   471→async function saveSetting(key: SettingKey, config: Record<string, unknown>, isActive: boolean) {
   472→  const res = await fetch("/api/settings", {
   473→    method: "POST",
   474→    headers: { "Content-Type": "application/json" },
   475→    body: JSON.stringify({ key, config, isActive }),
   476→  });
   477→  const data = await res.json();
   478→  if (!res.ok) throw new Error(data.error || `Failed to save ${key}`);
   479→  return data;
   480→}

   482→async function testChannel(channel: string, config: Record<string, unknown>, testTarget: string) {
   483→  const res = await fetch("/api/settings", {
   484→    method: "PATCH",
   485→    headers: { "Content-Type": "application/json" },
   486→    body: JSON.stringify({ action: "test", channel, config, testTarget }),
   487→  });
   488→  const data = await res.json();
   489→  if (!res.ok) {
   490→    throw new Error(data.message || data.error || "Test failed");
   491→  }
   492→  return data as { success?: boolean; message?: string; detail?: string };
   493→}
