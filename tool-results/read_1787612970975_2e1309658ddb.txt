   107→export function CommandCenter({ onSelectClient, onAddClient }: CommandCenterProps) {
   108→  const [clients, setClients] = useState<ClientSummary[]>([]);
   109→  const [feed, setFeed] = useState<ActivityFeed | null>(null);
   110→  const [loading, setLoading] = useState(true);
   111→  const [clock, setClock] = useState(new Date());
   112→  const [threatLevel, setThreatLevel] = useState(0);
   113→  const [warRoom, setWarRoom] = useState(false);
   114→  const [chatOpen, setChatOpen] = useState(false);
   115→  const [immersiveOpen, setImmersiveOpen] = useState(false);
   116→  const [opsLoading, setOpsLoading] = useState<string | null>(null);
   117→  const [selectedClientIds, setSelectedClientIds] = useState<string[]>([]); // multi-select for parallel ops
   118→  const [showClientSelector, setShowClientSelector] = useState(false);
   119→  const [launcherOpen, setLauncherOpen] = useState(false);
   120→  const [launcherClients, setLauncherClients] = useState<string[]>([]);
   121→  const logRef = useRef<HTMLDivElement>(null);
   122→
   123→  // Live clock
   124→  useEffect(() => {
   125→    const id = setInterval(() => setClock(new Date()), 1000);
   126→    return () => clearInterval(id);
   127→  }, []);
   128→
   129→  const load = useCallback(async () => {
   130→    try {
   131→      const [cRes, fRes] = await Promise.all([
   132→        fetch("/api/clients"),
   133→        fetch("/api/activity-feed"),
   134→      ]);
   135→      const c = await cRes.json();
   136→      const f = await fRes.json();

   138→      if (f && !f.error) setFeed(f);
   139→

   141→      if (f && !f.error) {
   142→        const criticalCount = f.stats.critical;
   143→        const activeCount = f.active_processes.total_active;
   144→        // Threat level: 0-100, scaled from critical findings + active processes
   145→        setThreatLevel(Math.min(100, criticalCount * 8 + activeCount * 5));
   146→      }
   147→    } catch {
   148→      // ignore
   149→    } finally {
   150→      setLoading(false);
   151→    }
   152→  }, []);
   153→
   154→  useEffect(() => {
   155→    load();
   156→    // Visibility-aware polling: pauses when the tab is hidden so the
   157→    // dashboard doesn't burn CPU + network while the user is elsewhere.
   158→    // Previously this was a bare setInterval(load, 15000).
   159→    let interval: ReturnType<typeof setInterval> | null = null;
   160→    const start = () => {
   161→      if (interval) return;
   162→      interval = setInterval(load, 15000);
   163→    };
   164→    const stop = () => {
   165→      if (interval) {
   166→        clearInterval(interval);
   167→        interval = null;
   168→      }
   169→    };
   170→    const onVisibility = () => {
   171→      if (document.hidden) stop();
   172→      else start();
   173→    };
   174→    document.addEventListener("visibilitychange", onVisibility);
   175→    start();
   176→    return () => {
   177→      stop();
   178→      document.removeEventListener("visibilitychange", onVisibility);
   179→    };
   180→  }, [load]);
   181→
   182→  // Compute aggregate stats
   183→  const stats = {

   185→    active_pipelines: clients.filter((c) => c.status !== "onboarding" && c.status !== "compliant").length,
   186→    total_scans: feed?.active_processes.sast_scans || 0,
   187→    total_patches: clients.reduce((s, c) => s + c.stats.patches, 0),
   188→    pending_patches: clients.reduce((s, c) => s + c.stats.pending_patches, 0),
   189→    total_findings: clients.reduce((s, c) => s + c.stats.findings, 0),
   190→    critical_findings: clients.reduce((s, c) => s + c.stats.critical_findings, 0),
   191→    compliant_clients: clients.filter((c) => c.status === "compliant").length,
   192→  };
   193→
   194→  const threatColor = threatLevel >= 60 ? "red" : threatLevel >= 30 ? "amber" : "emerald";
   195→  const threatCfg = COLOR_MAP[threatColor];
   196→
   197→  return (
   198→    <SignalBusProvider>
   199→    <div className="space-y-4">
   200→      {/* ═══ FUTURISTIC HEADER ═══ */}

   202→        <div aria-hidden className="cyber-grid pointer-events-none absolute inset-0 opacity-30" />
   203→        <div aria-hidden className="pointer-events-none absolute inset-0">
   204→          <div className="absolute -top-20 left-1/3 h-40 w-96 -translate-x-1/2 rounded-full bg-emerald-500/15 blur-3xl" />
   205→          <div className="absolute -bottom-10 right-0 h-32 w-64 rounded-full bg-red-500/10 blur-3xl" />
   206→        </div>

   208→          <div className="flex items-center gap-3">

   211→                <Radar className="size-6 text-emerald-400" />
   212→              </div>
   213→              <span className="absolute -right-1 -top-1 size-3 rounded-full bg-emerald-500 pulse-dot" />
   214→            </div>
   215→            <div>
   216→              <h2 className="text-2xl font-bold tracking-tight text-zinc-50">
   217→                <span className="neon-emerald">COMMAND</span>{" "}
   218→                <span className="neon-cyan">CENTER</span>
   219→              </h2>
   220→              <p className="font-mono text-[10px] uppercase tracking-widest text-emerald-500/60">
   221→                {"// real-time security operations // all clients // all services"}
   222→              </p>
   223→            </div>
   224→          </div>
   225→
   226→          <div className="flex items-center gap-3">
   227→            {/* Threat level gauge */}
   228→            <div className={`flex items-center gap-2 rounded-lg border ${threatCfg.border} ${threatCfg.bg} px-3 py-2`}>
   229→              <Gauge className={`size-4 ${threatCfg.text} ${threatLevel >= 30 ? "animate-pulse" : ""}`} />
   230→              <div>
   231→                <div className="font-mono text-[8px] uppercase tracking-wider text-zinc-500">Threat Level</div>
   232→                <div className={`font-mono text-sm font-bold ${threatCfg.text}`}>
   233→                  {threatLevel >= 60 ? "CRITICAL" : threatLevel >= 30 ? "ELEVATED" : "GUARDED"}
   234→                </div>
   235→              </div>
   236→              <div className="flex flex-col gap-0.5">
   237→                {[...Array(10)].map((_, i) => (
   238→                  <div
   239→                    key={i}
   240→                    className={`h-1 w-3 rounded-sm ${
   241→                      i < Math.floor(threatLevel / 10) ? threatCfg.dot : "bg-zinc-800"
   242→                    }`}
   243→                  />
   244→                ))}
   245→              </div>
   246→            </div>
   247→
   248→            {/* Live clock */}
   249→            <div className="rounded-lg border border-emerald-500/30 bg-zinc-950/80 px-3 py-2 font-mono">
   250→              <div className="text-[8px] uppercase tracking-wider text-emerald-500/60">SYS TIME</div>

   254→            </div>
   255→
   256→            <Button
   257→              onClick={() => setChatOpen(true)}

   261→              <Bot className="size-4" /> <span className="hidden sm:inline">Guardian AI</span>
   262→            </Button>
   263→            <Button
   264→              onClick={() => setWarRoom(true)}
   265→              variant="outline"
   266→              className="border-cyan-500/40 bg-cyan-500/10 text-cyan-300 hover:bg-cyan-500/20 neon-border-cyan"

   276→            </Button>

   278→              onClick={() => { setLauncherClients(selectedClientIds); setLauncherOpen(true); }}
   279→              className="bg-emerald-600 text-white hover:bg-emerald-500 neon-border"
   280→            >
   281→              <Rocket className="size-4" /> <span className="hidden sm:inline">Launch Service</span>
   282→            </Button>

   286→          </div>
   287→        </div>
   288→      </div>
   289→
   290→      {/* ═══ KPI STRIP, terminal-style with sparklines ═══ */}
   291→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-4 lg:grid-cols-8">
   292→        <KpiCard label="CLIENTS" value={stats.total_clients} icon={Building2} color="emerald" />

   295→        <KpiCard label="PATCHES" value={stats.total_patches} icon={ShieldCheck} color="violet" sparkMetric="patches" />
   296→        <KpiCard label="PENDING" value={stats.pending_patches} icon={AlertCircle} color="amber" pulse={stats.pending_patches > 0} sparkMetric="patches" />
   297→        <KpiCard label="FINDINGS" value={stats.total_findings} icon={Skull} color="red" sparkMetric="findings" />
   298→        <KpiCard label="CRITICAL" value={stats.critical_findings} icon={AlertTriangle} color="red" pulse={stats.critical_findings > 0} sparkMetric="critical" />
   299→        <KpiCard label="COMPLIANT" value={stats.compliant_clients} icon={CheckCircle2} color="emerald" />
   300→      </div>
   301→
   302→      {/* ═══ CLIENT CONTEXT SELECTOR + OPERATIONS TOOLBAR ═══ */}
   303→      <div className="holo-card-sharp hud-corners p-3">
   304→        <div className="flex flex-wrap items-center gap-2">
   305→          <span className="font-mono text-[10px] uppercase tracking-widest text-emerald-500/60">{"// Scope:"}</span>
   306→          {/* Client selector dropdown */}
   307→          <button
   308→            onClick={() => setShowClientSelector(!showClientSelector)}
   309→            className="inline-flex items-center gap-1.5 rounded-md border border-emerald-500/40 bg-emerald-500/10 px-3 py-1.5 text-xs font-medium text-emerald-300 transition-all hover:bg-emerald-500/20"
   310→          >
   311→            <Building2 className="size-3" />
   312→            {selectedClientIds.length === 0 ? "ALL CLIENTS" : `${selectedClientIds.length} SELECTED`}
   313→          </button>
   314→          {selectedClientIds.length > 0 && (
   315→            <button
   316→              onClick={() => setSelectedClientIds([])}
   317→              className="text-[10px] text-zinc-500 hover:text-red-400"
   318→            >
   319→              ✕ clear
   320→            </button>
   321→          )}
   322→
   323→          <span className="mx-1 text-zinc-700">│</span>

   362→            label="Threat Hunter"
   363→            icon={Radar}
   364→            color="cyan"
   365→            loading={opsLoading === "threat-hunter"}
   366→            disabled={!!opsLoading}
   367→            onClick={async () => {
   368→              setOpsLoading("threat-hunter");
   369→              try {
   370→                const res = await fetch("/api/threat-hunter", { method: "POST" });

   397→            <a href="/api/audit-export" target="_blank" rel="noopener noreferrer">
   398→              <OpsButton label="Audit" icon={FileDown} color="violet" onClick={() => {}} />

   400→            <a href="/api/sla-tracking" target="_blank" rel="noopener noreferrer">
   401→              <OpsButton label="SLA" icon={Gauge} color="rose" onClick={() => {}} />

   409→        {/* Client selector dropdown */}

   433→                <span className="truncate text-zinc-300">{c.name}</span>

   800→// ── Helper: determine stage status from client data ────────────────────────
   801→function getStageStatus(client: ClientSummary, stageKey: string): "pending" | "in-progress" | "completed" {
   802→  const s = client.stats;
   803→  switch (stageKey) {
   804→    case "onboarding":
   805→      return (s.codebases > 0 || s.targets > 0) && client.authorized ? "completed" : (s.codebases > 0 || s.targets > 0) ? "in-progress" : "pending";
   806→    case "scanning":
   807→      return s.patches > 0 || s.findings > 0 ? "completed" : client.status === "scanning" ? "in-progress" : "pending";
   808→    case "testing":
   809→      return s.findings > 0 ? "completed" : client.status === "testing" ? "in-progress" : "pending";
   810→    case "patching":
   811→      return s.approved_patches > 0 ? "completed" : s.pending_patches > 0 ? "in-progress" : "pending";
   812→    case "verifying":
   813→      return s.approved_patches > 0 ? "completed" : "pending";
   814→    case "defending":
   815→      return client.status === "defending" || client.status === "compliant" ? "completed" : "pending";
   816→    case "compliant":
   817→      return client.status === "compliant" ? "completed" : "pending";
   818→    default:
   819→      return "pending";
   820→  }
   821→}
