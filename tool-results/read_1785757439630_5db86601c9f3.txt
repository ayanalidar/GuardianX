   258→// ── Hook: load all settings once on mount ───────────────────────────────────
   259→function useSettingsLoader() {
   260→  const [settings, setSettings] = useState<SettingsMap>(DEFAULTS);
   261→  const [loaded, setLoaded] = useState(false);
   262→
   263→  const load = useCallback(async () => {
   264→    try {
   265→      const res = await fetch("/api/settings");
   266→      const data = await res.json();
   267→      if (!res.ok) throw new Error(data.error || "Failed to load settings");
   268→      const next: SettingsMap = { ...DEFAULTS };
   269→      const incoming = (data.settings || {}) as Partial<SettingsMap>;
   270→      (Object.keys(DEFAULTS) as SettingKey[]).forEach((k) => {
   271→        const row = incoming[k];
   272→        if (row && row.config && typeof row.config === "object") {
   273→          // Strip the internal _key marker if it slipped through
   274→          const clean = { ...row.config };
   275→          delete (clean as Record<string, unknown>)._key;
   276→          next[k] = {
   277→            id: row.id || "",
   278→            isActive: !!row.isActive,
   279→            createdAt: row.createdAt || "",
   280→            config: { ...(DEFAULTS[k].config as Record<string, unknown>), ...clean },
   281→          };
   282→        }
   283→      });
   284→      setSettings(next);
   285→    } catch {
   286→      // silently fall back to defaults
   287→      setSettings(DEFAULTS);
   288→    } finally {
   289→      setLoaded(true);
   290→    }
   291→  }, []);
   292→
   293→  useEffect(() => {
   294→    load();
   295→  }, [load]);
   296→
   297→  return { settings, loaded, reload: load, setSettings };
   298→}

   468→// ── Save & test helpers ─────────────────────────────────────────────────────
   469→async function saveSetting(key: SettingKey, config: Record<string, unknown>, isActive: boolean) {
   470→  const res = await fetch("/api/settings", {
   471→    method: "POST",
   472→    headers: { "Content-Type": "application/json" },
   473→    body: JSON.stringify({ key, config, isActive }),
   474→  });
   475→  const data = await res.json();
   476→  if (!res.ok) throw new Error(data.error || `Failed to save ${key}`);
   477→  return data;
   478→}

   480→async function testChannel(channel: string, config: Record<string, unknown>, testTarget: string) {
   481→  const res = await fetch("/api/settings", {
   482→    method: "PATCH",
   483→    headers: { "Content-Type": "application/json" },
   484→    body: JSON.stringify({ action: "test", channel, config, testTarget }),
   485→  });
   486→  const data = await res.json();
   487→  if (!res.ok) {
   488→    throw new Error(data.message || data.error || "Test failed");
   489→  }
   490→  return data as { success?: boolean; message?: string; detail?: string };
