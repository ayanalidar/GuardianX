     1→// GuardianX Outbound Connectors - 20 additional integration targets.
     2→//
     3→// Each entry is a ConnectorSchema (see ./engine.ts) with a `send`
     4→// implementation that performs the actual HTTP call. These complement
     5→// the 9 built-in connectors in engine.ts.
     6→//
     7→// Connector categories follow the engine.ts ConnectorCategory union.
     8→// We deliberately DO NOT use any HTTP libraries beyond the global
     9→// `fetch` - each send() is a single awaitable POST/PUT with a short
    10→// timeout and structured error reporting.
    11→
    12→import type { ConnectorSchema, SecurityEvent, ConnectorSendResult } from "./engine";
    13→
    14→// Helper: convert a SecurityEvent into a generic alert payload that
    15→// most notification/ITSM systems can consume.
    16→function toAlertPayload(event: SecurityEvent): Record<string, unknown> {
    17→  return {
    18→    title: event.title,
    19→    description: event.description || event.title,
    20→    severity: event.severity,
    21→    type: event.type,
    22→    source: event.source,
    23→    occurredAt: event.occurredAt || new Date().toISOString(),
    24→    details: event.data || {},
    25→  };
    26→}
    27→
    28→// Severity color palette (hex without #) for chat-style connectors.
    29→function severityColor(sev: SecurityEvent["severity"]): string {
    30→  switch (sev) {
    31→    case "critical": return "#FF0000";
    32→    case "high": return "#FF8C00";
    33→    case "medium": return "#FFD700";
    34→    case "low": return "#00BFFF";
    35→    default: return "#36A64F";
    36→  }
    37→}

    54→    async send(config, event): Promise<ConnectorSendResult> {
    55→      const sid = String(config.accountSid || "");
    56→      const token = String(config.authToken || "");

    62→        Body: `[${event.severity.toUpperCase()}] ${event.title}\n\n${event.description || ""}`.slice(0, 1500),

    64→      const res = await fetch(url, {
    65→        method: "POST",
    66→        headers: {
    67→          Authorization: `Basic ${Buffer.from(`${sid}:${token}`).toString("base64")}`,
    68→          "Content-Type": "application/x-www-form-urlencoded",
    69→        },
    70→        body: body.toString(),
    71→        signal: AbortSignal.timeout(10_000),
    72→      });
    73→      const data = await res.json().catch(() => null);
    74→      return { ok: res.ok, status: res.status, externalId: data?.sid, detail: res.ok ? "sent" : `HTTP ${res.status}` };

   239→    icon: "Cloud",
   240→    configFields: [
   241→      { key: "workspaceId", label: "Workspace ID", type: "string", required: true },
   242→      { key: "sharedKey", label: "Primary Key", type: "password", required: true },
   243→      { key: "logType", label: "Log Type", type: "string", required: false, defaultValue: "GuardianXAlert" },
   244→    ],
   245→    async send(config, event): Promise<ConnectorSendResult> {
   246→      const wid = String(config.workspaceId || "");
   247→      const key = String(config.sharedKey || "");
   248→      if (!wid || !key) return { ok: false, detail: "Missing workspaceId or sharedKey" };
   249→      const body = JSON.stringify(toAlertPayload(event));
   250→      const { createHmac } = await import("node:crypto");
   251→      const resource = `/api/logs`;
   252→      const date = new Date().toUTCString();
   253→      const stringToSign = `POST\n${body.length}\napplication/json\nx-ms-date:${date}\n${resource}`;
   254→      const signature = createHmac("sha256", Buffer.from(key, "base64")).update(stringToSign).digest("base64");
   255→      const res = await fetch(`https://${wid}.ods.opinsights.azure.com${resource}?api-version=2016-04-01`, {
   256→        method: "POST",
   257→        headers: {
   258→          Authorization: `SharedKey ${wid}:${signature}`,
   259→          "Content-Type": "application/json",
   260→          "x-ms-date": date,
   261→          "Log-Type": String(config.logType || "GuardianXAlert"),
   262→        },
   263→        body,
   264→        signal: AbortSignal.timeout(10_000),
   265→      });
   266→      return { ok: res.ok, status: res.status, detail: res.ok ? "logged" : `HTTP ${res.status}` };
   267→    },
